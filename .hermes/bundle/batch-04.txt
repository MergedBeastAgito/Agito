//! The tokio implementation of the runtime's `Pace`.
//!
//! The runtime crate keeps its own tokio implementation behind a feature
//! nothing enables, so that a capability never inherits an executor. This is
//! where the executor actually lives: the adapter that drives a browser is the
//! one thing in the workspace that must have one, and the composition root
//! hands this pace to every capability it composes.

use std::time::{Duration, Instant};

use AgenticRuntime::{Clock, FutureBox, Pace};

/// The clock and the wait, on tokio.
#[derive(Debug, Default)]
pub struct Tokio;

impl Clock for Tokio {
	fn Now(&self) -> Instant { Instant::now() }
}

impl Pace for Tokio {
	fn Sleep(&self, Span:Duration) -> FutureBox<'_, ()> { Box::pin(tokio::time::sleep(Span)) }
}

//! The port `agentic.browse` promises, written the way a consumer asks for it.
//!
//! `module.json` names three errors - `browser-unavailable`,
//! `navigation-failed` and `timeout` - and [`crate::Struct::Error`] is where
//! they live. A caller narrows on the variant; a provider that fails in some
//! fourth way is mapped onto one of the three at the boundary, so a consumer
//! never has to guess.

use AgenticRuntime::{Call, FutureBox};
use serde_json::Value;

/// Who owns the page behind a target, and therefore who may close it.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub enum Ownership {
	/// Somebody else owns the browser: disposal only disconnects.
	Attached,
	/// This session started the browser and may close it.
	Launched,
}

impl Ownership {
	/// The word the port uses.
	pub const fn Name(self) -> &'static str {
		match self {
			Ownership::Attached => "attached",
			Ownership::Launched => "launched",
		}
	}
}

/// The handle a session is bound to.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Target {
	/// The page this session is bound to.
	pub Url:String,
	/// The endpoint the transport behind that page is connected to.
	pub Ws:String,
	/// Whether this session launched the browser or attached to one.
	pub Ownership:Ownership,
}

/// What every operation is bounded by, plus whether to attach.
#[derive(Clone, Debug)]
pub struct OpenOptions {
	/// The deadline and the caller's cancellation.
	pub Call:Call,
	/// Attach to a browser already listening here instead of launching one.
	pub AttachTo:Option<String>,
	/// Bind the page that browser already has, named by target id or by a URL
	/// prefix, instead of opening a new one.
	///
	/// Only meaningful with `AttachTo`: a browser this session launched has no
	/// page to bind, and asking for one is refused rather than silently
	/// opening a tab.
	pub Page:Option<String>,
}

impl OpenOptions {
	/// Open a new browser, bounded by `Call`.
	pub fn Launch(Call:Call) -> Self { Self { Call, AttachTo:None, Page:None } }

	/// Attach to the browser at `Endpoint` and open a new page in it, bounded
	/// by `Call`.
	pub fn Attach(Endpoint:impl Into<String>, Call:Call) -> Self {
		Self { Call, AttachTo:Some(Endpoint.into()), Page:None }
	}

	/// Attach to the browser at `Endpoint` and bind the page it already has.
	pub fn Existing(Endpoint:impl Into<String>, Page:impl Into<String>, Call:Call) -> Self {
		Self { Call, AttachTo:Some(Endpoint.into()), Page:Some(Page.into()) }
	}

	/// The ownership this option implies: attaching means somebody else owns
	/// it.
	pub fn Ownership(&self) -> Ownership {
		match self.AttachTo {
			None => Ownership::Launched,
			Some(_) => Ownership::Attached,
		}
	}

	/// The page this option asks a launcher for.
	pub fn Want<'a>(&'a self, Url:&'a str) -> crate::Struct::Transport::PageWant<'a> {
		match &self.Page {
			Some(Key) => crate::Struct::Transport::PageWant::Existing(Key),
			None => crate::Struct::Transport::PageWant::Fresh(Url),
		}
	}
}

/// The one port this crate offers.
///
/// `Dispose` returns `()` and cannot fail. A release that fails is a fact about
/// the browser, and a caller that already has its answer must not lose it to a
/// failing cleanup; the reason goes to the session's report, which is the
/// supervisor's record.
pub trait Browse: Send + Sync {
	/// Open a page, returning the handle the session is bound to.
	fn Open<'a>(
		&'a self,
		Url:&'a str,
		Options:OpenOptions,
	) -> FutureBox<'a, Result<Target, crate::Struct::Error::Error>>;
	/// Evaluate an expression in the page and return its value.
	fn Evaluate<'a>(
		&'a self,
		Target:&'a Target,
		Expression:&'a str,
		Call:Call,
	) -> FutureBox<'a, Result<Value, crate::Struct::Error::Error>>;
	/// Release everything this session holds. Never fails.
	fn Dispose<'a>(&'a self, Target:Target) -> FutureBox<'a, ()>;
}

//! `Browse` over a transport the composition root supplies.
//!
//! The session is the supervisor the port's rules talk about: it owns the
//! transport map, the pending operations and the cleanup. A caller's
//! cancellation stops *its wait* and nothing else - it never closes an attached
//! page - and disposal is the only thing that ends a transport's life.
//!
//! Every classification rule lives in [`crate::Fn::Classify`], in one place:
//!
//! - silence is a timeout. Nothing but the deadline can end a page that never
//!   answers, so a hang is reported as `timeout` and never as a dead transport;
//! - a closed transport is `browser-unavailable`, whatever error arrived with
//!   it, because the socket is gone rather than the page being slow;
//! - a live transport that could not be used is `navigation-failed`.

use std::{
	collections::{HashMap, HashSet},
	sync::{Arc, Mutex, MutexGuard},
	time::Duration,
};

use AgenticRuntime::{Bound, Call, Expired, FutureBox, Never, Pace};
use serde_json::Value;

use crate::{
	Fn::Classify::ClassifyPage,
	Struct::{
		Error::Error,
		Port::{Browse, OpenOptions, Ownership, Target},
		Transport::{Launcher, Transport},
	},
};

/// A release that has not finished in this long is reported, not waited on.
pub const DefaultDisposeBudget:Duration = Duration::from_secs(5);

/// Where a dispose failure goes, because it may not be returned.
pub type Report = Arc<dyn Fn(&Target, &Error) + Send + Sync>;

/// How one session is configured.
pub struct Options {
	/// Where a dispose failure goes.
	pub Report:Report,
	/// How long a release may take before it is reported as a failure.
	pub DisposeBudget:Duration,
}

impl Default for Options {
	fn default() -> Self {
		Self {
			// Silent by default: a library that writes to the process's stderr
			// behind a caller's back is a library that corrupts a protocol.
			Report:Arc::new(|_, _| {}),
			DisposeBudget:DefaultDisposeBudget,
		}
	}
}

/// A session is one `Browse` bound to one launcher.
pub struct Session {
	Launcher:Arc<dyn Launcher>,
	Pace:Arc<dyn Pace>,
	Options:Options,
	/// What this session holds, by endpoint.
	Held:Mutex<HashMap<String, Arc<dyn Transport>>>,
	/// What it has already released, so a second dispose is silent.
	Released:Mutex<HashSet<String>>,
}

/// Build a session over one launcher, on the pace the caller supplied.
pub fn Create(Launcher:Arc<dyn Launcher>, Pace:Arc<dyn Pace>, Options:Options) -> Session {
	Session {
		Launcher,
		Pace,
		Options,
		Held:Mutex::new(HashMap::new()),
		Released:Mutex::new(HashSet::new()),
	}
}

impl Session {
	fn Lock<T>(Guard:&Mutex<T>) -> MutexGuard<'_, T> { Guard.lock().unwrap_or_else(|Poison| Poison.into_inner()) }

	/// A reporter that throws must not turn dispose into a throw.
	fn ReportQuietly(&self, Target:&Target, Why:&Error) {
		let Report = self.Options.Report.clone();
		let _ = std::panic::catch_unwind(std::panic::AssertUnwindSafe(move || {
			Report(Target, Why);
		}));
	}

	/// Release one transport, bounded, and report why if it did not finish.
	async fn Release(&self, Transport:Arc<dyn Transport>, Ownership:Ownership) -> Option<Error> {
		let Deadline = self
			.Pace
			.Now()
			.checked_add(self.Options.DisposeBudget)
			.unwrap_or_else(|| self.Pace.Now());
		let Call = Call { Deadline, Cancellation:Arc::new(Never) };
		let Work = match Ownership {
			Ownership::Launched => Transport.Close(),
			Ownership::Attached => Transport.Detach(),
		};
		match Bound(Work, &Call, self.Pace.as_ref()).await {
			Ok(()) => None,
			Err(Why) => {
				Some(Error::Timeout(
					format!("dispose exceeded its {}ms budget", self.Options.DisposeBudget.as_millis()),
					Why.Name(),
				))
			},
		}
	}
}

impl Browse for Session {
	fn Open<'a>(&'a self, Url:&'a str, Options:OpenOptions) -> FutureBox<'a, Result<Target, Error>> {
		Box::pin(async move {
			let Ownership = Options.Ownership();
			let Phase = match Ownership {
				Ownership::Launched => "open (launch)",
				Ownership::Attached => "open (attach)",
			};
			// A caller that has already given up must not start a browser.
			if Options.Call.Cancelled() {
				return Err(Error::Timeout(
					format!("{Phase} was cancelled before it started"),
					"the signal was already fired",
				));
			}
			let Work = match &Options.AttachTo {
				None if Options.Page.is_some() => {
					return Err(Error::Unavailable(
						"a page can only be bound on a browser this session attached to",
						"the options asked for an existing page but named no endpoint",
					));
				},
				None => self.Launcher.Launch(Url, Options.Call.clone()),
				Some(Endpoint) => {
					self.Launcher
						.Attach(Endpoint, Options.Want(Url), Options.Call.clone())
				},
			};
			let Transport = match Bound(Work, &Options.Call, self.Pace.as_ref()).await {
				Ok(Ok(Transport)) => Transport,
				Ok(Err(Why)) => return Err(Why),
				Err(Expired::Timeout) => {
					return Err(Error::Timeout(
						format!("{Phase} exceeded its deadline"),
						"the browser never answered",
					));
				},
				Err(Expired::Cancelled) => {
					return Err(Error::Timeout(
						format!("{Phase} was cancelled by the caller"),
						"the signal fired while the browser was being opened",
					));
				},
			};
			let Target = Target {
				Url:Transport.Url().to_string(),
				Ws:Transport.Endpoint().to_string(),
				Ownership,
			};
			if Transport.Closed() {
				if let Some(Why) = self.Release(Transport.clone(), Ownership).await {
					self.ReportQuietly(&Target, &Why);
				}
				return Err(Error::Unavailable(
					"the transport was already closed when it arrived",
					format!("{Url} on {}", Target.Ws),
				));
			}
			Self::Lock(&self.Held).insert(Target.Ws.clone(), Transport);
			Self::Lock(&self.Released).remove(&Target.Ws);
			Ok(Target)
		})
	}

	fn Evaluate<'a>(&'a self, Target:&'a Target, Expression:&'a str, Call:Call) -> FutureBox<'a, Result<Value, Error>> {
		Box::pin(async move {
			let Transport = Self::Lock(&self.Held).get(&Target.Ws).cloned();
			let Some(Transport) = Transport else {
				return Err(Error::Unavailable(
					"this session holds no transport for the target",
					format!("{} was never opened here, or it was already disposed", Target.Ws),
				));
			};
			if Transport.Closed() {
				return Err(Error::Unavailable(
					"the transport is closed",
					format!("{} is no longer connected", Target.Ws),
				));
			}
			// A caller that has already given up must not ask the page
			// anything.
			if Call.Cancelled() {
				return Err(Error::Timeout(
					"evaluate was cancelled before it started",
					"the signal was already fired",
				));
			}
			let Work = Transport.Evaluate(Expression, Call.clone());
			match Bound(Work, &Call, self.Pace.as_ref()).await {
				Ok(Ok(Value)) => Ok(Value),
				Ok(Err(Why)) => Err(ClassifyPage(Why, Transport.Closed())),
				Err(Expired::Timeout) => {
					Err(Error::Timeout("evaluate exceeded its deadline", "the page never answered"))
				},
				Err(Expired::Cancelled) => {
					Err(Error::Timeout(
						"evaluate was cancelled by the caller",
						"the signal fired while the page was being asked",
					))
				},
			}
		})
	}

	fn Dispose<'a>(&'a self, Target:Target) -> FutureBox<'a, ()> {
		Box::pin(async move {
			// Never throws: not on a failed release, and not on a target this
			// session never opened.
			if !Self::Lock(&self.Released).insert(Target.Ws.clone()) {
				return;
			}
			let Transport = Self::Lock(&self.Held).remove(&Target.Ws);
			let Some(Transport) = Transport else {
				return;
			};
			if let Some(Why) = self.Release(Transport, Target.Ownership).await {
				self.ReportQuietly(&Target, &Why);
			}
		})
	}
}

--- FILE: Crate/BrowserCDP/Source/Struct/Transport.rs ---
//! The seam a provider implements, and the fake in `Test/` stands in for.
//!
//! Nothing in this file starts a browser. A [`Launcher`] is supplied by the
//! composition root - it may speak CDP over a WebSocket, drive a subprocess, or
//! answer from a fixture - and the session applies the port's rules to whatever
//! it returns.
//!
//! The contract on a provider:
//!
//! - a transport reports its own death through `Closed`, so the session can
//!   tell a dead socket from a page that is merely not ready;
//! - a failed `Attach`/`Launch` carries a [`crate::Struct::Error`] when it
//!   knows why (`Navigation` for a page that did not load); anything else is
//!   read as an unreachable browser;
//! - `Close` and `Detach` are idempotent: the session may call them on a target
//!   it has already released.

use std::sync::Arc;

use AgenticRuntime::{Call, FutureBox};
use serde_json::Value;

use crate::Struct::Error::Error;

/// One live page connection.
pub trait Transport: Send + Sync {
	/// The endpoint this transport is bound to; it becomes `Target::Ws`.
	fn Endpoint(&self) -> &str;
	/// The URL the page is actually on, which is not always the URL a caller
	/// asked for: a page bound by target id keeps whatever it was showing.
	fn Url(&self) -> &str;
	/// True once the connection is gone. A closed transport is dead, not slow.
	fn Closed(&self) -> bool;
	/// Ask the page one expression and return what it produced.
	fn Evaluate<'a>(&'a self, Expression:&'a str, Call:Call) -> FutureBox<'a, Result<Value, Error>>;
	/// Close the page this session launched.
	fn Close(&self) -> FutureBox<'_, ()>;
	/// Let go of a page somebody else owns, without closing it.
	fn Detach(&self) -> FutureBox<'_, ()>;
}

/// What an attach asks for: a page that already exists, or a new one.
///
/// Attaching to a browser and then opening a page in it is not the same
/// operation as binding the page a person is already looking at. The first
/// leaves a tab behind that nobody asked for; the second is the only way to act
/// "on this tab". So the difference is a value the caller states, not a
/// boolean a provider guesses.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum PageWant<'a> {
	/// A new page at this URL.
	Fresh(&'a str),
	/// The page already open whose target id, or failing that whose URL
	/// starts with this, is this.
	Existing(&'a str),
}

/// Where a transport comes from: an existing browser, or a new one.
pub trait Launcher: Send + Sync {
	/// Attach to a browser already listening at `Endpoint`, bound to `Want`.
	fn Attach<'a>(
		&'a self,
		Endpoint:&'a str,
		Want:PageWant<'a>,
		Call:Call,
	) -> FutureBox<'a, Result<Arc<dyn Transport>, Error>>;
	/// Start a browser and open `Url` in it.
	fn Launch<'a>(&'a self, Url:&'a str, Call:Call) -> FutureBox<'a, Result<Arc<dyn Transport>, Error>>;
}

//! `agentic.browse`, exercised against a fake launcher and a clock the test
//! owns.
//!
//! No browser, no network and no sleeping: a launcher that never answers is one
//! whose script says so, and a deadline passes because the test advanced its
//! own clock.

use std::{
	sync::{
		Arc,
		Mutex,
		atomic::{AtomicBool, AtomicU64, Ordering},
	},
	time::{Duration, Instant},
};

use AgenticBrowserCDP::{
	Browse,
	PageWant,
	Code,
	Create,
	Error,
	Launcher,
	OpenOptions,
	Options,
	Ownership,
	Report,
	Session,
	Target,
	Transport,
};
use AgenticRuntime::{BlockOn, Call, Clock, FutureBox, Never, Pace, Signal};
use serde_json::{Value, json};

fn Span(Milliseconds:u64) -> Duration { Duration::from_millis(Milliseconds) }

/// A clock the test owns, and a wait that advances it.
struct TestPace {
	Now:Mutex<Instant>,
	Fire:Mutex<Option<Arc<Signal>>>,
}

impl TestPace {
	fn New() -> Arc<Self> { Arc::new(Self { Now:Mutex::new(Instant::now()), Fire:Mutex::new(None) }) }

	fn Firing(self: &Arc<Self>, Signal:Arc<Signal>) -> Arc<Self> {
		*self.Fire.lock().unwrap() = Some(Signal);
		self.clone()
	}
}

impl Clock for TestPace {
	fn Now(&self) -> Instant { *self.Now.lock().unwrap() }
}

impl Pace for TestPace {
	fn Sleep(&self, Span:Duration) -> FutureBox<'_, ()> {
		// A pace that is about to fire models a caller giving up, not time
		// passing: it never finishes its wait, so the cancellation is
		// what ends the call.
		if let Some(Signal) = self.Fire.lock().unwrap().take() {
			*self.Now.lock().unwrap() += Duration::from_millis(1);
			Signal.Fire();
			return Box::pin(std::future::pending());
		}
		*self.Now.lock().unwrap() += Span.max(Duration::from_millis(1));
		Box::pin(std::future::ready(()))
	}
}

/// One live page, scripted.
struct FakeTransport {
	Endpoint:String,
	/// What the scripted page says it is on.
	Url:String,
	Closed:AtomicBool,
	Evaluations:AtomicU64,
	Closes:AtomicU64,
	Detaches:AtomicU64,
	Answer:Mutex<Result<Value, Error>>,
	Hang:bool,
	Dies:bool,
	Release:Release,
}

#[derive(PartialEq)]
enum Release {
	Quick,
	Hang,
}

impl FakeTransport {
	fn Base(Endpoint:&str) -> Self {
		Self {
			Endpoint:Endpoint.to_string(),
			Url:"https://example.test/".to_string(),
			Closed:AtomicBool::new(false),
			Evaluations:AtomicU64::new(0),
			Closes:AtomicU64::new(0),
			Detaches:AtomicU64::new(0),
			Answer:Mutex::new(Ok(json!(null))),
			Hang:false,
			Dies:false,
			Release:Release::Quick,
		}
	}

	fn New(Endpoint:&str) -> Arc<Self> { Arc::new(Self::Base(Endpoint)) }

	fn Answering(Endpoint:&str, Value:Value) -> Arc<Self> {
		let Transport = Self::New(Endpoint);
		*Transport.Answer.lock().unwrap() = Ok(Value);
		Transport
	}

	fn Failing(Endpoint:&str, Why:Error) -> Arc<Self> {
		let Transport = Self::New(Endpoint);
		*Transport.Answer.lock().unwrap() = Err(Why);
		Transport
	}

	fn Hanging(Endpoint:&str) -> Arc<Self> { Arc::new(Self { Hang:true, ..Self::Base(Endpoint) }) }

	fn Dead(Endpoint:&str) -> Arc<Self> {
		let Transport = Self::New(Endpoint);
		Transport.Closed.store(true, Ordering::SeqCst);
		Transport
	}

	/// A transport whose release never finishes.
	fn Stuck(Endpoint:&str) -> Arc<Self> { Arc::new(Self { Release:Release::Hang, ..Self::Base(Endpoint) }) }
}

impl Transport for FakeTransport {
	fn Endpoint(&self) -> &str { &self.Endpoint }

	fn Url(&self) -> &str { &self.Url }

	fn Closed(&self) -> bool { self.Closed.load(Ordering::SeqCst) }

	fn Evaluate<'a>(&'a self, _Expression:&'a str, _Call:Call) -> FutureBox<'a, Result<Value, Error>> {
		self.Evaluations.fetch_add(1, Ordering::SeqCst);
		let Hang = self.Hang;
		if self.Dies {
			self.Closed.store(true, Ordering::SeqCst);
		}
		let Answer = self.Answer.lock().unwrap().clone();
		Box::pin(async move {
			if Hang {
				std::future::pending::<()>().await;
			}
			Answer
		})
	}

	fn Close(&self) -> FutureBox<'_, ()> {
		self.Closes.fetch_add(1, Ordering::SeqCst);
		let Hang = self.Release == Release::Hang;
		Box::pin(async move {
			if Hang {
				std::future::pending::<()>().await;
			}
			self.Closed.store(true, Ordering::SeqCst);
		})
	}

	fn Detach(&self) -> FutureBox<'_, ()> {
		self.Detaches.fetch_add(1, Ordering::SeqCst);
		let Hang = self.Release == Release::Hang;
		Box::pin(async move {
			if Hang {
				std::future::pending::<()>().await;
			}
			self.Closed.store(true, Ordering::SeqCst);
		})
	}
}

/// A launcher that hands out one scripted transport, or one scripted failure.
struct FakeLauncher {
	Hands:Mutex<Option<Arc<FakeTransport>>>,
	Failure:Mutex<Option<Error>>,
	Hang:bool,
	Launches:AtomicU64,
	Attaches:AtomicU64,
}

impl FakeLauncher {
	fn Base(Transport:Arc<FakeTransport>) -> Self {
		Self {
			Hands:Mutex::new(Some(Transport)),
			Failure:Mutex::new(None),
			Hang:false,
			Launches:AtomicU64::new(0),
			Attaches:AtomicU64::new(0),
		}
	}

	fn Handing(Transport:Arc<FakeTransport>) -> Arc<Self> { Arc::new(Self::Base(Transport)) }

	fn Failing(Why:Error) -> Arc<Self> {
		let Launcher = Self::Handing(FakeTransport::New("ws://none"));
		*Launcher.Failure.lock().unwrap() = Some(Why);
		Launcher
	}

	/// A launcher that never answers.
	fn Hanging() -> Arc<Self> { Arc::new(Self { Hang:true, ..Self::Base(FakeTransport::New("ws://none")) }) }

	fn Answer(&self) -> FutureBox<'_, Result<Arc<dyn Transport>, Error>> {
		let Hand = self.Hands.lock().unwrap().clone();
		let Failure = self.Failure.lock().unwrap().clone();
		let Hang = self.Hang;
		Box::pin(async move {
			if Hang {
				std::future::pending::<()>().await;
			}
			if let Some(Why) = Failure {
				return Err(Why);
			}
			match Hand {
				Some(Transport) => Ok(Transport as Arc<dyn Transport>),
				None => Err(Error::Unavailable("nothing to hand out", "")),
			}
		})
	}
}

impl Launcher for FakeLauncher {
	fn Attach<'a>(
		&'a self,
		_Endpoint:&'a str,
		_Want:PageWant<'a>,
		_Call:Call,
	) -> FutureBox<'a, Result<Arc<dyn Transport>, Error>> {
		self.Attaches.fetch_add(1, Ordering::SeqCst);
		self.Answer()
	}

	fn Launch<'a>(&'a self, _Url:&'a str, _Call:Call) -> FutureBox<'a, Result<Arc<dyn Transport>, Error>> {
		self.Launches.fetch_add(1, Ordering::SeqCst);
		self.Answer()
	}
}

//! The Rust side of the differential run for `agentic.browse`.
//!
//! These cases are not written here: they are read from
//! `scenarios/browser-cdp.json` and `scenarios/generated/browser-cdp.timing.json`,
//! the same files `Package/browser-cdp/Test/Parity.test.ts` reads. What this
//! file adds is only the fake, the interpreter and the assertion - the fixture,
//! the steps, the expected result and the expected ordered effect list all come
//! from the scenario files, so the two implementations cannot drift by being
//! tested against two hand-written suites.
//!
//! `effects` is the protocol the implementation spoke: every launch, attach,
//! evaluation and release the fake observed, in order. Two ports that returned
//! the same value by different routes disagree here, which is the whole point.
//!
//! `Test/Browse.rs` is kept as it is: that suite is the contract, this one is
//! the comparison. `AGENTIC_PARITY_OUT=<path>` makes this test also write what
//! it observed as JSON, which is how `scenarios/run.mjs --compare` gets the Rust
//! half of the differential run.

#![allow(non_snake_case, non_upper_case_globals)]

use std::{
	collections::BTreeMap,
	sync::{
		Arc, Mutex,
		atomic::{AtomicBool, AtomicU64, Ordering},
	},
	time::{Duration, Instant},
};

use AgenticBrowserCDP::{
	Browse, Create, Error, Launcher, OpenOptions, Options, Ownership, PageWant, Report, Target, Transport,
};
use AgenticRuntime::{BlockOn, Call, Cancellation, Clock, FutureBox, Never, Pace, Signal};
use serde_json::{Map, Value, json};

/// Where every observation the fake made is written, in order.
type Effects = Arc<Mutex<Vec<String>>>;

/// One scripted page connection.
struct FakeTransport {
	Endpoint: String,
	/// Where the scripted page says it is.
	Url: String,
	Closed: AtomicBool,
	/// `answer`, `hang`, `throw` or `dead`.
	Behaviour: Mutex<String>,
	Answer: Mutex<Value>,
	HangOnClose: AtomicBool,
	Closes: AtomicU64,
	Detaches: AtomicU64,
	Asked: Mutex<Vec<String>>,
	Effects: Effects,
}

impl FakeTransport {
	fn New(Fixture: &Value, Effects: Effects) -> Self {
		let Endpoint = string_at(Fixture, &["endpoint"]).unwrap_or_else(|| "ws://none".to_string());
		let State = Fixture.get("transport").cloned().unwrap_or(Value::Null);
		Self {
			Endpoint,
			Url: "https://example.test/".to_string(),
			Closed: AtomicBool::new(bool_at(&State, &["closed"]).unwrap_or(false)),
			Behaviour: Mutex::new(string_at(&State, &["behaviour"]).unwrap_or_else(|| "answer".to_string())),
			Answer: Mutex::new(State.get("answer").cloned().unwrap_or(Value::Null)),
			HangOnClose: AtomicBool::new(bool_at(&State, &["hangOnClose"]).unwrap_or(false)),
			Closes: AtomicU64::new(0),
			Detaches: AtomicU64::new(0),
			Asked: Mutex::new(Vec::new()),
			Effects,
		}
	}

	/// Apply the `before` patch of one step.
	fn Patch(&self, Values: &Value) {
		if let Some(Behaviour) = Values.get("behaviour").and_then(Value::as_str) {
			*self.Behaviour.lock().unwrap() = Behaviour.to_string();
		}
		if let Some(Answer) = Values.get("answer") {
			*self.Answer.lock().unwrap() = Answer.clone();
		}
		if let Some(Closed) = Values.get("closed").and_then(Value::as_bool) {
			self.Closed.store(Closed, Ordering::SeqCst);
		}
		if let Some(Hang) = Values.get("hangOnClose").and_then(Value::as_bool) {
			self.HangOnClose.store(Hang, Ordering::SeqCst);
		}
	}

	fn Note(&self, Effect: String) { self.Effects.lock().unwrap().push(Effect); }
}

impl Transport for FakeTransport {
	fn Endpoint(&self) -> &str { &self.Endpoint }

	fn Url(&self) -> &str { &self.Url }

	fn Closed(&self) -> bool { self.Closed.load(Ordering::SeqCst) }

	fn Evaluate<'a>(&'a self, Expression: &'a str, _Call: Call) -> FutureBox<'a, Result<Value, Error>> {
		self.Note(format!("evaluate:{Expression}"));
		self.Asked.lock().unwrap().push(Expression.to_string());
		let Behaviour = self.Behaviour.lock().unwrap().clone();
		let Endpoint = self.Endpoint.clone();
		if Behaviour == "dead" || self.Closed() {
			// The socket died under the call: the fake reports the state as well
			// as the error, so the session can classify it as a dead transport.
			self.Closed.store(true, Ordering::SeqCst);
			self.Note(format!("refused:evaluate:{Endpoint}"));
			return Box::pin(async move {
				Err(Error::Navigation(
					"WebSocket is already in CLOSING or CLOSED state",
					"the socket died under the call",
				))
			});
		}
		if Behaviour == "throw" {
			return Box::pin(async move {
				Err(Error::Navigation("the page could not answer", "ReferenceError: nothing is defined"))
			});
		}
		if Behaviour == "hang" {
			return Box::pin(std::future::pending::<Result<Value, Error>>());
		}
		let Answer = self.Answer.lock().unwrap().clone();
		Box::pin(async move { Ok(Answer) })
	}

	// `Close` and `Detach` answer `()`, so a release cannot fail - only a hang is
	// reportable, which is why the scenario file marks the one case that needs a
	// *rejecting* release as unported on this side.
	fn Close(&self) -> FutureBox<'_, ()> {
		self.Note(format!("close:{}", self.Endpoint));
		self.Closes.fetch_add(1, Ordering::SeqCst);
		if self.HangOnClose.load(Ordering::SeqCst) {
			return Box::pin(std::future::pending());
		}
		self.Closed.store(true, Ordering::SeqCst);
		Box::pin(std::future::ready(()))
	}

	fn Detach(&self) -> FutureBox<'_, ()> {
		self.Note(format!("detach:{}", self.Endpoint));
		self.Detaches.fetch_add(1, Ordering::SeqCst);
		self.Closed.store(true, Ordering::SeqCst);
		Box::pin(std::future::ready(()))
	}
}

/// A launcher that hands out one scripted transport, or one scripted failure.
struct FakeLauncher {
	Transport: Arc<FakeTransport>,
	/// `answer`, `hang` or `throw`.
	Open: Mutex<String>,
	Failure: Mutex<Option<Value>>,
	DeadOnArrival: AtomicBool,
	Launches: AtomicU64,
	Attaches: AtomicU64,
	Effects: Effects,
}

impl FakeLauncher {
	fn New(Fixture: &Value, Transport: Arc<FakeTransport>, Effects: Effects) -> Self {
		let State = Fixture.get("launcher").cloned().unwrap_or(Value::Null);
		Self {
			Transport,
			Open: Mutex::new(string_at(&State, &["open"]).unwrap_or_else(|| "answer".to_string())),
			Failure: Mutex::new(State.get("failure").cloned().filter(|Value| !Value.is_null())),
			DeadOnArrival: AtomicBool::new(bool_at(&State, &["deadOnArrival"]).unwrap_or(false)),
			Launches: AtomicU64::new(0),
			Attaches: AtomicU64::new(0),
			Effects,
		}
	}

	fn Patch(&self, Values: &Value) {
		if let Some(Open) = Values.get("open").and_then(Value::as_str) {
			*self.Open.lock().unwrap() = Open.to_string();
		}
		if let Some(Failure) = Values.get("failure") {
			*self.Failure.lock().unwrap() = if Failure.is_null() { None } else { Some(Failure.clone()) };
		}
		if let Some(Dead) = Values.get("deadOnArrival").and_then(Value::as_bool) {
			self.DeadOnArrival.store(Dead, Ordering::SeqCst);
		}
	}

	fn Answer(&self) -> FutureBox<'_, Result<Arc<dyn Transport>, Error>> {
		let Open = self.Open.lock().unwrap().clone();
		if Open == "hang" {
			return Box::pin(std::future::pending::<Result<Arc<dyn Transport>, Error>>());
		}
		if Open == "throw" {
			let Failure = self.Failure.lock().unwrap().clone().unwrap_or(Value::Null);
			let Why = failure_of(&Failure);
			return Box::pin(async move { Err(Why) });
		}
		if self.DeadOnArrival.load(Ordering::SeqCst) {
			self.Transport.Closed.store(true, Ordering::SeqCst);
		}
		self.Effects.lock().unwrap().push(format!("handed:{}", self.Transport.Endpoint));
		let Transport: Arc<dyn Transport> = self.Transport.clone();
		Box::pin(async move { Ok(Transport) })
	}
}

impl Launcher for FakeLauncher {
	fn Attach<'a>(
		&'a self,
		Endpoint: &'a str,
		_Want: PageWant<'a>,
		_Call: Call,
	) -> FutureBox<'a, Result<Arc<dyn Transport>, Error>> {
		self.Effects.lock().unwrap().push(format!("attach:{Endpoint}"));
		self.Attaches.fetch_add(1, Ordering::SeqCst);
		self.Answer()
	}

	fn Launch<'a>(&'a self, Url: &'a str, _Call: Call) -> FutureBox<'a, Result<Arc<dyn Transport>, Error>> {
		self.Effects.lock().unwrap().push(format!("launch:{Url}"));
		self.Launches.fetch_add(1, Ordering::SeqCst);
		self.Answer()
	}
}

/// The error a scripted launcher failure describes.
fn failure_of(Failure: &Value) -> Error {
	let Message = string_at(Failure, &["message"]).unwrap_or_else(|| "the browser failed".to_string());
	let Detail = string_at(Failure, &["detail"]).unwrap_or_default();
	match string_at(Failure, &["code"]).unwrap_or_else(|| "unavailable".to_string()).as_str() {
		"navigation" => Error::Navigation(Message, Detail),
		"timeout" => Error::Timeout(Message, Detail),
		_ => Error::Unavailable(Message, Detail),
	}
}

/// A clock the test owns, and a wait that advances it.
///
/// A pace that is about to fire models a caller giving up, not time passing: it
/// never finishes its wait, so the cancellation is what ends the call. That is
/// what makes `signal: "after:N"` deterministic without a real timer.
struct TestPace {
	Now: Mutex<Instant>,
	Fire: Mutex<Option<Arc<Signal>>>,
}

impl TestPace {
	fn New() -> Arc<Self> { Arc::new(Self { Now: Mutex::new(Instant::now()), Fire: Mutex::new(None) }) }

	fn Firing(self: &Arc<Self>, Signal: Arc<Signal>) -> Arc<Self> {
		*self.Fire.lock().unwrap() = Some(Signal);
		self.clone()
	}
}

impl Clock for TestPace {
	fn Now(&self) -> Instant { *self.Now.lock().unwrap() }
}

impl Pace for TestPace {
	fn Sleep(&self, Span: Duration) -> FutureBox<'_, ()> {
		if let Some(Signal) = self.Fire.lock().unwrap().take() {
			*self.Now.lock().unwrap() += Duration::from_millis(1);
			Signal.Fire();
			return Box::pin(std::future::pending());
		}
		*self.Now.lock().unwrap() += Span.max(Duration::from_millis(1));
		Box::pin(std::future::ready(()))
	}
}

/// Read a dotted path out of a JSON value.
fn at<'a>(Document: &'a Value, Path: &[&str]) -> Option<&'a Value> {
	let mut Cursor = Document;
	for Step in Path {
		Cursor = Cursor.get(*Step)?;
	}
	Some(Cursor)
}

fn string_at(Document: &Value, Path: &[&str]) -> Option<String> {
	at(Document, Path).and_then(Value::as_str).map(str::to_string)
}

fn bool_at(Document: &Value, Path: &[&str]) -> Option<bool> { at(Document, Path).and_then(Value::as_bool) }

fn number_at(Document: &Value, Path: &[&str]) -> Option<u64> { at(Document, Path).and_then(Value::as_u64) }

/// Merge a patch into a fixture, one level deep per section, recursively. A
/// missing patch is not a patch: it leaves the fixture alone.
fn merge(Base: &Value, Patch: &Value) -> Value {
	if Patch.is_null() {
		return Base.clone();
	}
	match (Base, Patch) {
		(Value::Object(Base), Value::Object(Patch)) => {
			let mut Merged = Base.clone();
			for (Key, Patched) in Patch {
				let Next = match Merged.get(Key) {
					Some(Existing) => merge(Existing, Patched),
					None => Patched.clone(),
				};
				Merged.insert(Key.clone(), Next);
			}
			Value::Object(Merged)
		}
		_ => Patch.clone(),
	}
}

/// The scenario files this crate answers to.
fn scenario_files() -> Vec<Value> {
	let Base = std::path::Path::new(env!("CARGO_MANIFEST_DIR")).join("../../scenarios");
	["browser-cdp.json", "generated/browser-cdp.timing.json"]
		.iter()
		.map(|Name| Base.join(Name))
		.filter(|Path| Path.exists())
		.filter_map(|Path| std::fs::read_to_string(Path).ok())
		.filter_map(|Text| serde_json::from_str::<Value>(&Text).ok())
		.collect()
}

/// A case this side cannot express is skipped, and the file says why.

		.get("skip")
		.and_then(Value::as_array)
		.map(|Sides| Sides.iter().any(|Side| Side.as_str() == Some("rust")))
		.unwrap_or(false)

/// One step's bound: its budget, and whatever cancellation the step names.
fn call_for(Step: &Value, Pace: &Arc<TestPace>, Cancellation: Arc<dyn Cancellation>) -> Call {
	let Budget = Duration::from_millis(number_at(Step, &["timeoutMs"]).unwrap_or(40));
	Call::Within(Budget, Pace.as_ref(), Cancellation)
}

fn target_of(Document: &Value) -> Target {

/// Run one scenario, and answer what it did: the result, the ordered effects and
/// how many failures were reported.
fn run_case(Fixture: &Value, Entry: &Value) -> Value {
	let Fixture = merge(Fixture, Entry.get("fixture").unwrap_or(&Value::Null));
	let Effects: Effects = Arc::new(Mutex::new(Vec::new()));
	let Reports = Arc::new(AtomicU64::new(0));
	let ReportMode = Arc::new(Mutex::new(
		string_at(&Fixture, &["session", "report"]).unwrap_or_else(|| "record".to_string()),
	));
	let DisposeBudget = Duration::from_millis(number_at(&Fixture, &["session", "disposeTimeoutMs"]).unwrap_or(40));

	let Transport = Arc::new(FakeTransport::New(&Fixture, Effects.clone()));
	let Launcher = Arc::new(FakeLauncher::New(&Fixture, Transport.clone(), Effects.clone()));

	let Steps = Entry.get("steps").and_then(Value::as_array).cloned().unwrap_or_default();
	// A step that asks to be cancelled mid-operation needs a pace that fires, and
	// the pace fires on its first wait - which is that operation's guard, because
	// every step before it answers without waiting.
	let Wanted = Steps
		.iter()
		.filter_map(|Step| Step.get("signal").and_then(Value::as_str))
		.find(|Kind| Kind.starts_with("after:") || *Kind == "aborted");
	let Shared = Signal::New();
	let Pace = match Wanted {
		Some("aborted") => {
			Shared.Fire();
			TestPace::New()
		}
		Some(_) => TestPace::New().Firing(Shared.clone()),
		None => TestPace::New(),
	};

	let Report: Report = {
		let Effects = Effects.clone();
		let Reports = Reports.clone();
		let Mode = ReportMode.clone();
		Arc::new(move |Target: &Target, _Why: &Error| {
			Effects.lock().unwrap().push(format!("report:{}", Target.Ws));
			Reports.fetch_add(1, Ordering::SeqCst);
			if Mode.lock().unwrap().as_str() == "throw" {
				panic!("the reporter is broken too");
			}
		})
	};

	let Session = Create(Launcher.clone(), Pace.clone(), Options { Report, DisposeBudget });

	let mut Held: Option<Target> = None;
	let mut Result = Value::Null;

	for Step in &Steps {
		if let Some(Before) = Step.get("before") {
			if let Some(Values) = Before.get("transport") {
				Transport.Patch(Values);
			}
			if let Some(Values) = Before.get("launcher") {
				Launcher.Patch(Values);
			}
			if let Some(Values) = Before.get("session") {
				if let Some(Mode) = Values.get("report").and_then(Value::as_str) {
					*ReportMode.lock().unwrap() = Mode.to_string();
				}
			}
		}

		let Cancellation: Arc<dyn Cancellation> = match Step.get("signal").and_then(Value::as_str) {
			Some(Kind) if Kind == "aborted" || Kind.starts_with("after:") => Shared.clone(),
			_ => Arc::new(Never),
		};

		match string_at(Step, &["op"]).unwrap_or_default().as_str() {
			"open" => {
				let Url = string_at(Step, &["url"]).unwrap_or_default();
				let Bound = call_for(Step, &Pace, Cancellation);
				let Options = match string_at(Step, &["attachTo"]) {
					Some(Endpoint) => OpenOptions::Attach(Endpoint, Bound),
					None => OpenOptions::Launch(Bound),
				};
				match BlockOn(Session.Open(&Url, Options)) {
					Ok(Target) => {
						Result = json!({
							"kind": "target",
							"ownership": Target.Ownership.Name(),
							"url": Target.Url,
							"ws": Target.Ws,
						});
						Held = Some(Target);
					}
					Err(Why) => {
						Result = error_of(&Why);
						break;
					}
				}
			}
			"evaluate" => {
				let One = match Step.get("target") {
					Some(Value::Object(_)) => Some(target_of(Step.get("target").unwrap_or(&Value::Null))),
					_ => Held.clone(),
				};
				let Some(One) = One else {
					Result = json!({ "kind": "error", "code": "browser-unavailable", "message": "no target" });
					break;
				};
				let Expression = string_at(Step, &["expression"]).unwrap_or_default();
				let Bound = call_for(Step, &Pace, Cancellation);
				match BlockOn(Session.Evaluate(&One, &Expression, Bound)) {
					Ok(Produced) => Result = json!({ "kind": "value", "value": Produced }),
					Err(Why) => {
						Result = error_of(&Why);
						break;
					}
				}
			}
			"dispose" => {
				let One = match Step.get("target") {
					Some(Value::Object(_)) => Some(target_of(Step.get("target").unwrap_or(&Value::Null))),
					_ => Held.clone(),
				};
				let Some(One) = One else {
					Result = json!({ "kind": "void" });
					continue;
				};
				for _ in 0..number_at(Step, &["times"]).unwrap_or(1) {
					BlockOn(Session.Dispose(One.clone()));
				}
				Result = json!({ "kind": "void" });
			}
			Other => panic!("unknown browse step {Other}"),
		}
	}

	json!({
		"result": Result,
		"effects": Effects.lock().unwrap().clone(),
		"reports": Reports.load(Ordering::SeqCst),
	})
}

/// What one observation got wrong, as sentences. Empty means the case agreed.
fn mismatches(Observed: &Value, Expect: &Value) -> Vec<String> {
	let mut Wrong = Vec::new();
	let Want = Expect.get("result").cloned().unwrap_or(Value::Null);
	let Got = Observed.get("result").cloned().unwrap_or(Value::Null);
	let Kind = |Document: &Value| Document.get("kind").and_then(Value::as_str).unwrap_or("").to_string();
	if Kind(&Want) != Kind(&Got) {
		Wrong.push(format!("result kind: wanted {}, got {}", Kind(&Want), Kind(&Got)));
	} else if Kind(&Want) == "error" {
		if Want.get("code") != Got.get("code") {
			Wrong.push(format!("result code: wanted {:?}, got {:?}", Want.get("code"), Got.get("code")));
		}
		if let Some(Message) = Want.get("message").and_then(Value::as_str) {
			if !Got.get("message").and_then(Value::as_str).unwrap_or("").contains(Message) {
				Wrong.push(format!("result message: wanted to contain {Message:?}, got {:?}", Got.get("message")));
			}
		}
		if let Some(Token) = Want.get("observed").and_then(Value::as_str) {
			let Both = format!(
				"{} {}",
				Got.get("message").and_then(Value::as_str).unwrap_or(""),
				Got.get("detail").and_then(Value::as_str).unwrap_or("")
			);
			if !Both.contains(Token) {
				Wrong.push(format!("result observation: wanted {Token:?} in message or detail, got {Both:?}"));
			}
		}
	} else if Kind(&Want) == "target" {
		for Field in ["ownership", "url", "ws"] {
			if Want.get(Field) != Got.get(Field) {
				Wrong.push(format!("result {Field}: wanted {:?}, got {:?}", Want.get(Field), Got.get(Field)));
			}
		}
	} else if Kind(&Want) == "value" && Want.get("value") != Got.get("value") {
		Wrong.push(format!("result value: wanted {:?}, got {:?}", Want.get("value"), Got.get("value")));
	}

	if let Some(Effects) = Expect.get("effects") {
		if Effects != Observed.get("effects").unwrap_or(&Value::Null) {
			Wrong.push(format!("effects: wanted {Effects}, got {}", Observed.get("effects").unwrap_or(&Value::Null)));
		}
	}
	if let Some(Reports) = Expect.get("reports").and_then(Value::as_u64) {
		if Some(Reports) != Observed.get("reports").and_then(Value::as_u64) {
			Wrong.push(format!("reports: wanted {Reports}, got {:?}", Observed.get("reports")));
		}
	}
	Wrong
}

/// Every scenario, against its expectation, with what was observed left behind
/// for the differential run to compare against the TypeScript half.
#[test]
fn every_scenario_agrees_with_its_expectation() {
	let mut Observed: BTreeMap<String, Value> = BTreeMap::new();
	let mut Wrong: Vec<String> = Vec::new();
	let mut Seen = 0usize;

	for File in scenario_files() {
		let Fixture = File.get("fixture").cloned().unwrap_or(Value::Null);
		let Cases = File.get("cases").and_then(Value::as_array).cloned().unwrap_or_default();
		for Entry in Cases {
			if skipped_on_rust(&Entry) {
				continue;
			}
			let Id = string_at(&Entry, &["id"]).unwrap_or_else(|| "<unnamed>".to_string());
			let One = run_case(&Fixture, &Entry);
			let Expect = Entry.get("expect").cloned().unwrap_or(Value::Null);
			for Sentence in mismatches(&One, &Expect) {
				Wrong.push(format!("{Id}: {Sentence}"));
			}
			Observed.insert(Id, One);
			Seen += 1;
		}
	}

	if let Ok(Path) = std::env::var("AGENTIC_PARITY_OUT") {
		let Document =
			serde_json::to_string_pretty(&Value::Object(Observed.clone().into_iter().collect::<Map<String, Value>>()))
				.unwrap_or_default();
		let _ = std::fs::write(Path, Document);
	}

	assert!(Seen > 0, "no scenario files were found");
	assert!(
		Wrong.is_empty(),
		"the Rust port disagrees with the scenario files:\n{}",
		Wrong.join("\n")
	);
}

//! The expressions the driver sends to a page.
//!
//! These are the whole protocol between the capability and the browser. They
//! are deliberately small and literal: each one reads one thing, returns a
//! plain value, and touches nothing but `document`. That is what lets the same
//! driver run against a real tab and against a fake, which answers by
//! recognising the expression it was handed rather than by matching a string
//! this crate happened to send.
//!
//! Two conventions hold across all of them:
//!
//! - a value the page does not have is `null`, never a guess. A missing toggle
//!   is not "search", a missing label is not the model the caller asked for,
//!   and a missing answer is not an empty answer;
//! - the text they return is trimmed, because a page's own whitespace is not
//!   part of what it said, and an answer that differs only in trailing spaces
//!   has not changed.
//!
//! # Where the JavaScript is
//!
//! Not in this file. Each expression is one file under `Source/Script/`, taken
//! whole by `include_str!` and sent as it stands, so an editor, a linter and a
//! compiler can see it - which a `format!` never let them do. A value the caller
//! supplies is written where it belongs as a `{{name}}` placeholder and filled
//! by `AgenticRuntime::Fill`; the parameters a function declares are the
//! parameters its file names, and the test at the bottom of this module holds
//! that agreement to the bytes.
//!
//! What did not move: the functions, their names and their signatures. This
//! module is still the list of what the driver can say, and the script behind
//! each one is still one expression that reads one thing.

use AgenticRuntime::Fill;

use crate::Struct::{Format::Format, Selectors::Selectors};

/// A string as a JavaScript literal, so a selector cannot end the expression.
fn Literal(Value:&str) -> String { serde_json::to_string(Value).unwrap_or_else(|_| "\"\"".to_string()) }

/// How many answers the thread holds, the text of the newest one, and whether
/// the page says it is still producing it.
///
/// `busy` is read here rather than in a probe of its own because it is a fact
/// about the same moment as the text: a separate read would describe a
/// different instant, and the two would disagree exactly when it matters.
///
/// It has two halves, and both are the page's own statements. A visible stop
/// control means the page is working now. A turn whose text does not yet read
/// as finished means the page has not said it is done - which covers the window
/// before a stop control appears, where a turn is demonstrably not finished and
/// nothing on the page says so.
///
/// `Format` decides what one answer reads as, and it is a parameter rather than
/// a second function because the rest of the read is identical either way: the
/// same element, the same scroll, the same busy control, the same marks. Only
/// the closure that turns an element into text changes, which is why the
/// observation stream and the completion rule cannot tell the two apart - they
/// compare text to text, not text to a format.
pub fn ReadAnswers(Selectors:&Selectors, Format:Format) -> String {
	// An empty `done` is not a pattern that matches nothing: it is the caller
	// saying the product has no terminal marker, which turns the check off
	// rather than making every turn look unfinished.
	let Pattern = if Selectors.Done.trim().is_empty() {
		"null".to_string()
	} else {
		format!("new RegExp({})", Literal(&Selectors.Done))
	};
	// The serializer is spliced in only when it is asked for: it is thirteen
	// kilobytes the page would parse on every read of the thread, and the
	// recorded baselines were read without it.
	let Serialise = match Format {
		Format::Text => "",
		Format::Markdown => SerialiseScript,
	};
	// What one answer reads as. The same closure produces the newest answer's
	// text and every answer's mark, so the driver's own mark and the page's
	// cannot be computed from two different readings of the same turn.
	let Seen = match Format {
		Format::Text => "(Answer) => (Answer.innerText || \"\").trim()",
		Format::Markdown => "(Answer) => Serialise(Answer)",
	};
	Fill(
		ReadAnswersScript,
		&[
			("serialise", Serialise),
			("scroll", &Literal(&Selectors.Scroll)),
			("answer", &Literal(&Selectors.Answer)),
			("busy", &Literal(&Selectors.Busy)),
			("pattern", &Pattern),
			("turn", &Literal(&Selectors.Turn)),
			("seen", Seen),
		],
	)
}

///

			Done:"^\\s*done\\s*$".to_string(),
			Turn:"[data-testid='turn']".to_string(),
			..DefaultSelectors()
		};
		let Question = "a question";
		vec![
			("ReadMode", ReadMode(&Selectors)),
			("PressMode", PressMode(&Selectors)),
			("ReadModel", ReadModel(&Selectors)),
			("TypeQuestion", TypeQuestion(&Selectors, Question)),
			("PressSubmit", PressSubmit(&Selectors, Question)),
			("ReadAnswers/text", ReadAnswers(&Selectors, Format::Text)),
			("ReadAnswers/markdown", ReadAnswers(&Selectors, Format::Markdown)),
			("ReadCollapse", ReadCollapse(&Selectors)),
			("ReadTrace", ReadTrace(&Selectors)),
			("ReadSources", ReadSources(&Selectors)),
		]
	}

	/// A script is filled completely, and the check that says so bites.
	///
	/// The second half is the one that matters: `Unfilled` answering `None` for
	/// every expression proves nothing unless it answers `Some` for a string that
	/// still carries a placeholder, and the raw script is that string.
	#[test]
	fn every_expression_is_filled_and_no_script_is_left_with_a_placeholder() {
		for (Name, Raw, Parameters) in Scripts {
			for Parameter in Parameters {
				assert!(
					Raw.contains(&format!("{{{{{Parameter}}}}}")),
					"{Name} does not name {{{{{Parameter}}}}}, so the value would go nowhere"
				);
			}
			assert_eq!(
				Unfilled(Raw).as_deref(),
				Some(Parameters[0]),
				"{Name}'s script carries no placeholder, so nothing proves the fill ran"
			);
		}

		for (Name, Text) in Built() {
			assert_eq!(Unfilled(&Text), None, "{Name} left a placeholder unsubstituted in the expression it sends");
		}
	}
}

//! Turning what a page answered into the port's types, or into `None`.
//!
//! Every value the page returns crosses the port boundary as a
//! [`serde_json::Value`]: the expressions are the driver's own, but the provider is
//! not, and a provider that answers a number where a label was asked for must
//! not be able to make `ask` fail outside the four codes the manifest declares.
//! So a value is read as text when it is text and rendered as text when it is
//! not - the rules then refuse it on its merits, with a code the manifest
//! actually has.

use serde_json::Value;

use crate::Struct::Port::{Source, Step};

/// One read of the thread.
///
/// A thread with no answers is `{ Count: 0, Text: "", Busy: false }`, so "there
/// is no answer yet" and "the page did not answer the question I asked" are
/// different values. Folding them together would let a provider's wrong shape
/// look like an empty thread and cost the caller its whole deadline before
/// `not-ready`.
#[derive(Clone, Debug, Default, PartialEq)]
pub struct Answers {
	/// How many answers the thread holds.
	pub Count:f64,
	/// The last answer's text, trimmed; empty when there is none.
	pub Text:String,
	/// Whether the page says it is still producing the answer.
	///
	/// This is the page's own statement, not an inference from its text: a
	/// pause between tokens, a reasoning step and a slow code block all leave
	/// the text unchanged while the answer is demonstrably unfinished. A page
	/// that offers no such signal answers `false`, which is the honest reading
	/// ("nothing says it is still working") and leaves the text rule in charge.
	pub Busy:bool,
	/// One identifier per answer in the thread, in document order.
	///
	/// An identifier is the text's length and its first characters, which is
	/// enough to tell one turn from another without carrying every turn's text
	/// through the port. It exists so "is this answer new" is answered by
	/// membership rather than by position: a virtualised thread can reorder or
	/// unmount its turns, and a newest-by-position read then hands back an
	/// answer that was already there.
	pub Marks:Vec<String>,
}

/// A value as text: as it is when it is text, rendered when it is not.
pub fn TextOf(Value:Option<&Value>) -> String {
	match Value {
		None | Some(Value::Null) => String::new(),
		Some(Value::String(Text)) => Text.clone(),
		Some(Other) => Other.to_string(),
	}
}

/// A label as it is compared: case and runs of whitespace are not part of a
/// model's name, so both sides are folded. Nothing else is folded: a different
/// model is a different model.
pub fn Normalise(Value:Option<&Value>) -> String {
	TextOf(Value).split_whitespace().collect::<Vec<&str>>().join(" ").to_lowercase()
}

/// What was read, for a failure's detail, without pretending `null` is a value.
pub fn Describe(Value:Option<&Value>) -> String {
	match Value {
		None | Some(Value::Null) => "nothing".to_string(),
		Some(Value::String(Text)) if Text.trim().is_empty() => "nothing".to_string(),
		Some(Other) => Other.to_string(),
	}
}

/// What a text read, for a failure's detail.
pub fn DescribeText(Value:Option<&str>) -> String {
	match Value {
		None => "nothing".to_string(),
		Some(Text) if Text.trim().is_empty() => "nothing".to_string(),
		Some(Text) => serde_json::to_string(Text).unwrap_or_else(|_| format!("{Text:?}")),
	}
}

/// An answer's identifier, computed the way the page computes it: its length in
/// code points and its opening.
///
/// The two sides must agree, which is why the count is code points rather than
/// UTF-16 units: an answer with an emoji in it would otherwise mark differently
/// in the driver than in the page and every read would look new.
pub fn Mark(Text:&str) -> String {
	format!(
		"{}:{}",
		Text.chars().count(),
		Text.chars().take(48).collect::<String>()
	)
}

/// One read of the thread, or `None` when the page answered with something that
/// is not one.
pub fn AsAnswers(Value:&Value) -> Option<Answers> {
	let Object = Value.as_object()?;
	let Count = Object.get("count")?.as_f64()?;
	Some(Answers {
		Count,
		Text:TextOf(Object.get("text")),
		Busy:Object.get("busy").and_then(Value::as_bool).unwrap_or(false),
		Marks:Object
			.get("marks")
			.and_then(Value::as_array)
			.map(|Marks| Marks.iter().map(|Mark| TextOf(Some(Mark))).collect())
			.unwrap_or_default(),
	})
}

/// The steps the page reported, or `None` when it reported something else.
pub fn AsSteps(Value:&Value) -> Option<Vec<Step>> {
	let Entries = Value.as_array()?;
	Some(
		Entries
			.iter()
			.map(|Entry| {
				let Object = Entry.as_object();
				Step {
					Depth:Object
						.and_then(|Object| Object.get("depth"))
						.and_then(Value::as_f64)
						.map(|Depth| Depth as i64)
						.unwrap_or(0),
					Label:TextOf(Object.and_then(|Object| Object.get("label"))),
					Detail:TextOf(Object.and_then(|Object| Object.get("detail"))),
				}
			})
			.collect(),
	)
}

/// The sources the page cited, or `None` when it cited something else.
pub fn AsSources(Value:&Value) -> Option<Vec<Source>> {
	let Entries = Value.as_array()?;
	Some(
		Entries
			.iter()
			.map(|Entry| {
				let Object = Entry.as_object();
				Source {
					Url:TextOf(Object.and_then(|Object| Object.get("url"))),
					Label:TextOf(Object.and_then(|Object| Object.get("label"))),
				}
			})
			.collect(),
	)
}

//! `agentic.chat.ask` over a page the composition root supplies.
//!
//! The four rules of the port, applied in the order that makes each of them
//! mean something:
//!
//! 1. **The mode is a precondition.** The composer's toggle is read before
//!    anything is typed. When it does not read `search` it is pressed once and
//!    read again; when it still does not, the call fails with `wrong-mode` and
//!    nothing was sent. One press, not a loop: a toggle that does not move on
//!    the first press is a page that will not move on the fifth, and a retry
//!    loop would spend the caller's deadline proving it.
//! 2. **The model is a precondition.** The label is compared before anything is
//!    typed, so a page set to another model is refused rather than asked.
//! 3. **Completion is a rule, not a sleep.** The answer's text is read until
//!    three consecutive reads report no change, which is [`Settlement`]. A
//!    sleep would return whatever the network happened to have delivered; a
//!    count of reads would return an answer that had stopped at a paragraph
//!    break. A deadline hit is `not-ready`, never a partial answer presented as
//!    a whole one.
//! 4. **A collapsed answer is `truncated`.** The control that hides the rest of
//!    an answer is looked for inside the newest answer once it has settled,
//!    because the text in the DOM is then demonstrably not all of it.
//!
//! Everything the page does that is not one of those four is `not-ready`: a
//! page that will not open, a page that fails mid-read, a composer that is not
//! there, a send control that never enables. The manifest has four codes and
//! the driver does not invent a fifth; the cause rides along in `detail`
//! instead.
//!
//! Nothing here opens a browser, knows a transport, or reads an environment
//! variable: the page is the composition root's to supply, and the fake in
//! `Test/` is the proof that the rules hold without one.

/// How long between two reads of an answer while it is being written.
pub const DefaultPoll:Duration = Duration::from_millis(250);




/// A release that has not finished in this long is reported, not waited on.
pub const DefaultDisposeBudget:Duration = Duration::from_secs(5);

/// Where a dispose failure goes, because it may not be returned.
pub type Report = Arc<dyn Fn(&Target, &Error) + Send + Sync>;

/// How one driver is configured.
#[derive(Clone, Debug)]
pub struct Options {

	pub Url:String,
	/// How the page is read.

	/// How long between two reads of the answer while it is being written.
	pub Poll:Duration,
	/// How long a release may take before it is reported as a failure.
	pub DisposeBudget:Duration,
}

impl Options {
	/// The defaults: the reference product's selectors, a quarter-second

	pub fn New(Url:impl Into<String>) -> Self {

			Selectors:DefaultSelectors(),

/// A `Chat` over one `Page`.
pub struct Driver {
	Page:Arc<dyn Page>,
	Options:Options,
	Pace:Arc<dyn Pace>,
	Report:Report,
}

/// Build a `Chat` over one `Page`, on the pace the caller supplied.
///
/// The pace is a parameter for the same reason the clock is: a driver that
/// reached for a timer would decide which executor is running it, and a driver
/// whose waits are supplied can be tested by advancing a clock instead of
/// sleeping.
pub fn Create(Page:Arc<dyn Page>, Options:Options, Pace:Arc<dyn Pace>) -> Driver {
	Driver {
		Page,
		Options,
		Pace,
		// Silent by default: a library that writes to the process's stderr behind a
		// caller's back is a library that corrupts a protocol. The composition root
		// supplies the record.
		Report:Arc::new(|_, _| {}),
	}
}

impl Driver {
	/// The same driver, with somewhere to put a dispose failure.
	pub fn Reporting(mut self, Report:Report) -> Self {
		self.Report = Report;
		self
	}

	/// How much of the caller's budget is left, for a failure's wording.
	fn Budget(&self, Call:&Call) -> Duration { Call.Remaining(self.Pace.as_ref()) }

	/// One expression, run inside whatever budget is left.
	async fn Probe(
		&self,
		Phase:&str,
		Expression:String,
		Target:&Target,
		Call:&Call,
		Budget:Duration,
	) -> Result<Value, Error> {
		if Call.Cancelled() {
			return Err(Error::NotReady(
				format!("{Phase} was cancelled before it started"),
				"the caller's signal had already fired",
			));
		}
		if Call.Expired(self.Pace.as_ref()) {
			return Err(Error::NotReady(
				format!("{Phase} exceeded the {}ms deadline", Budget.as_millis()),
				"the page was still working when the budget ran out",
			));
		}
		let Work = self.Page.Evaluate(Target, &Expression, Call.clone());
		match Bound(Work, Call, self.Pace.as_ref()).await {
			Ok(Ok(Value)) => Ok(Value),
			Ok(Err(Failure)) => {
				Err(Error::NotReady(
					format!("{Phase} could not be read from the page"),
					Failure.Message,
				))
			},
			Err(Expired::Timeout) => {
				Err(Error::NotReady(
					format!("{Phase} exceeded the {}ms deadline", Budget.as_millis()),
					"the page never answered",
				))
			},
			Err(Expired::Cancelled) => {
				Err(Error::NotReady(
					format!("{Phase} was cancelled by the caller"),
					"the signal fired while the page was being asked",
				))
			},
		}
	}

	/// Open the configured page, bounded.
	async fn Open(&self, Call:&Call, Budget:Duration) -> Result<Target, Error> {
		if Call.Cancelled() {
			return Err(Error::NotReady(
				"open was cancelled before it started",
				"the caller's signal had already fired",
			));
		}
		if Call.Expired(self.Pace.as_ref()) {
			return Err(Error::NotReady(
				format!("open exceeded the {}ms deadline", Budget.as_millis()),
				"the page was still working when the budget ran out",
			));
		}
		let Work = self.Page.Open(&self.Options.Url, Call.clone());
		match Bound(Work, Call, self.Pace.as_ref()).await {
			Ok(Ok(Target)) => Ok(Target),
			Ok(Err(Failure)) => Err(Error::NotReady("the chat page could not be opened", Failure.Message)),
			Err(Expired::Timeout) => {
				Err(Error::NotReady("the chat page could not be opened", "the page never answered"))
			},
			Err(Expired::Cancelled) => {
				Err(Error::NotReady(
					"the chat page could not be opened",
					"the caller gave up before the page opened",
				))
			},
		}
	}

	/// One read of the thread, refusing a page that answers something else.
	async fn ReadAnswers(&self, Phase:&str, Target:&Target, Call:&Call, Budget:Duration) -> Result<Answers, Error> {
		let Selectors = &self.Options.Selectors;
		let Expression = ReadAnswers(Selectors, Selectors.Format);
		let Value = self.Probe(Phase, Expression, Target, Call, Budget).await?;
		AsAnswers(&Value).ok_or_else(|| {
			Error::NotReady(
				format!("{Phase} could not be read from the page"),
				format!("the page answered {} instead of a thread", Describe(Some(&Value))),
			)
		})
	}

	/// Release the target, bounded, and never fail.
	async fn Release(&self, Target:&Target) {
		let Deadline = self
			.Pace
			.Now()
			.checked_add(self.Options.DisposeBudget)
			.unwrap_or_else(|| self.Pace.Now());
		let Call = Call { Deadline, Cancellation:Arc::new(Never) };
		let Work = self.Page.Dispose(Target.clone());
		if let Err(Why) = Bound(Work, &Call, self.Pace.as_ref()).await {
			let Error = Error::NotReady(
				format!(
					"dispose of {} exceeded its {}ms budget",
					Target.Url,
					self.Options.DisposeBudget.as_millis()
				),
				Why.Name(),
			);
			// A reporter that panics must not turn a release into a failure the
			// caller never asked for. Under `panic = "abort"` this cannot catch
			// anything, which is the stated cost of the release profile.
			let Report = self.Report.clone();
			let _ = std::panic::catch_unwind(std::panic::AssertUnwindSafe(move || {
				Report(Target, &Error);
			}));
		}
	}

	/// Everything between an open page and a released one.
	async fn Body<'a>(
		&'a self,
		Question:&Question,
		Call:&Call,
		Budget:Duration,
		Target:&Target,
		mut Sink:Option<&'a mut (dyn FnMut(&Observation) -> Control + Send)>,
	) -> Result<Answer, Error> {
		let Selectors = &self.Options.Selectors;

		// The mode is a precondition, not a preference.
		let mut Mode = self.Probe("mode", ReadMode(Selectors), Target, Call, Budget).await?;
		if Normalise(Some(&Mode)) != "search" {
			self.Probe("mode press", PressMode(Selectors), Target, Call, Budget).await?;
			Mode = self.Probe("mode", ReadMode(Selectors), Target, Call, Budget).await?;
			if Normalise(Some(&Mode)) != "search" {
				return Err(Error::WrongMode(
					"the composer does not offer search mode",
					format!("after pressing the toggle it reads {}", Describe(Some(&Mode))),
				));
			}
		}

		// The model is a precondition, and it is compared before the send.
		let Label = self.Probe("model", ReadModel(Selectors), Target, Call, Budget).await?;
		let Wanted = Value::String(Question.Model.clone());
		if Normalise(Some(&Label)) != Normalise(Some(&Wanted)) {
			return Err(Error::WrongModel(
				"the page is set to a different model",
				format!(
					"wanted {}, the page reads {}",
					serde_json::to_string(&Question.Model).unwrap_or_else(|_| format!("{:?}", Question.Model)),
					Describe(Some(&Label))
				),
			));
		}

		// Correlate the turn: what is already in the thread is not this answer.
		let Before = self.ReadAnswers("answers", Target, Call, Budget).await?;
		// A page that is already producing an answer is not a page this call
		// may type into: the question would land in a turn that is still being
		// written, and the answer read back would belong to neither.
		if Before.Busy {
			return Err(Error::NotReady(
				"the page was already producing an answer",
				"the busy control was visible before the question was typed",
			));
		}

		let Typed = self
			.Probe("composer", TypeQuestion(Selectors, &Question.Text), Target, Call, Budget)
			.await?;
		if Typed.as_bool() != Some(true) {
			return Err(Error::NotReady(
				"the composer could not be found",
				format!("nothing matched {:?}", Selectors.Composer),
			));
		}

		let Sent = self
			.Probe("submit", PressSubmit(Selectors, &Question.Text), Target, Call, Budget)
			.await?;
		match Sent.as_str() {
			Some("sent") => {},
			Some("mismatch") => {
				return Err(Error::NotReady(
					"the composer does not hold the question",
					"the editor kept a different text than the one that was typed, so nothing was sent",
				));
			},
			Some("disabled") => {
				return Err(Error::NotReady(
					"the question could not be sent",
					"the send control is disabled after the question was typed",
				));
			},
			_ => {
				return Err(Error::NotReady(
					"the question could not be sent",
					format!("nothing matched {:?}", Selectors.Submit),
				));
			},
		}

		// Completion is stable text: unchanged for three consecutive reads.
		let mut Progress = Settlement::Empty;
		let Text = loop {
			// The deadline is checked here, not only inside the read, so a page
			// that keeps writing is reported as an answer that
			// never settled rather than as one slow read.
			if Call.Expired(self.Pace.as_ref()) {
				return Err(Error::NotReady(
					format!("the answer did not settle within the {}ms deadline", Budget.as_millis()),
					format!("the last read was {}", DescribeText(Progress.Text())),
				));
			}
			let Read = self.ReadAnswers("answer", Target, Call, Budget).await?;
			// A new turn is recognised by membership, not by position: a
			// virtualised thread reorders and unmounts its turns, so the last
			// element in document order is not reliably the newest turn, and a
			// count is not reliably growing. An answer whose identity was not
			// in the thread before the question is the only reading that
			// survives both.
			let Fresh = !Read.Text.is_empty() && !Before.Marks.contains(&Mark(&Read.Text));
			let Current = if Fresh { Some(Read.Text.as_str()) } else { None };
			// A page that says it is still producing the answer has not
			// finished, whatever its text is doing: a pause between tokens, a
			// reasoning step and a slow code block all leave the text unchanged
			// while the answer is demonstrably incomplete. So a busy read
			// resets the confirmations instead of advancing them.
			let Settled = if Read.Busy {
				Progress = Settlement::Empty;
				false
			} else {
				Progress.Observe(Current)
			};
			// The observation is pushed before the loop acts on it, so an
			// observer sees the read that settled the answer rather than
			// missing the one that mattered.
			if let Some(Sink) = Sink.as_deref_mut() {
				let Seen = Observation {
					Count:Read.Count,
					Text:Read.Text.clone(),
					Busy:Read.Busy,
					Fresh,
					Settled,
				};
				if Sink(&Seen) == Control::Stop {
					break Read.Text;
				}
			}
			if Settled {
				break Progress.Text().unwrap_or_default().to_string();
			}
			let Wait = self.Options.Poll.min(Call.Remaining(self.Pace.as_ref()));
			self.Pace.Sleep(Wait).await;
		};

		// A collapsed answer is not the whole answer.
		let Collapsed = self.Probe("collapse", ReadCollapse(Selectors), Target, Call, Budget).await?;
		let Collapsed = TextOf(Some(&Collapsed)).trim().to_string();
		if !Collapsed.is_empty() {
			return Err(Error::Truncated(
				"the answer is collapsed behind a control",
				format!(
					"the page shows {} over {} characters",
					serde_json::to_string(&Collapsed).unwrap_or_else(|_| format!("{Collapsed:?}")),
					Text.chars().count()
				),
			));
		}

		let Trace = self.Probe("trace", ReadTrace(Selectors), Target, Call, Budget).await?;
		let Trace = AsSteps(&Trace).ok_or_else(|| {
			Error::NotReady(
				"the trace could not be read from the page",
				format!("the page answered {} instead of a list of steps", Describe(Some(&Trace))),
			)
		})?;

		let Sources = self.Probe("sources", ReadSources(Selectors), Target, Call, Budget).await?;
		let Sources = AsSources(&Sources).ok_or_else(|| {
			Error::NotReady(
				"the sources could not be read from the page",
				format!("the page answered {} instead of a list of sources", Describe(Some(&Sources))),
			)
		})?;

		let Model = {
			let Read = TextOf(Some(&Label)).trim().to_string();
			if Read.is_empty() { Question.Model.trim().to_string() } else { Read }
		};

		Ok(Answer { Text, Trace, Sources, Model, Mode:"search".to_string() })
	}

	/// The whole call, with the release guaranteed on every path.
	async fn Perform<'a>(
		&'a self,
		Question:Question,
		Call:Call,
		Sink:Option<&'a mut (dyn FnMut(&Observation) -> Control + Send)>,
	) -> Result<Answer, Error> {
		let Budget = self.Budget(&Call);

		// The port is defined for search mode, so a question that asks for
		// another one is refused here, before a page is opened: there
		// is nothing to verify against a toggle the driver is not
		// willing to use.
		if !Question.Mode.IsSearch() {
			return Err(Error::WrongMode(
				"this port only asks in search mode",
				format!(
					"the question asked for {}",
					serde_json::to_string(Question.Mode.Name())
						.unwrap_or_else(|_| format!("{:?}", Question.Mode.Name()))
				),
			));
		}

		let Target = self.Open(&Call, Budget).await?;
		let Outcome = self.Body(&Question, &Call, Budget, &Target, Sink).await;
		self.Release(&Target).await;
		Outcome
	}
}

impl Chat for Driver {
	fn Ask<'a>(&'a self, Question:Question, Call:Call) -> FutureBox<'a, Result<Answer, Error>> {
		Box::pin(async move { self.Perform(Question, Call, None).await })
	}

	fn Observe<'a>(
		&'a self,
		Question:Question,
		Call:Call,
		Sink:&'a mut (dyn FnMut(&Observation) -> Control + Send),
	) -> FutureBox<'a, Result<Answer, Error>> {
		Box::pin(async move { self.Perform(Question, Call, Some(Sink)).await })
	}
}

//! The four failure modes `module.json` declares, and no others.
//!
//! One enum, one variant per code, `detail` kept apart from the message in
//! every one of them.

use std::fmt;

/// The codes, in the order the manifest names them.
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub enum Code {
	/// The page could not answer.
	NotReady,
	/// The composer would not enter search mode.
	WrongMode,
	/// The page is set to a different model.
	WrongModel,
	/// The answer is collapsed behind a control.
	Truncated,
}

impl Code {
	/// The string the manifest uses.
	pub const fn Name(self) -> &'static str {
		match self {
			Code::NotReady => "not-ready",
			Code::WrongMode => "wrong-mode",
			Code::WrongModel => "wrong-model",
			Code::Truncated => "truncated",
		}
	}

	/// Every code, so a harness can compare the manifest against the type.
	pub const fn All() -> [Code; 4] { [Code::NotReady, Code::WrongMode, Code::WrongModel, Code::Truncated] }

	/// The code a manifest string names, if it names one of the four.
	pub fn Parse(Value:&str) -> Option<Code> { Code::All().into_iter().find(|Code| Code.Name() == Value) }
}

/// The one error type an `Ask` call fails with.
///
/// `NotReady` is the one code that covers a provider's failures as well,
/// because the manifest has no fifth code and "the page could not answer" is
/// what a dead transport, a navigation failure and a slow page all mean to this
/// capability. The cause is carried in `detail` rather than flattened, so the
/// difference survives.
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
	/// The page is not in a state where it can answer.
	NotReady {
		/// The sentence a human reads.
		Message:String,
		/// What was observed, kept apart from the sentence.
		Detail:String,
	},
	/// The composer would not enter search mode, so nothing was sent.
	WrongMode {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
	/// The page is set to a different model, so nothing was sent.
	WrongModel {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
	/// The answer is collapsed, so the text in the DOM is only part of it.
	Truncated {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
}

impl Error {
	/// A page that could not answer.
	pub fn NotReady(Message:impl Into<String>, Detail:impl Into<String>) -> Self {
		Error::NotReady { Message:Message.into(), Detail:Detail.into() }
	}

	/// A composer that would not enter search mode.
	pub fn WrongMode(Message:impl Into<String>, Detail:impl Into<String>) -> Self {
		Error::WrongMode { Message:Message.into(), Detail:Detail.into() }
	}

	/// A page set to a different model.
	pub fn WrongModel(Message:impl Into<String>, Detail:impl Into<String>) -> Self {
		Error::WrongModel { Message:Message.into(), Detail:Detail.into() }
	}

	/// An answer that is only part of itself.
	pub fn Truncated(Message:impl Into<String>, Detail:impl Into<String>) -> Self {
		Error::Truncated { Message:Message.into(), Detail:Detail.into() }
	}

	/// The code this failure carries.
	pub const fn Code(&self) -> Code {
		match self {
			Error::NotReady { .. } => Code::NotReady,
			Error::WrongMode { .. } => Code::WrongMode,
			Error::WrongModel { .. } => Code::WrongModel,
			Error::Truncated { .. } => Code::Truncated,
		}
	}

	/// The sentence a human reads.
	pub fn Message(&self) -> &str {
		match self {
			Error::NotReady { Message, .. }
			| Error::WrongMode { Message, .. }
			| Error::WrongModel { Message, .. }
			| Error::Truncated { Message, .. } => Message,
		}
	}

	/// What was observed, which may be empty.
	pub fn Detail(&self) -> &str {
		match self {
			Error::NotReady { Detail, .. }
			| Error::WrongMode { Detail, .. }
			| Error::WrongModel { Detail, .. }
			| Error::Truncated { Detail, .. } => Detail,
		}
	}
}

impl fmt::Display for Error {
	fn fmt(&self, Formatter:&mut fmt::Formatter<'_>) -> fmt::Result {
		write!(Formatter, "{}: {}", self.Code().Name(), self.Message())
	}
}

impl std::error::Error for Error {}

//! How the text of an answer is read out of the page.
//!
//! The page is the same either way; what changes is which of its statements is
//! taken as the answer. `Text` is `innerText`: what the product put on screen,
//! with a table's columns reduced to tabs, a list's structure reduced to line
//! breaks, and a formula reduced to its glyphs. `Markdown` serialises the same
//! DOM through `Source/Script/Extract.js`, so a table keeps its columns, a code
//! block keeps its fence and a formula is its TeX.
//!
//! `Text` is the default, and deliberately so: the parity scenarios were
//! baselined against it, and a format that silently changed every recorded
//! answer would turn a baseline into a diff nobody asked for. A capture asks
//! for `Markdown`; anything that compares against a recorded text keeps
//! `Text`.
//!
//! It is a field of [`crate::Struct::Selectors::Selectors`] rather than of the
//! driver's options because it is a fact about the page: one product's answer
//! markup is worth serialising and another's is not, and the selector file is
//! where a product's reading is described.

/// How an answer's text is read.
#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
pub enum Format {
	/// `innerText`, trimmed: what the page rendered, as text.
	#[default]
	Text,
	/// The DOM serialised as Markdown by the page itself.
	Markdown,
}

impl Format {
	/// The word the command line and the selector file use.
	pub const fn Name(self) -> &'static str {
		match self {
			Format::Text => "text",
			Format::Markdown => "markdown",
		}
	}

	/// Every format, in the order the command line lists them.
	pub const fn All() -> [Format; 2] { [Format::Text, Format::Markdown] }

	/// The format a word names, or `None` when it names none of them.
	///
	/// Case is folded and nothing else is: a format that is nearly right is a
	/// caller's mistake, and a silent fallback would capture a whole thread in
	/// the format they did not ask for.
	pub fn From(Word:&str) -> Option<Self> {
		let Word = Word.trim().to_lowercase();
		Self::All().into_iter().find(|Format| Format.Name() == Word)
	}
}

//! The port `agentic.chat.ask` promises, written the way its consumer asks for
//! it.
//!
//! `module.json` names four errors - `not-ready`, `wrong-mode`, `wrong-model`
//! and `truncated` - and [`crate::Struct::Error`] is where they live. A caller
//! narrows on the variant; anything a page does that is not one of those four
//! is mapped onto one of them at the boundary, so a consumer never has to guess
//! what a failure meant.
//!
//! The port states more than data shapes, because two drivers that agreed on
//! the shapes alone would still behave differently:
//!
//! - **A question is answered or it is not.** `Ask` returns an answer whose
//!   text the page had stopped changing, or it fails. A partial answer is never
//!   returned as a whole one, and an empty string is never an answer.
//! - **Mode and model are preconditions.** The composer is in search mode and
//!   set to the requested model before anything is typed, so a wrong answer is
//!   refused rather than produced.
//! - **The answer is the one this question caused.** A thread already holds the
//!   answers to earlier questions, so the turn is correlated by counting them
//!   before the send and reading only the one that appeared after it.
//! - **A collapsed answer is refused.** When the page hides the rest of an
//!   answer behind a control, the text in the DOM is not the answer, and
//!   returning it would be a quiet lie.
//! - **Cancellation is `not-ready`.** A signal that fires and a deadline that
//!   passes are the same event to a consumer - the answer did not arrive - so
//!   both are `not-ready`. `detail` says which one happened.

use AgenticRuntime::{Call, FutureBox};
use serde_json::Value;

/// One step of the work an answer reports having done.
#[derive(Clone, Debug, Default, PartialEq, Eq)]
pub struct Step {
	/// Nesting: a step inside a step is one level deeper than it.
	pub Depth:i64,
	/// What the step is called, as the page labels it.
	pub Label:String,
	/// The page's own note about it - a duration, a state - when it has one.
	pub Detail:String,
}

/// One cited source behind an answer.
#[derive(Clone, Debug, Default, PartialEq, Eq)]
pub struct Source {
	/// Where the citation points.
	pub Url:String,
	/// How the page labels the citation.
	pub Label:String,
}

/// The composer mode a question must be asked in.
///
/// The port is defined for search only. An agent mode answers by running tools
/// on the caller's behalf, which is a different capability with different costs
/// and a different budget, so a driver that silently used one would be spending
/// somebody else's allowance. [`Mode::Other`] exists so that asking for one is
/// a refusal with a code rather than a value the type system quietly forbids: a
/// caller migrating from another driver learns why from `wrong-mode`.
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
	/// The one mode this port asks in.
	Search,
	/// Some other mode, named as the caller named it.
	Other(String),
}

impl Mode {
	/// The mode's name, as a caller would write it.
	pub fn Name(&self) -> &str {
		match self {
			Mode::Search => "search",
			Mode::Other(Name) => Name,
		}
	}

	/// Whether this is the mode the port asks in.
	pub fn IsSearch(&self) -> bool { matches!(self, Mode::Search) }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Question {

	pub Text:String,

	pub Mode:Mode,

	pub Model:String,

/// What a settled turn produced.

pub struct Answer {

	/// The work the answer reports having done, in the order it did it.
	pub Trace:Vec<Step>,
	/// The sources the answer cites.
	pub Sources:Vec<Source>,

/// The handle a page is bound to.
///
/// The driver passes it back to the `Page` that issued it and never reads it,
/// so a provider may hang whatever it needs there - a target id, a session, a
/// socket. `Url` is the one field the port keeps, because a failure report that
/// cannot say which page failed is not a report.
#[derive(Clone, Debug, Default, PartialEq, Eq)]
pub struct Target {
	/// The page this session is bound to.
	pub Url:String,
}

/// What a page operation reports when it fails.
///
/// A page is not this capability, so it does not speak this capability's four
/// codes. It says what it observed, and the driver maps that onto `not-ready`
/// with the observation in `detail` - which is the whole point of keeping the
/// detail apart from the sentence.
#[derive(Clone, Debug, Default, PartialEq, Eq)]
pub struct Failure {
	/// The sentence a human reads.
	pub Message:String,
	/// What was observed.
	pub Detail:String,
}

impl Failure {
	/// A failure with a sentence and nothing observed.
	pub fn Of(Message:impl Into<String>) -> Self { Self { Message:Message.into(), Detail:String::new() } }
}

/// The browser as this capability needs it: one page, asked one expression at a
/// time.
///
/// What the port demands of an implementation:
///
/// - `Evaluate` returns the value of the expression, not a wrapper around it,
///   and the expression is a complete JavaScript expression whose value is
///   JSON-serialisable. `document` is the only host object it may touch, so a
///   page that is a tab and a page that is a fixture answer the same driver.
/// - the [`Call`] is honoured: a page that never answers is failed, not
///   awaited. The driver checks the same deadline on its side, so an
///   implementation that ignores it cannot quietly turn a bounded call into an
///   unbounded one.
/// - `Dispose` never fails. A release that fails is a fact about the browser,
///   and a caller that already has its answer must not lose it to a failing
///   cleanup.
pub trait Page: Send + Sync {
	/// Open the page at `Url`.
	fn Open<'a>(&'a self, Url:&'a str, Call:Call) -> FutureBox<'a, Result<Target, Failure>>;
	/// Evaluate one expression in the page and return its value.
	fn Evaluate<'a>(
		&'a self,
		Target:&'a Target,
		Expression:&'a str,
		Call:Call,
	) -> FutureBox<'a, Result<Value, Failure>>;
	/// Release everything the page holds. Never fails.
	fn Dispose<'a>(&'a self, Target:Target) -> FutureBox<'a, ()>;
}

/// The one port this capability offers.
pub trait Chat: Send + Sync {
	/// Ask one question and return the answer once the page has settled.
	fn Ask<'a>(&'a self, Question:Question, Call:Call) -> FutureBox<'a, Result<Answer, crate::Struct::Error::Error>>;
	/// Ask one question, and report the thread on every read until it settles.
	///
	/// This is `Ask` with its inside exposed: the same preconditions, the same
	/// turn correlation and the same completion rule, plus one [`Observation`]
	/// pushed to the sink for every read the driver takes.
	///
	/// The sink is the observer and the [`Call`] is the abort: returning
	/// [`Control::Stop`] disconnects it the way a `MutationObserver`'s
	/// `disconnect()` does, and the call's cancellation ends the wait from the
	/// outside. Both end the observation, neither is an error - the answer so
	/// far comes back either way, so a caller that stopped early can still read
	/// what was there.
	fn Observe<'a>(
		&'a self,
		Question:Question,
		Call:Call,
		Sink:&'a mut (dyn FnMut(&Observation) -> Control + Send),
	) -> FutureBox<'a, Result<Answer, crate::Struct::Error::Error>>;
}

/// One look at the thread, as the driver saw it.
///
/// It carries what was read and not what to do about it: a consumer deciding
/// "still working" from `Busy` and `Settled` is the consumer's business, and a
/// driver that returned a verdict would be answering its own question.
#[derive(Clone, Debug, Default, PartialEq)]
pub struct Observation {
	/// How many answers the thread holds.
	pub Count:f64,
	/// The newest answer's text, trimmed.
	pub Text:String,
	/// Whether the page says it is still producing that answer.
	pub Busy:bool,
	/// Whether this answer was not in the thread before the question.
	pub Fresh:bool,
	/// Whether the text has held still long enough, with the page idle, to be
	/// read as settled.
	pub Settled:bool,
}

/// What an observer says back to the driver.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Control {
	/// Keep observing.
	Continue,
	/// Stop observing now; the answer so far is returned.
	Stop,
}

//! Where a chat product keeps the things this driver has to read.
//!
//! A driver written against one product's markup is not reusable, and a driver
//! written against nothing cannot apply its rules at all: the mode and the
//! model are preconditions because there is a toggle and a label to read, and

//! it. So the shape is fixed here and the strings are configuration.
//!
//! A selector may be a CSS selector list - `"textarea, [contenteditable]"` -
//! which is how one product is covered by one entry without the driver knowing
//! which of the two a given page used.

use crate::Struct::Format::Format;

/// Where one chat product keeps the things this driver reads, and how the
/// answer is read out of it.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selectors {
	/// The composer's mode toggles. Each carries its mode as its label and
	/// `aria-pressed="true"` on the one that is active, so the mode is the
	/// label of the single pressed toggle, and unknown when no toggle is
	/// pressed or several are.
	pub Mode:String,
	/// The element naming the model the composer will send to.
	pub Model:String,
	/// Where the question is typed: a `textarea`, an `input` or a
	/// contenteditable.
	pub Composer:String,
	/// The control that sends the question.
	pub Submit:String,
	/// Every answer in the thread, in document order; the last one is the
	/// newest.
	pub Answer:String,
	/// The control that collapses an answer, searched for *inside* the answer
	/// container. Not across the page: an earlier answer's control is not this
	/// answer's.
	pub Collapse:String,
	/// One step of the answer's trace, searched for *inside* the newest answer.
	pub Trace:String,
	/// One cited source: an anchor whose `href` is the citation, searched for
	/// *inside* the newest answer, for the same reason as `Trace`.
	pub Sources:String,
	/// The control that means the page is still producing the answer - a stop
	/// or a cancel control.
	///
	/// Searched for across the whole page, not inside the answer, because a
	/// product keeps it in the composer: it is a fact about the turn, not about
	/// the text the turn has produced so far.
	pub Busy:String,
	/// The answer's own turn container, when the product has one.
	///
	/// Searched for as an ancestor of the answer, so a product that wraps a turn
	/// (body, steps, duration, source chips) can be asked what the *turn* says
	/// rather than what its body says.
	pub Turn:String,
	/// What the turn's text reads once the turn is over, as a pattern.
	///

	/// negation - a cancelled turn is neither.
	///
	/// Empty by default, and deliberately so. A header that reads like a
	/// completion marker is often a duration label that is present while the
	/// answer streams, and a pattern that matches it makes every turn look
	/// finished from its first token - the exact false settle the check exists

	pub Format:Format,

/// The defaults, with whichever of these keys the caller stated replaced.
///

/// should not have to restate the seven it is happy with, and because a key that
/// is absent must keep the default instead of becoming an empty selector that
/// matches everything.
///

///

pub fn Merged(Overrides:&serde_json::Value) -> Selectors {

	let Read = |Key:&str, Field:&mut String| {
		if let Some(Text) = Overrides.get(Key).and_then(|Value| Value.as_str()) {
			*Field = Text.to_string();
		}
	};
	Read("mode", &mut Base.Mode);
	Read("model", &mut Base.Model);
	Read("composer", &mut Base.Composer);
	Read("submit", &mut Base.Submit);
	Read("answer", &mut Base.Answer);
	Read("collapse", &mut Base.Collapse);
	Read("trace", &mut Base.Trace);
	Read("sources", &mut Base.Sources);
	Read("busy", &mut Base.Busy);
	Read("turn", &mut Base.Turn);
	Read("done", &mut Base.Done);
	Read("scroll", &mut Base.Scroll);
	if let Some(Named) = Overrides.get("format").and_then(|Value| Value.as_str()) {
		if let Some(Format) = Format::From(Named) {
			Base.Format = Format;
		}
	}
	Base
}

//! Completion as an explicit state, not a loop with a counter.
//!
//! Three consecutive reads must report no change before an answer counts as
//! settled, and the counter counts *comparisons*: the third read that equals
//! the one before it is what settles the answer, so a page that delivers its
//! last token and stops is read four times in all. Fewer than three is a page
//! that paused mid-sentence; more than three is latency nobody asked for.
//!
//! Written as a state machine because a loop with a counter cannot say what it
//! is doing: `Reading { Text, Confirmed }` is a claim about the page - "this
//! text has been confirmed twice" - and `Settled(Text)` is the only state a
//! caller may read an answer out of. The empty state is not an answer either: a
//! thread with no answer yet resets the machine rather than confirming an empty
//! string.

/// How many comparisons must agree before an answer counts as settled.
pub const StableReads:u8 = 3;

/// How many consecutive reads have confirmed the text currently held.
#[derive(Clone, Copy, Debug, Default, PartialEq, Eq, PartialOrd, Ord)]
pub struct Confirmations(u8);

impl Confirmations {
	/// No confirmation yet: the text was just read for the first time.
	pub const fn First() -> Self { Self(0) }

	/// One more read that agreed.
	pub const fn Next(self) -> Self { Self(self.0.saturating_add(1)) }

	/// How many comparisons have agreed.
	pub const fn Count(self) -> u8 { self.0 }

	/// Whether this many confirmations is enough to settle an answer.
	pub const fn Settles(self) -> bool { self.0 >= StableReads }
}

/// What the driver knows about the answer it is reading.
#[derive(Clone, Debug, Default, PartialEq, Eq)]
pub enum Settlement {
	/// Nothing has been read yet, or the newest answer is still empty.
	#[default]
	Empty,
	/// Text the page has shown, and how many reads have agreed on it.
	Reading {
		/// The text as last read.
		Text:String,
		/// How many consecutive reads have confirmed it.
		Confirmed:Confirmations,
	},
	/// Text three reads agreed on. This is the only state an answer comes from.
	Settled(String),
}

impl Settlement {
	/// Fold one read into the state, and say whether it is settled.
	///
	/// `Current` is the answer's text when this question caused a new answer,
	/// and `None` when the thread has not grown: an unchanged thread is not
	/// a confirmation of anything, so it resets the state instead of
	/// advancing it.
	pub fn Observe(&mut self, Current:Option<&str>) -> bool {
		let Current = match Current {
			Some(Text) if !Text.is_empty() => Text,
			_ => {
				*self = Settlement::Empty;
				return false;
			},
		};
		*self = match std::mem::take(self) {
			Settlement::Reading { Text, Confirmed } if Text == Current => {
				let Confirmed = Confirmed.Next();
				if Confirmed.Settles() {
					Settlement::Settled(Current.to_string())
				} else {
					Settlement::Reading { Text, Confirmed }
				}
			},
			_ => Settlement::Reading { Text:Current.to_string(), Confirmed:Confirmations::First() },
		};
		matches!(self, Settlement::Settled(_))
	}

	/// The text this state holds, if it holds one.
	pub fn Text(&self) -> Option<&str> {
		match self {
			Settlement::Empty => None,
			Settlement::Reading { Text, .. } => Some(Text),
			Settlement::Settled(Text) => Some(Text),
		}
	}
}

//! A page that answers the driver's own expressions, with no browser in sight.
//!
//! The TypeScript fake evaluates the expressions against a page-shaped object.
//! Rust has no JavaScript engine in a std-only test, so this fake answers by
//! recognising the expression it was handed - and it recognises it by calling
//! the same function the driver calls, so a change to the protocol breaks the
//! fake rather than passing silently.

use std::{
	sync::{
		Arc,
		Mutex,
		atomic::{AtomicU64, Ordering},
	},
	time::{Duration, Instant},
};

use AgenticChatDriver::{DefaultSelectors, Expressions, Failure, Format, Page, Selectors, Target};
use AgenticRuntime::{Call, Clock, FutureBox, Pace, Signal};
use serde_json::{Value, json};

/// What the fake answers for a thread read.
pub enum Answers {
	/// Successive reads; the last value repeats.
	Fixed(Vec<Value>),
	/// A thread whose newest answer never stops changing.
	Changing,
}

/// What the fake's release does.
pub enum Release {
	/// Finishes at once, the way a healthy release does.
	Quick,
	/// Never finishes, the way a hung one does.
	Hang,
}

/// Everything the fake answers with.
pub struct Script {
	/// The mode the composer reports before anything is pressed.
	pub Mode:Value,
	/// The mode it reports after the toggle is pressed.
	pub ModeAfterPress:Value,
	/// What pressing the toggle answers.
	pub Press:Value,
	/// The model label.
	pub Model:Value,
	/// The thread, read by read.
	pub Thread:Answers,
	/// The collapse control's label.
	pub Collapse:Value,
	/// The trace.
	pub Trace:Value,
	/// The sources.
	pub Sources:Value,
	/// What typing the question answers.
	pub Typed:Value,
	/// What pressing the send control answers.
	pub Submit:Value,
	/// The question the fake expects to be typed, so it can build the same
	/// expression the driver builds.
	pub Question:String,
	/// A failure for every open, instead of a page.
	pub FailOpen:Option<String>,
	/// A failure for every evaluate, instead of an answer.
	pub FailEvaluate:Option<String>,
	/// What the release does.
	pub Release:Release,
}

impl Script {
	/// A page that is in search mode, set to `Model`, and answers `Text`.
	pub fn New(Question:&str, Model:&str, Text:&str) -> Self {
		Self {
			Mode:json!("Search"),
			ModeAfterPress:json!("Search"),
			Press:json!(true),
			Model:json!(Model),
			Thread:Answers::Fixed(vec![json!({ "count": 0, "text": "" }), json!({ "count": 1, "text": Text })]),
			Collapse:Value::Null,
			Trace:json!([]),
			Sources:json!([]),
			Typed:json!(true),
			Submit:json!("sent"),
			Question:Question.to_string(),
			FailOpen:None,
			FailEvaluate:None,
			Release:Release::Quick,
		}
	}
}

/// The expression's identity, decided by the driver's own functions.
///
/// A read of the thread has two expressions - one per format - and both are a
/// read of the thread: the fake answers the same value either way, because the
/// shape it answers is the port's, not the page's.
pub fn KindOf(Selectors:&Selectors, Question:&str, Expression:&str) -> Kind {
	if Expression == Expressions::ReadMode(Selectors) {
		Kind::Mode
	} else if Expression == Expressions::PressMode(Selectors) {
		Kind::Press
	} else if Expression == Expressions::ReadModel(Selectors) {
		Kind::Model
	} else if Expression == Expressions::TypeQuestion(Selectors, Question) {
		Kind::Composer
	} else if Expression == Expressions::PressSubmit(Selectors, Question) {
		Kind::Submit
	} else if Expression == Expressions::ReadAnswers(Selectors, Format::Text)
		|| Expression == Expressions::ReadAnswers(Selectors, Format::Markdown)
	{
		Kind::Answers
	} else if Expression == Expressions::ReadCollapse(Selectors) {
		Kind::Collapse
	} else if Expression == Expressions::ReadTrace(Selectors) {
		Kind::Trace
	} else if Expression == Expressions::ReadSources(Selectors) {
		Kind::Sources
	} else {
		Kind::Unknown
	}
}

/// A page, scripted.
pub struct FakePage {
	Script:Mutex<Script>,
	Selectors:Selectors,
	/// Every expression kind the driver asked for, in order.
	pub Asked:Mutex<Vec<Kind>>,
	/// How many times the page was opened.
	pub Opens:AtomicU64,
	/// How many times it was released.
	pub Releases:AtomicU64,
	/// How many times the toggle was pressed.
	pub Presses:AtomicU64,
	/// How many times an answer was read.
	pub Reads:AtomicU64,
	/// The last deadline a call handed the page.
	pub Deadline:Mutex<Option<Instant>>,
}

impl FakePage {
	/// A fake with the default selectors.
	pub fn New(Script:Script) -> Arc<Self> { Self::With(DefaultSelectors(), Script) }

	/// A fake with the caller's selectors.
	pub fn With(Selectors:Selectors, Script:Script) -> Arc<Self> {
		Arc::new(Self {
			Script:Mutex::new(Script),
			Selectors,
			Asked:Mutex::new(Vec::new()),
			Opens:AtomicU64::new(0),
			Releases:AtomicU64::new(0),
			Presses:AtomicU64::new(0),
			Reads:AtomicU64::new(0),
			Deadline:Mutex::new(None),
		})
	}

	/// The kinds asked for so far.
	pub fn Asked(&self) -> Vec<Kind> { self.Asked.lock().unwrap().clone() }

	/// Whether the driver ever asked the page to type anything.
	pub fn Typed(&self) -> bool { self.Asked().contains(&Kind::Composer) }

	/// Whether the driver ever asked the page to send.
	pub fn Sent(&self) -> bool { self.Asked().contains(&Kind::Submit) }
}

impl Page for FakePage {
	fn Open<'a>(&'a self, Url:&'a str, Call:Call) -> FutureBox<'a, Result<Target, Failure>> {
		*self.Deadline.lock().unwrap() = Some(Call.Deadline);
		let Fail = self.Script.lock().unwrap().FailOpen.clone();
		Box::pin(async move {
			if let Some(Message) = Fail {
				return Err(Failure::Of(Message));
			}
			self.Opens.fetch_add(1, Ordering::SeqCst);
			Ok(Target { Url:Url.to_string() })
		})
	}

	fn Evaluate<'a>(
		&'a self,
		_Target:&'a Target,
		Expression:&'a str,
		Call:Call,
	) -> FutureBox<'a, Result<Value, Failure>> {
		*self.Deadline.lock().unwrap() = Some(Call.Deadline);
		let Script = self.Script.lock().unwrap();
		let Kind = KindOf(&self.Selectors, &Script.Question, Expression);
		self.Asked.lock().unwrap().push(Kind);
		if let Some(Message) = Script.FailEvaluate.clone() {
			return Box::pin(async move { Err(Failure::Of(Message)) });
		}
		let Answer = match Kind {
			Kind::Mode => {
				if self.Presses.load(Ordering::SeqCst) == 0 {
					Script.Mode.clone()
				} else {
					Script.ModeAfterPress.clone()
				}
			},
			Kind::Press => {
				self.Presses.fetch_add(1, Ordering::SeqCst);
				Script.Press.clone()
			},
			Kind::Model => Script.Model.clone(),
			Kind::Composer => Script.Typed.clone(),
			Kind::Submit => Script.Submit.clone(),
			Kind::Answers => {
				let Read = self.Reads.fetch_add(1, Ordering::SeqCst);
				match &Script.Thread {
					Answers::Changing => json!({ "count": 1, "text": format!("read {Read}") }),
					Answers::Fixed(Values) => {
						let At = (Read as usize).min(Values.len().saturating_sub(1));
						Values.get(At).cloned().unwrap_or_else(|| json!({ "count": 0, "text": "" }))
					},
				}
			},
			Kind::Collapse => Script.Collapse.clone(),
			Kind::Trace => Script.Trace.clone(),
			Kind::Sources => Script.Sources.clone(),
			Kind::Unknown => Value::Null,
		};
		Box::pin(async move { Ok(Answer) })
	}

	fn Dispose<'a>(&'a self, _Target:Target) -> FutureBox<'a, ()> {
		self.Releases.fetch_add(1, Ordering::SeqCst);
		let Hang = matches!(self.Script.lock().unwrap().Release, Release::Hang);
		Box::pin(async move {
			if Hang {
				std::future::pending::<()>().await;
			}
		})
	}
}

/// A clock the test owns, and a wait that advances it.
///
/// The wait moves the clock by at least a millisecond, so a poll loop with a
/// zero interval still terminates: time passing is the test's decision, not the

pub struct TestPace {

	Fire:Mutex<Option<Arc<Signal>>>,

	pub fn New() -> Arc<Self> { Arc::new(Self { Now:Mutex::new(Instant::now()), Fire:Mutex::new(None) }) }

	/// The same pace, which fires `Signal` on its first wait.
	pub fn Firing(self: &Arc<Self>, Signal:Arc<Signal>) -> Arc<Self> {

	}

}

impl Pace for TestPace {

		Box::pin(std::future::ready(()))
	}
}

/// A page, scripted by the fixture.
///
/// The script is the fixture's `page` object, kept whole, so a step's `before`
/// patch is the same one-level merge the TypeScript half applies. Every field is
/// flat, and `answersHangFrom` / `answersFailFrom` count *every* read of the
/// thread - the correlation read the driver takes before the send is read one -
/// so both halves index the same sequence.
struct FakePage {
	Page: Mutex<Value>,
	Question: String,
	Selectors: Selectors,
	Effects: Effects,
	/// Every expression kind the driver asked for, in order.
	Asked: Mutex<Vec<Kind>>,
	Opens: AtomicU64,
	Releases: AtomicU64,
	Presses: AtomicU64,
	Reads: AtomicU64,
}

impl FakePage {
	fn New(Fixture: &Value, Effects: Effects) -> Arc<Self> {
		Arc::new(Self {
			Page: Mutex::new(Fixture.get("page").cloned().unwrap_or_else(|| json!({}))),
			Question: string_at(Fixture, &["question", "text"]).unwrap_or_default(),
			Selectors: DefaultSelectors(),
			Effects,
			Asked: Mutex::new(Vec::new()),
			Opens: AtomicU64::new(0),
			Releases: AtomicU64::new(0),
			Presses: AtomicU64::new(0),
			Reads: AtomicU64::new(0),
		})
	}

	/// The script as it stands. Cloned rather than borrowed, so no lock is held
	/// across an await.
	fn State(&self) -> Value { self.Page.lock().unwrap().clone() }

	/// Apply the `page` half of a step's `before` patch, one level deep.
	fn Patch(&self, Values: &Value) {
		let mut Current = self.Page.lock().unwrap();
		let (Some(Current), Some(Patch)) = (Current.as_object_mut(), Values.as_object()) else {
			return;
		};
		for (Key, Value) in Patch {
			Current.insert(Key.clone(), Value.clone());
		}
	}

	fn Note(&self, Effect: String) { self.Effects.lock().unwrap().push(Effect); }

	/// The thread on its `Read`th read, counting from zero.
	fn ThreadAt(&self, State: &Value, Read: u64) -> Value {
		if bool_at(State, &["answersStall"]).unwrap_or(false) {
			return json!({ "count": 1, "text": format!("still writing {Read}") });
		}
		let Reads = State.get("answersReads").and_then(Value::as_array);
		match Reads {
			None => json!({ "count": 0, "text": "" }),
			Some(Reads) if Reads.is_empty() => json!({ "count": 0, "text": "" }),
			Some(Reads) => {
				let At = (Read as usize).min(Reads.len() - 1);
				Reads.get(At).cloned().unwrap_or(Value::Null)
			}
		}
	}
}

impl Page for FakePage {
	fn Open<'a>(&'a self, Url: &'a str, _Call: Call) -> FutureBox<'a, Result<Target, Failure>> {
		let State = self.State();
		self.Note(format!("open:{Url}"));
		match string_at(&State, &["open"]).unwrap_or_else(|| "answer".to_string()).as_str() {
			"hang" => return Box::pin(std::future::pending::<Result<Target, Failure>>()),
			"throw" => {
				let Message = string_at(&State, &["openFailure"]).unwrap_or_default();
				self.Note(format!("failed:open:{Message}"));
				return Box::pin(async move { Err(Failure::Of(Message)) });
			}
			_ => {}
		}
		self.Opens.fetch_add(1, Ordering::SeqCst);
		self.Note(format!("opened:{Url}"));
		Box::pin(async move { Ok(Target { Url: Url.to_string() }) })
	}

	fn Evaluate<'a>(
		&'a self,
		_Target: &'a Target,
		Expression: &'a str,
		_Call: Call,
	) -> FutureBox<'a, Result<Value, Failure>> {
		let State = self.State();
		let Kind = KindOf(&self.Selectors, &self.Question, Expression);
		self.Asked.lock().unwrap().push(Kind);
		self.Note(format!("evaluate:{}", Kind.Name()));

		if string_at(&State, &["evaluate"]).unwrap_or_else(|| "answer".to_string()) == "hang" {
			return Box::pin(std::future::pending::<Result<Value, Failure>>());
		}
		if string_at(&State, &["evaluate"]).unwrap_or_else(|| "answer".to_string()) == "throw" {
			let Message = string_at(&State, &["evaluateFailure"]).unwrap_or_default();
			self.Note(format!("failed:{}:{Message}", Kind.Name()));
			return Box::pin(async move { Err(Failure::Of(Message)) });
		}

		let Answer = match Kind {
			// A page that cannot name its mode is answered as the script says it,
			// whatever shape that is; a page that can is answered with the label
			// the expression would have folded.
			Kind::Mode => {
				let Label = if self.Presses.load(Ordering::SeqCst) == 0 {
					State.get("mode").cloned().unwrap_or(Value::Null)
				} else {
					State.get("modeAfterPress").cloned().unwrap_or(Value::Null)
				};
				match Label.as_str() {
					Some(Text) => json!(Text.to_lowercase()),
					None => Label,
				}
			}
			Kind::Press => {
				self.Presses.fetch_add(1, Ordering::SeqCst);
				json!(true)
			}
			Kind::Model => match State.get("model").and_then(Value::as_str) {
				Some(Label) => json!(Label.trim()),
				None => State.get("model").cloned().unwrap_or(Value::Null),
			},
			Kind::Composer => {
				if string_at(&State, &["composer"]).unwrap_or_else(|| "textarea".to_string())
					== "missing"
				{
					json!(false)
				} else {
					self.Note(format!("written:{}", self.Question));
					json!(true)
				}
			}
			Kind::Submit => {
				let Outcome =
					string_at(&State, &["submit"]).unwrap_or_else(|| "sent".to_string());
				match Outcome.as_str() {
					"sent" => self.Note("sent".to_string()),
					"disabled" => self.Note("refused:submit:disabled".to_string()),
					_ => {}
				}
				json!(Outcome)
			}
			Kind::Answers => {
				let Read = self.Reads.fetch_add(1, Ordering::SeqCst);
				let At = Read + 1;
				if number_at(&State, &["answersHangFrom"]) == Some(At) {
					return Box::pin(std::future::pending::<Result<Value, Failure>>());
				}
				if number_at(&State, &["answersFailFrom"]) == Some(At) {
					let Message = string_at(&State, &["answersFailMessage"]).unwrap_or_default();
					self.Note(format!("failed:answers:{Message}"));
					return Box::pin(async move { Err(Failure::Of(Message)) });
				}
				self.ThreadAt(&State, Read)
			}
			Kind::Collapse => match State.get("collapse") {
				Some(Value::String(Label)) => json!(Label.trim()),
				Some(Other) => Other.clone(),
				None => Value::Null,
			},
			Kind::Trace => State.get("trace").cloned().unwrap_or_else(|| json!([])),
			Kind::Sources => State.get("sources").cloned().unwrap_or_else(|| json!([])),
			Kind::Unknown => Value::Null,
		};

		Box::pin(async move { Ok(Answer) })
	}

	fn Dispose<'a>(&'a self, Target: Target) -> FutureBox<'a, ()> {
		self.Releases.fetch_add(1, Ordering::SeqCst);
		self.Note(format!("release:{}", Target.Url));
		let Hang = string_at(&self.State(), &["release"]).unwrap_or_else(|| "answer".to_string())
			== "hang";
		Box::pin(async move {
			if Hang {
				std::future::pending::<()>().await;
			}
		})
	}
}

/// Read a dotted path out of a JSON value.

}

/// Merge a patch into a fixture the way `scenarios/lib/load.mjs` does: section by
/// section, one level deep. A patch that is not an object replaces the section,
/// and a section that is not an object is replaced rather than merged into.
fn merge(Base: &Value, Patch: &Value) -> Value {
	if Patch.is_null() {
		return Base.clone();
	}
	let (Some(BaseObject), Some(PatchObject)) = (Base.as_object(), Patch.as_object()) else {
		return Patch.clone();
	};
	let mut Merged = BaseObject.clone();
	for (Key, Patched) in PatchObject {
		let Next = match Patched.as_object() {
			Some(Values) => {
				let mut Section = Merged
					.get(Key)
					.and_then(Value::as_object)
					.cloned()
					.unwrap_or_default();
				for (Field, Value) in Values {
					Section.insert(Field.clone(), Value.clone());
				}
				Value::Object(Section)
			}
			None => Patched.clone(),
		};
		Merged.insert(Key.clone(), Next);
	}
	Value::Object(Merged)
}

/// Run one scenario, and answer what it did: the result, the ordered effects and
/// how many failures were reported.
fn run_case(Fixture: &Value, Entry: &Value) -> Value {

	let Effects: Effects = Arc::new(Mutex::new(Vec::new()));
	let Reports = Arc::new(AtomicU64::new(0));

	let Signal = Signal::New();
	let Wanted = string_at(&Fixture, &["driver", "signal"]).unwrap_or_else(|| "fresh".to_string());
	let Pace = match Wanted.as_str() {
		"aborted" => {
			Signal.Fire();
			TestPace::New()
		}
		Kind if Kind.starts_with("after:") => TestPace::New().Firing(Signal.clone()),
		_ => TestPace::New(),
	};

	let Report: Arc<dyn Fn(&Target, &Error) + Send + Sync> = {
		let Effects = Effects.clone();
		let Reports = Reports.clone();
		Arc::new(move |_Target: &Target, Why: &Error| {
			Effects
				.lock()
				.unwrap()
				.push(format!("report:{}", Why.Code().Name()));
			Reports.fetch_add(1, Ordering::SeqCst);
		})
	};

	let Driver = Create(
		Page.clone(),
		Options {
			Url: string_at(&Fixture, &["url"]).unwrap_or_default(),
			Selectors: DefaultSelectors(),
			Poll: Duration::from_millis(number_at(&Fixture, &["driver", "pollMs"]).unwrap_or(0)),
			DisposeBudget: Duration::from_millis(
				number_at(&Fixture, &["driver", "disposeTimeoutMs"]).unwrap_or(40),
			),
		},
		Pace.clone(),
	)
	.Reporting(Report);

	let Budget = Duration::from_millis(number_at(&Fixture, &["driver", "timeoutMs"]).unwrap_or(400));
	let Mode = match string_at(&Fixture, &["question", "mode"]).unwrap_or_else(|| "search".to_string()).as_str() {
		"search" => Mode::Search,
		Other => Mode::Other(Other.to_string()),
	};
	let Question = Question {
		Text: string_at(&Fixture, &["question", "text"]).unwrap_or_default(),
		Mode,
		Model: string_at(&Fixture, &["question", "model"]).unwrap_or_default(),
	};

	let Steps = Entry.get("steps").and_then(Value::as_array).cloned().unwrap_or_default();
	let mut Result = json!({ "kind": "void" });

	for Step in &Steps {
		if let Some(Before) = Step.get("before") {
			if let Some(Values) = Before.get("page") {
				Page.Patch(Values);
			}
		}
		let Op = string_at(Step, &["op"]).unwrap_or_default();
		assert_eq!(Op, "ask", "unknown chat step {Op}");

		let Cancellation: Arc<dyn Cancellation> = match Wanted.as_str() {
			"aborted" => Signal.clone(),
			Kind if Kind.starts_with("after:") => Signal.clone(),
			_ => Arc::new(Never),
		};
		let Call = Call { Deadline: Pace.Now() + Budget, Cancellation };

		match BlockOn(Driver.Ask(Question.clone(), Call)) {
			Ok(Answer) => Result = answer_of(&Answer),
			Err(Why) => {
				Result = error_of(&Why);
				break;
			}
		}
	}

	json!({
		"result": Result,
		"effects": Effects.lock().unwrap().clone(),
		"reports": Reports.load(Ordering::SeqCst),
	})
}

/// What one observation got wrong, as sentences. Empty means the case agreed.
fn mismatches(Observed: &Value, Expect: &Value) -> Vec<String> {
	let mut Wrong = Vec::new();
	let Want = Expect.get("result").cloned().unwrap_or(Value::Null);
	let Got = Observed.get("result").cloned().unwrap_or(Value::Null);
	let Kind = |Document: &Value| Document.get("kind").and_then(Value::as_str).unwrap_or("").to_string();

	if Kind(&Want) != Kind(&Got) {
		Wrong.push(format!("result kind: wanted {}, got {}", Kind(&Want), Kind(&Got)));
	} else if Kind(&Want) == "error" {
		if Want.get("code") != Got.get("code") {
			Wrong.push(format!("result code: wanted {:?}, got {:?}", Want.get("code"), Got.get("code")));
		}
		if let Some(Message) = Want.get("message").and_then(Value::as_str) {
			if !Got.get("message").and_then(Value::as_str).unwrap_or("").contains(Message) {
				Wrong.push(format!("result message: wanted to contain {Message:?}, got {:?}", Got.get("message")));
			}
		}
		if let Some(Token) = Want.get("observed").and_then(Value::as_str) {
			let Both = format!(
				"{} {}",
				Got.get("message").and_then(Value::as_str).unwrap_or(""),
				Got.get("detail").and_then(Value::as_str).unwrap_or("")
			);
			if !Both.contains(Token) {
				Wrong.push(format!("result observation: wanted {Token:?} in message or detail, got {Both:?}"));
			}
		}
	} else if Kind(&Want) == "answer" {
		for Field in ["text", "model", "mode"] {
			if let Some(Wanted) = Want.get(Field) {
				if Got.get(Field) != Some(Wanted) {
					Wrong.push(format!("result {Field}: wanted {Wanted}, got {:?}", Got.get(Field)));
				}
			}
		}
		for Field in ["trace", "sources"] {
			if let Some(Wanted) = Want.get(Field) {
				if Got.get(Field) != Some(Wanted) {
					Wrong.push(format!("result {Field}: wanted {Wanted}, got {:?}", Got.get(Field)));
				}
			}
		}
	}

	if let Some(Effects) = Expect.get("effects") {
		if Effects != Observed.get("effects").unwrap_or(&Value::Null) {
			Wrong.push(format!("effects: wanted {Effects}, got {}", Observed.get("effects").unwrap_or(&Value::Null)));
		}
	}
	if let Some(Reports) = Expect.get("reports").and_then(Value::as_u64) {
		if Some(Reports) != Observed.get("reports").and_then(Value::as_u64) {
			Wrong.push(format!("reports: wanted {Reports}, got {:?}", Observed.get("reports")));
		}
	}
	Wrong
}

/// Every scenario, against its expectation, with what was observed left behind
/// for the differential run to compare against the TypeScript half.
#[test]
fn every_scenario_agrees_with_its_expectation() {
	let mut Observed: BTreeMap<String, Value> = BTreeMap::new();
	let mut Wrong: Vec<String> = Vec::new();
	let mut Seen = 0usize;

	for File in scenario_files() {
		let Fixture = File.get("fixture").cloned().unwrap_or(Value::Null);
		let Cases = File.get("cases").and_then(Value::as_array).cloned().unwrap_or_default();
		for Entry in Cases {
			if skipped_on_rust(&Entry) {
				continue;
			}
			let Id = string_at(&Entry, &["id"]).unwrap_or_else(|| "<unnamed>".to_string());
			let One = run_case(&Fixture, &Entry);
			let Expect = Entry.get("expect").cloned().unwrap_or(Value::Null);
			for Sentence in mismatches(&One, &Expect) {
				Wrong.push(format!("{Id}: {Sentence}"));
			}
			Observed.insert(Id, One);
			Seen += 1;
		}
	}

	if let Ok(Path) = std::env::var("AGENTIC_PARITY_OUT") {
		let Document =
			serde_json::to_string_pretty(&Value::Object(Observed.clone().into_iter().collect::<Map<String, Value>>()))
				.unwrap_or_default();
		let _ = std::fs::write(Path, Document);
	}

	assert!(Seen > 0, "no scenario files were found");
	assert!(
		Wrong.is_empty(),
		"the Rust port disagrees with the scenario files:\n{}",
		Wrong.join("\n")
	);
}

//! Judging measurements against invariants. Pure, and it never fails.
//!
//! The absent list is built first and independently of the invariants, because
//! a selector that matched nothing fails the run even when every row that reads
//! through it happened to hold: the design named an element, and the page does
//! not have it.

use std::panic::{AssertUnwindSafe, catch_unwind};

use serde_json::Value;

use crate::Struct::Port::{Evidence, Invariant, Measurement, Outcome, Verdict, Want};

/// Whether one invariant held, without letting any part of it escape.
///
/// A reading that fails, a derivation that fails and a comparator that panics
/// are each recorded as a row that did not hold, because a page defect must not
/// be able to end a run as an exception.
pub fn Judge(Invariant:&Invariant, Measurements:&[Measurement]) -> Evidence {
	let Want = match &Invariant.Want {
		Want::Value(Value) => Outcome::Value(Value.clone()),
		Want::Derive(Derive) => {
			match Derive(Measurements) {
				Ok(Value) => Outcome::Value(Value),
				Err(Why) => Outcome::Threw(Why),
			}
		},
	};
	let Got = match (Invariant.Read)(Measurements) {
		Ok(Value) => Outcome::Value(Value),
		Err(Why) => Outcome::Threw(Why),
	};
	let Held = match (&Got, &Want) {
		(Outcome::Value(Got), Outcome::Value(Want)) => {
			match &Invariant.Test {
				Some(Test) => catch_unwind(AssertUnwindSafe(|| Test(Got, Want))).unwrap_or(false),
				None => Got == Want,
			}
		},
		// A comparison that could not be made has not been satisfied. The row still
		// carries what was read, which is what a reader needs.
		_ => false,
	};
	Evidence { Label:Invariant.Label.clone(), Got, Want, Held }
}

/// Judge a set of measurements.
pub fn Check(Invariants:&[Invariant], Measurements:&[Measurement]) -> Verdict {
	let mut Absent:Vec<String> = Vec::new();
	for Measurement in Measurements {
		if Measurement.Present {
			continue;
		}
		if !Absent.iter().any(|Selector| Selector == &Measurement.Selector) {
			Absent.push(Measurement.Selector.clone());
		}
	}
	let Evidence:Vec<Evidence> = Invariants.iter().map(|Invariant| Judge(Invariant, Measurements)).collect();
	Verdict { Held:Absent.is_empty() && Evidence.iter().all(|Row| Row.Held), Evidence, Absent }
}

/// The measurements a selector names, for a caller writing a reading.
pub fn Measured<'a>(Measurements:&'a [Measurement], Selector:&str) -> Option<&'a Measurement> {
	Measurements.iter().find(|Measurement| Measurement.Selector == Selector)
}

/// A measurement's value as a number, for a caller writing a reading.
pub fn Number(Measurements:&[Measurement], Selector:&str) -> Result<Value, String> {
	let Measurement = Measured(Measurements, Selector).ok_or_else(|| format!("nothing was measured for {Selector}"))?;
	if !Measurement.Present {
		return Err(format!("{Selector} is not in the page"));
	}
	match &Measurement.Value {
		Value::Number(Number) => Ok(Value::Number(Number.clone())),
		Other => Err(format!("{Selector} read {Other}, which is not a number")),
	}
}

//! Reconciling what a transport answered with what the caller asked for.
//!
//! The answer is matched by selector rather than by position, so a provider
//! that batches or reorders cannot hand an invariant another element's number.
//! A selector the transport did not answer for is a broken transport, not an
//! absent element: absence is something the page was observed to have, and an
//! answer that never arrived was not observed at all. Reporting it as absent
//! would invent a page defect out of a provider's omission.

use serde_json::Value;

use crate::Struct::{Error::Error, Port::Measurement};

/// A value as a detail line.
///
/// A provider that answered with the wrong shape is reported by what it sent,
/// not by `[object Object]`, because the detail is the only thing that says
/// which part of the contract it broke.
pub fn Describe(Value:&Value) -> String {
	match Value {
		Value::Null => "null".to_string(),
		Value::String(Text) => serde_json::to_string(Text).unwrap_or_else(|_| format!("{Text:?}")),
		Value::Number(_) | Value::Bool(_) => Value.to_string(),
		Value::Array(Items) => format!("an array of {}", Items.len()),
		Value::Object(_) => "an object".to_string(),
	}
}

/// What a measurement's value may be.
///
/// A provider that answers an object has not finished serialising what it read,
/// and flattening that to `Null` would report "nothing to report" about an
/// element that did report something. The port refuses it instead, naming the
/// selector, so the provider is fixed rather than silently trusted.
fn AsValue(Value:&Value, Selector:&str) -> Result<Value, Error> {
	match Value {
		Value::Null | Value::Number(_) | Value::String(_) => Ok(Value.clone()),
		Other => {
			Err(Error::NotReady(
				"the transport answered with a value the port does not carry",
				format!(
					"{:?} read {}; a measurement is a number, a string or null",
					Selector,
					Describe(Other)
				),
			))
		},
	}
}

/// One entry of a transport's answer, as the port defines a measurement.
fn AsMeasurement(Value:&Value) -> Result<Measurement, Error> {
	let Some(Object) = Value.as_object() else {
		return Err(Error::NotReady(
			"the transport answered with a measurement that is not an object",
			format!("it answered {}", Describe(Value)),
		));
	};
	let Selector = match Object.get("selector") {
		Some(Value::String(Selector)) if !Selector.is_empty() => Selector.clone(),
		_ => {
			return Err(Error::NotReady(
				"the transport answered with a measurement for no selector",
				format!("it answered {}", Describe(Value)),
			));
		},
	};
	let Present = match Object.get("present") {
		Some(Value::Bool(Present)) => *Present,
		Other => {
			return Err(Error::NotReady(
				"the transport answered without saying whether the element was there",
				format!(
					"{:?} carried {} as present",
					Selector,
					Other.map(Describe).unwrap_or_else(|| "nothing".to_string())
				),
			));
		},
	};
	// An absent element cannot carry a value: a stale number travelling with
	// `present: false` is what an invariant that forgot to check would read.
	let Value = if Present {
		AsValue(Object.get("value").unwrap_or(&Value::Null), &Selector)?
	} else {
		Value::Null
	};
	Ok(Measurement { Selector, Present, Value })
}

/// One measurement per asked selector, in the order it was asked for.
pub fn Reconcile(Asked:&[String], Answered:&[Value]) -> Result<Vec<Measurement>, Error> {
	let mut BySelector:Vec<Measurement> = Vec::new();
	for Entry in Answered {
		let Measurement = AsMeasurement(Entry)?;
		BySelector.push(Measurement);
	}
	let mut Ordered = Vec::new();
	for Selector in Asked {
		let Some(Found) = BySelector.iter().find(|Measurement| &Measurement.Selector == Selector) else {
			return Err(Error::NotReady(
				"the transport did not answer for a selector it was asked about",
				format!("nothing was measured for {Selector:?}"),
			));
		};
		// A copy per position: two entries naming one selector must not be one
		// mutable object, or a caller editing one edits the other.
		Ordered.push(Found.clone());
	}
	Ok(Ordered)
}

//! The three failure modes `module.json` declares, and no others.
//!
//! One enum, one variant per code, `detail` kept apart from the message in
//! every one of them.

use std::fmt;

use crate::Struct::Port::Verdict;

/// The codes, in the order the manifest names them.
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub enum Code {
	/// An element the design names is not in the page.
	Absent,
	/// An invariant did not hold.
	Broke,
	/// The page was never read.
	NotReady,
}

impl Code {
	/// The string the manifest uses.
	pub const fn Name(self) -> &'static str {
		match self {
			Code::Absent => "absent-element",
			Code::Broke => "invariant-broke",
			Code::NotReady => "not-ready",
		}
	}

	/// Every code, so a harness can compare the manifest against the type.
	pub const fn All() -> [Code; 3] { [Code::Absent, Code::Broke, Code::NotReady] }

	/// The code a manifest string names, if it names one of the three.
	pub fn Parse(Value:&str) -> Option<Code> { Code::All().into_iter().find(|Code| Code.Name() == Value) }
}

/// The one error type this capability has.
///
/// `NotReady` is the only code `Measure` fails with, because the manifest has
/// no fourth code and "the page was never read" is what a dead transport, a
/// failure mid-measurement, a passed deadline and a cancellation all mean to
/// this capability. `detail` says which one happened.
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
	/// An element the design names is not in the page.
	Absent {
		/// The sentence a human reads.
		Message:String,
		/// The selectors, kept apart from the sentence.
		Detail:String,
	},
	/// An invariant did not hold.
	Broke {
		/// The sentence a human reads.
		Message:String,
		/// The invariants that broke.
		Detail:String,
	},
	/// The page was never read.
	NotReady {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
}

impl Error {
	/// A page that was never read.
	pub fn NotReady(Message:impl Into<String>, Detail:impl Into<String>) -> Self {
		Error::NotReady { Message:Message.into(), Detail:Detail.into() }
	}

	/// A verdict as the error the manifest names, or `None` when it held.
	///
	/// `Check` never fails, because a verdict is a result. A caller that must
	/// fail loudly - a command that exits non-zero - asks for the error
	/// here. Absence is reported before a broken invariant, because a page
	/// that lost a part is the more actionable failure and the rows that
	/// read through it are consequences of it.
	pub fn From(Verdict:&Verdict) -> Option<Error> {
		if !Verdict.Absent.is_empty() {
			return Some(Error::Absent {
				Message:format!("{} element(s) the design names are absent", Verdict.Absent.len()),
				Detail:Verdict.Absent.join(", "),
			});
		}
		let Broken:Vec<&str> = Verdict
			.Evidence
			.iter()
			.filter(|Row| !Row.Held)
			.map(|Row| Row.Label.as_str())
			.collect();
		if !Broken.is_empty() {
			return Some(Error::Broke {
				Message:format!("{} invariant(s) broke", Broken.len()),
				Detail:Broken.join(", "),
			});
		}
		None
	}

	/// The code this failure carries.
	pub const fn Code(&self) -> Code {
		match self {
			Error::Absent { .. } => Code::Absent,
			Error::Broke { .. } => Code::Broke,
			Error::NotReady { .. } => Code::NotReady,
		}
	}

	/// The sentence a human reads.
	pub fn Message(&self) -> &str {
		match self {
			Error::Absent { Message, .. } | Error::Broke { Message, .. } | Error::NotReady { Message, .. } => Message,
		}
	}

	/// What was observed, which may be empty.
	pub fn Detail(&self) -> &str {
		match self {
			Error::Absent { Detail, .. } | Error::Broke { Detail, .. } | Error::NotReady { Detail, .. } => Detail,
		}
	}
}

impl fmt::Display for Error {
	fn fmt(&self, Formatter:&mut fmt::Formatter<'_>) -> fmt::Result {
		write!(Formatter, "{}: {}", self.Code().Name(), self.Message())
	}
}

impl std::error::Error for Error {}

--- FILE: Crate/DesignVerification/Source/Struct/Port.rs ---
//! The port `agentic.verify.design` promises, written the way its consumer asks
//! for it.
//!
//! `module.json` names three errors - `absent-element`, `invariant-broke` and
//! `not-ready` - and [`crate::Struct::Error`] is where they live. A caller
//! narrows on the variant; anything a provider does that is not one of the
//! three is mapped onto one of them at the boundary.
//!
//! The port states more than data shapes, because two implementations that
//! agreed on the shapes alone would still disagree about what a failure is:
//!
//! - **An absent element is a failure, never a precondition.** `Check` reports
//!   the selector in `Absent` and answers `Held: false`. It never fails, and it
//!   never reports "could not run": the run happened, and the page was wrong.
//! - **This package holds no design.** No selector, no pixel value, no colour.
//! - **`Measure` never fails for a missing element.** A selector that matches
//!   nothing is a measurement with `Present: false`, because "there is nothing
//!   there" is an observation. A dead transport is the only failure, and it is
//!   `not-ready`, so the two answers cannot be confused.
//! - **Evidence is per invariant.** One entry per invariant, in the order the
//!   caller gave them, so a failure names which invariant broke and what it
//!   read.

use AgenticRuntime::{Call, FutureBox};
use serde_json::Value;

/// One element, as the page answered for it.
///
/// The type exists to keep "not observed" apart from a real zero, `false` or
/// empty string: a measurement engine that cannot say "there is nothing there"
/// produces evidence that reads as a measurement.
#[derive(Clone, Debug, PartialEq)]
pub struct Measurement {
	/// The selector this was asked about, as the caller wrote it.
	pub Selector:String,
	/// Whether the element was there. `false` is an observation, not an error.
	pub Present:bool,
	/// What was read from it - a width, a count, a label - or `Null`.
	///
	/// An absent element always carries `Null`, so a stale number cannot travel
	/// with `Present: false` and be read by an invariant that forgot to check.
	/// A present element that read `Null` is a value, not an absence: the
	/// element was there and had nothing to report.
	pub Value:Value,
}

impl Measurement {
	/// A measurement of an element that was there.
	pub fn Present(Selector:impl Into<String>, Value:Value) -> Self {
		Self { Selector:Selector.into(), Present:true, Value }
	}

	/// A measurement of an element that was not there. It cannot carry a value.
	pub fn Absent(Selector:impl Into<String>) -> Self {
		Self { Selector:Selector.into(), Present:false, Value:Value::Null }
	}
}

/// A reading of the measurements, or why it could not be read.
///
/// A reading that fails - because it dereferenced a measurement it assumed was
/// there - is a failed invariant, not a blocked run.
pub type Read = Box<dyn Fn(&[Measurement]) -> Result<Value, String> + Send + Sync>;

/// A derivation of an expectation from the measurements themselves.
pub type Derive = Box<dyn Fn(&[Measurement]) -> Result<Value, String> + Send + Sync>;

/// A comparison of a reading against an expectation.
pub type Test = Box<dyn Fn(&Value, &Value) -> bool + Send + Sync>;

/// What an invariant's expectation is.
///
/// Either a value, compared for equality, or a function of the measurements
/// when the expectation is derived from the page itself - "the rail sits level
/// with the drawer" wants the drawer's own top, and no constant states it.
pub enum Want {
	/// A value the reading must equal.
	Value(Value),
	/// A reading derived from the measurements. A derivation that fails counts
	/// as not held, and the row records why.
	Derive(Derive),
}

impl std::fmt::Debug for Want {
	fn fmt(&self, Formatter:&mut std::fmt::Formatter<'_>) -> std::fmt::Result {
		match self {
			Want::Value(Value) => Formatter.debug_tuple("Value").field(Value).finish(),
			Want::Derive(_) => Formatter.write_str("Derive(..)"),
		}
	}
}

/// One thing the design says must be true of the page.
pub struct Invariant {
	/// What the invariant is about, as a reader of a failure should hear it.
	pub Label:String,
	/// What to read out of the measurements.
	pub Read:Read,
	/// What the reading must equal, or how it is derived.
	pub Want:Want,
	/// How to compare, for an invariant that is a relation rather than an
	/// equality. Absent means equality; a comparator that panics counts as
	/// not held, because a comparison that could not be made has not been
	/// satisfied.
	pub Test:Option<Test>,
}

impl std::fmt::Debug for Invariant {
	fn fmt(&self, Formatter:&mut std::fmt::Formatter<'_>) -> std::fmt::Result {
		Formatter
			.debug_struct("Invariant")
			.field("Label", &self.Label)
			.field("Want", &self.Want)
			.finish_non_exhaustive()
	}
}

impl Invariant {
	/// An invariant that reads one value and requires it to equal another.
	pub fn Equal(
		Label:impl Into<String>,
		Read:impl Fn(&[Measurement]) -> Result<Value, String> + Send + Sync + 'static,
		Want:Value,
	) -> Self {
		Self { Label:Label.into(), Read:Box::new(Read), Want:Want::Value(Want), Test:None }
	}

	/// An invariant whose expectation is derived from the measurements.
	pub fn Derived(
		Label:impl Into<String>,
		Read:impl Fn(&[Measurement]) -> Result<Value, String> + Send + Sync + 'static,
		Derive:impl Fn(&[Measurement]) -> Result<Value, String> + Send + Sync + 'static,
	) -> Self {
		Self {
			Label:Label.into(),
			Read:Box::new(Read),
			Want:Want::Derive(Box::new(Derive)),
			Test:None,
		}
	}

	/// The same invariant, compared by the caller's own relation.
	pub fn Tested(mut self, Test:impl Fn(&Value, &Value) -> bool + Send + Sync + 'static) -> Self {
		self.Test = Some(Box::new(Test));
		self
	}
}

/// One side of a comparison: what was read, or why it could not be.
///
/// A reading that failed is recorded as `Threw`, which cannot be mistaken for a
/// value it read - the Rust port's version of the TypeScript `{ threw: message
/// }`.
#[derive(Clone, Debug, PartialEq)]
pub enum Outcome {
	/// A value that was read, or an expectation that was derived.
	Value(Value),
	/// Why it could not be read or derived.
	Threw(String),
}

impl Outcome {
	/// Whether this side produced a value.
	pub fn Held(&self) -> bool { matches!(self, Outcome::Value(_)) }
}

/// One invariant's row in a verdict: what it read, what it wanted, and whether.
#[derive(Clone, Debug, PartialEq)]
pub struct Evidence {
	/// The invariant's own label, copied unchanged.
	pub Label:String,
	/// What the invariant read.
	pub Got:Outcome,
	/// What it was compared against.
	pub Want:Outcome,
	/// Whether the comparison held.
	pub Held:bool,
}

/// The whole answer to one run.
///
/// `Held` is true only when nothing was absent and every invariant held, so a
/// caller can branch on one field and still read the reason from the others.
#[derive(Clone, Debug, Default, PartialEq)]
pub struct Verdict {
	/// Every invariant held and every selector matched something.
	pub Held:bool,
	/// One row per invariant, in the order the caller gave them.
	pub Evidence:Vec<Evidence>,
	/// Every selector that was observed to match nothing, once each.
	pub Absent:Vec<String>,
}

/// What a transport reports when it fails.
///
/// A transport is not this capability, so it does not speak this capability's
/// three codes. It says what it observed, and the capability maps that onto
/// `not-ready` with the observation in `detail`.
#[derive(Clone, Debug, Default, PartialEq, Eq)]
pub struct Failure {
	/// The sentence a human reads.
	pub Message:String,
	/// What was observed.
	pub Detail:String,
}

impl Failure {
	/// A failure with a sentence and nothing observed.
	pub fn Of(Message:impl Into<String>) -> Self { Self { Message:Message.into(), Detail:String::new() } }
}

/// The browser as this capability needs it: read every named element.
///
/// The transport answers the *wire* shape - one JSON object per selector,
/// carrying `selector`, `present` and `value` - and the capability validates
/// it. That is the TypeScript port's arrangement, kept because it is what makes
/// the classification checkable: a provider is not this crate, so a provider
/// that answers a list of numbers, an object where a number belongs, or no
/// answer for a selector it was asked about is refused with `not-ready` rather
/// than trusted.
///
/// What the port demands of an implementation:
///
/// - one entry per selector it was asked about, carrying that selector, so the
///   capability can match by `Selector` rather than by position;
/// - a selector that matches nothing is `present: false` - an observation, not
///   a failure. A provider that fails because an element is missing turns a
///   page defect into a failure of the observer;
/// - `value` is a number, a string or `null`. Anything else is a provider that
///   has not finished serialising what it read, and it is refused rather than
///   flattened, because flattening it to `null` would read as "nothing to
///   report";
/// - a failure means the page was never read, and it is reported as
///   `not-ready`.
pub trait Browse: Send + Sync {
	/// Read every named element, or report that it is not in the page.
	fn Measure<'a>(&'a self, Selectors:&'a [String], Call:Call) -> FutureBox<'a, Result<Vec<Value>, Failure>>;
}

/// The one port this capability offers.
pub trait Verify: Send + Sync {
	/// Read the named elements from the page, one measurement per selector.
	fn Measure<'a>(
		&'a self,
		Selectors:&'a [String],
		Call:Call,
	) -> FutureBox<'a, Result<Vec<Measurement>, crate::Struct::Error::Error>>;
	/// Judge a set of measurements against a set of invariants. Pure, and it
	/// never fails: a verdict is a result, not an exception.
	fn Check(&self, Invariants:&[Invariant], Measurements:&[Measurement]) -> Verdict;
}

//! `agentic.verify.design` over a transport the composition root supplies.
//!
//! Two verbs, and only the first touches a page:
//!
//! 1. **`Measure` asks and bounds.** The selectors go to the transport as they
//!    were given, the answer is matched back by selector, and the deadline is
//!    enforced on this side as well as the provider's - a transport that
//!    ignores its budget must not be able to hang a caller. The one failure is
//!    `not-ready`: the page was never read. An element that is not in the page
//!    is a measurement, never a failure.
//! 2. **`Check` is pure.** It reads no page and keeps no state, so the same
//!    measurements answer the same verdict twice. It cannot fail either: a
//!    reading that fails, an expectation that fails and a comparator that
//!    panics are each recorded as a row that did not hold, because a page
//!    defect must not be able to end a run as an exception.
//!
//! Nothing here names a selector, a length or a colour. The transport supplies
//! what a page says; the caller supplies what the design expects.

use std::sync::Arc;

use AgenticRuntime::{Bound, Call, Expired, FutureBox, Pace};

use crate::{
	Fn::{Judge::Check, Measure::Reconcile},
	Struct::{
		Error::Error,
		Port::{Browse, Invariant, Measurement, Verdict, Verify},
	},
};

/// A `Verify` over one transport.
///
/// Nothing here opens a browser or reads an environment variable: the transport
/// is the composition root's to supply, and the fake in `Test/` is the proof
/// that the rules hold without one.
pub struct Verifier {
	Browse:Arc<dyn Browse>,
	Pace:Arc<dyn Pace>,
}

/// Build a `Verify` over one transport, on the pace the caller supplied.
pub fn Create(Browse:Arc<dyn Browse>, Pace:Arc<dyn Pace>) -> Verifier { Verifier { Browse, Pace } }

impl Verify for Verifier {
	fn Measure<'a>(&'a self, Selectors:&'a [String], Call:Call) -> FutureBox<'a, Result<Vec<Measurement>, Error>> {
		Box::pin(async move {
			// Nothing asked is nothing read: a no-op request must not touch a
			// page.
			if Selectors.is_empty() {
				return Ok(Vec::new());
			}
			if Call.Cancelled() {
				return Err(Error::NotReady(
					"measure was cancelled before it started",
					"the caller's signal had already fired",
				));
			}
			if Call.Expired(self.Pace.as_ref()) {
				return Err(Error::NotReady(
					"measure exceeded its deadline",
					"the transport was still working when the budget ran out",
				));
			}
			let Work = self.Browse.Measure(Selectors, Call.clone());
			let Answered = match Bound(Work, &Call, self.Pace.as_ref()).await {
				Ok(Ok(Answered)) => Answered,
				Ok(Err(Failure)) => {
					return Err(Error::NotReady(
						"the transport failed while measuring the page",
						Failure.Message,
					));
				},
				Err(Expired::Timeout) => {
					return Err(Error::NotReady("measure exceeded its deadline", "the transport never answered"));
				},
				Err(Expired::Cancelled) => {
					return Err(Error::NotReady(
						"measure was cancelled by the caller",
						"the signal fired while the page was being read",
					));
				},
			};
			Reconcile(Selectors, &Answered)
		})
	}

	fn Check(&self, Invariants:&[Invariant], Measurements:&[Measurement]) -> Verdict { Check(Invariants, Measurements) }
}
