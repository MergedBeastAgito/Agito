/**
 * The archive's tables. Reducers live in `index.ts`: they encode
 * behaviour, and behaviour is not derived from a field list.
 */

/** One row per post. Media and observations live in their own tables. */
const post = table(
	{ name: "post", public: true },
	{
		/** X status id - the same key the file store uses, and the filename it becomes. */
		id: t.string().primaryKey(),
		/** The permalink the page showed. */
		url: t.string(),
		/** The author's handle; the `author` table is keyed by it. */
		author: t.string().index(),
		/** The post's text, as the page rendered it. */
		text: t.string(),
		/** When the post was written, when the page showed it. Null when it did not. */
		postedAt: t.string().optional(),
		/** Reply count at `lastScrapedAt`. */
		replies: t.u32(),
		/** Repost count at `lastScrapedAt`. */
		reposts: t.u32(),
		/** Like count at `lastScrapedAt`. */
		likes: t.u32(),
		/** View count at `lastScrapedAt`. */
		views: t.u32(),
		/** When this pipeline first saw the post - never when it was saved. */
		firstSeenAt: t.string(),
		/** When the post's fields were last read from the page. */
		lastScrapedAt: t.string(),
		/** The only field that decides whether the post may reach the site. */
		visibility: t.string().index(),
		/** The list slugs as a JSON array, because their number is the person's. */
		lists: t.string(),
	},
);

/** One row per account seen in the archive. */
const author = table(
	{ name: "author", public: true },

		/** The handle without its `@`. */
		handle: t.string().primaryKey(),

index.ts:

import { t } from "spacetimedb/server";
// This sub-package resolves imports the way the SpacetimeDB module bundler
// does - without an extension - which is why it is the one place in the
// repository that does not spell `.ts`.
import { spacetimedb } from "./Schema";

/**
 * Timeline archive, live view.
 *
 * The tables are rendered from `Maintain/Source/Model/Schema/Entities.ts` into
 * `Schema.ts`; this file holds only behaviour. Three rules survive from the file
 * store, and each one is a reducer's job rather than a column's:
 *
 * 1. Observations and events are append-only. A re-scrape updates the post's
 *    metrics in place but never rewrites history.
 * 2. `visibility` is the only field that decides what may reach the public site.
 * 3. Every derived row id is built from its content, so replaying the same
 *    scrape is idempotent instead of duplicating rows.
 *
 * There is exactly one reducer that writes a post - `observe` - because a
 * second write path is a second set of rules waiting to disagree with the first.
 * `store publish`, `store hide` and the backfill all go through it.
 */

export default spacetimedb;

export const init = spacetimedb.init((_ctx) => {
	// Nothing to seed: the archive arrives through `observe`.
});

export const onConnect = spacetimedb.clientConnected((_ctx) => {});
export const onDisconnect = spacetimedb.clientDisconnected((_ctx) => {});

/**
 * Write one post record and append the observation that produced it.
 *
 * The arguments are the record, not a page reading: a caller that has already
 * folded a reading into the archive record (the scrape path) and a caller that
 * is restoring a record it holds (the export path) send the same thing. That is
 * what keeps the two paths from growing different rules.
 *
 * `at` is the observation's own moment, never the wall clock, so replaying a
 * scrape rewrites the same row instead of appending a second one.
 */
export const observe = spacetimedb.reducer(
	{
		id: t.string(),
		url: t.string(),
		authorHandle: t.string(),
		authorName: t.string(),
		authorVerified: t.bool(),
		text: t.string(),
		postedAt: t.string().optional(),
		firstSeenAt: t.string(),
		lastScrapedAt: t.string(),
		/** shadow | public | hidden */
		visibility: t.string(),
		/** The list slugs as a JSON array. */
		lists: t.string(),
		replies: t.u32(),
		reposts: t.u32(),
		likes: t.u32(),
		views: t.u32(),
		/** bookmark | like | history */
		source: t.string(),
		at: t.string(),
		/** observed | published | hidden */
		action: t.string(),
	},
	(ctx, args) => {
		const existing = ctx.db.post.id.find(args.id);

		if (existing === null) {
			ctx.db.post.insert({
				id: args.id,
				url: args.url,
				author: args.authorHandle,
				text: args.text,
				postedAt: args.postedAt,
				replies: args.replies,
				reposts: args.reposts,
				likes: args.likes,
				views: args.views,
				firstSeenAt: args.firstSeenAt,
				lastScrapedAt: args.lastScrapedAt,
				visibility: args.visibility,
				lists: args.lists,
			});
		} else {
			// `firstSeenAt` is the archive's own fact and survives every fold.
			ctx.db.post.id.update({
				...existing,
				url: args.url,
				author: args.authorHandle,
				text: args.text,
				postedAt: args.postedAt,
				replies: args.replies,
				reposts: args.reposts,
				likes: args.likes,
				views: args.views,
				lastScrapedAt: args.lastScrapedAt,
				visibility: args.visibility,
				lists: args.lists,
			});
		}

		const known = ctx.db.author.handle.find(args.authorHandle);
		if (known === null) {
			ctx.db.author.insert({
				handle: args.authorHandle,
				name: args.authorName,
				verified: args.authorVerified,
				firstSeenAt: args.firstSeenAt,
				postCount: existing === null ? 1 : 0,
			});
		} else {
			ctx.db.author.handle.update({
				...known,
				name: args.authorName,
				verified: args.authorVerified,
				// A new post is a new count; re-observing one already held is not.
				postCount: existing === null ? known.postCount + 1 : known.postCount,
			});
		}

		// Append-only, keyed by content: replaying the same scrape writes nothing.
		const observationId = `${args.id}:${args.source}:${args.at}:${args.action}`;
		if (ctx.db.observation.id.find(observationId) === null) {
			ctx.db.observation.insert({
				id: observationId,
				at: args.at,
				source: args.source,
				postId: args.id,
				action: args.action,
			});
		}
	},
);

/** Append one event. The payload is opaque JSON text. */
export const record = spacetimedb.reducer(
	{
		kind: t.string(),
		subject: t.string(),
		payload: t.string(),
		at: t.string(),
	},
	(ctx, args) => {
		const id = `${args.at}:${args.kind}:${args.subject}`;
		if (ctx.db.event.id.find(id) !== null) return;
		ctx.db.event.insert({
			id,
			at: args.at,
			kind: args.kind,
			subject: args.subject,
			payload: args.payload,
		});
	},
);

/** Record which curated list a handle belongs to. The latest decision for a pair wins. */
export const classify = spacetimedb.reducer(
	{
		handle: t.string(),
		list: t.string(),
		by: t.string(),
		confidence: t.f32(),
		reason: t.string(),
		at: t.string(),
	},
	(ctx, args) => {
		const id = `${args.list}:${args.handle}`;
		const row = {
			id,
			list: args.list,
			handle: args.handle,
			by: args.by,
			confidence: args.confidence,
			reason: args.reason,
			at: args.at,
		};
		if (ctx.db.membership.id.find(id) === null) {
			ctx.db.membership.insert(row);
		} else {
			ctx.db.membership.id.update(row);
		}
	},
);

/** Attach one media item to a post, at the position the page showed it. */
export const attach = spacetimedb.reducer(
	{ postId: t.string(), kind: t.string(), url: t.string(), position: t.u32() },
	(ctx, args) => {
		const id = `${args.postId}:${args.url}`;
		const row = {
			id,
			postId: args.postId,
			kind: args.kind,
			url: args.url,
			position: args.position,
		};
		if (ctx.db.media.id.find(id) === null) ctx.db.media.insert(row);
		else ctx.db.media.id.update(row);
	},
);

Design the complete SpacetimeDB model for an agentic system whose local
capabilities are: browsing (a CDP-driven browser session), research (asking a
web chat product and reading the answer plus its step trace), design
verification (measuring a built page), skill validation, and the agent's own
task/lease lifecycle. Requirements:

1. The database must be self-describing: the schema of the schema lives in
   tables, so a running process can read the declared capabilities, their
   versions, their state classes, and their admission rules without the
   TypeScript source.
2. The process that runs the database must be able to modify the schema based on
   auth - so model identity, roles, and what a role may alter, as data rather
   than code.
3. Everything the agent needs locally should be answerable from this one
   database: UI state, browser sessions and their leases, research threads and
   their step traces, verification evidence, skill manifests.
4. Keep the existing archive tables working; say which of them survive unchanged
   and which must move. Give: the table list with columns and keys, the reducers
   and what each enforces, the self-describing tables and their invariants, how
   a lease is represented so two agents cannot claim one browser, and what you
   would refuse to put in the database. Reason from SpacetimeDB's actual model
   (tables, reducers, subscriptions, identity) and say where you are unsure.

Use one SpacetimeDB database with three layers: a typed operational model, a
versioned capability/schema catalog, and data-driven authorization. The
essential distinction is between the logical schema that agents may change
through authorized reducers and the physical schema compiled into the published
module.

SpacetimeDB does not turn a row inserted into a “table definition” table into a
new native table. Physical table and reducer changes require publishing an
updated module. Consequently, runtime schema modification needs either a
bounded, interpreted logical-schema layer or an authorized build-and-publish
controller—not a reducer executing arbitrary
DDL.[spacetimedb](https://spacetimedb.com/docs/intro/faq)

My recommendation is a **hybrid**: native tables for operational facts and
concurrency; an interpreted schema for extensible capability state; and an
explicitly privileged deployment path for physical changes.

## 1. Execution and trust boundaries

The database owns:

- Declared capabilities, versions, input/output contracts, state classes,
  admission policies.
- Identities, roles, scoped grants, approvals, and audit history.
- Tasks, execution attempts, resource leases, commands, and results.
- Browser observations, research conversations and visible step traces,
  verification evidence, skill manifests, and persisted UI state.
- Your existing archive.

An external worker owns the actual CDP connection, browser process, web-chat
interaction, measurement tooling, and skill execution. Reducers cannot perform
network requests or filesystem operations. Procedures can perform external I/O,
but their external effects do not share the atomicity of a reducer transaction.
A local CDP worker remains the appropriate boundary for your stated
capabilities.spacetimedb+1

Every exposed mutation authenticates using `ctx.sender`, not an identity
supplied in arguments. SpacetimeDB provides caller identity; your module
supplies the role and permission
interpretation.[spacetimedb](https://spacetimedb.com/docs/intro/key-architecture)

All new operational tables should be private. Expose caller-authorized views for
subscriptions. A client-side `WHERE` filter is not authorization: public tables
are readable by clients independently of the UI’s query choices. Views can
filter private rows and omit sensitive
columns.[spacetimedb](https://spacetimedb.com/docs/tables/access-permissions)

## 2. Table conventions

The following is a concrete proposed model, not a claim that SpacetimeDB
supplies these application tables.

Notation:

- `PK`: native primary key.
- `UQ`: native unique key where applicable.
- `IX`: nonunique index.
- `?`: optional.
- IDs: opaque strings, except authenticated principals use native `Identity`.
- Times: native `Timestamp`; durations: integer microseconds.
- Counters/fencing epochs: `u64`; measurements: `f64`; ordinals: `u32`.
- Enumerated statuses: declared algebraic types where fixed by the module, not
  unrestricted strings.
- `…Json`: bounded, validated canonical JSON, never executable code.
- References are logical foreign keys checked by reducers. Do not assume
  automatic SQL foreign-key enforcement; the transaction documentation
  explicitly places foreign-key-like relationships in application
  logic.[spacetimedb](https://spacetimedb.com/docs/databases/transactions-atomicity/)

For tuple keys, store the component columns and derive one collision-safe key
using a canonical encoding. Do not introduce more unescaped `${a}:${b}` keys.

All operational rows carry `scopeId IX`. A scope is the local workspace/security
boundary. All mutable current-state rows carry `revision:u64`, checked on
updates. Immutable records instead carry their provenance and recording
timestamp.

`logical_lookup` is a reducer-maintained search projection for declared logical
indexes. A unique logical index gets a globally unique claim encoding the key
definition and canonical value; nonunique entries do not claim uniqueness.

Do not route native task, browser, and lease state through `logical_entity`.
Those need explicit types, direct indexes, and carefully bounded update paths.

## Catalog invariants

1. The catalog describes its own tables and types. Bootstrap a finite set of
   metatypes, then describe `type_def`, `field_def`, `table_def`, and the
   remaining catalog using those types. References between definitions express
   the recursion.
2. An activated release is immutable. Changes create a new release.
3. Capability versions pin exact schema and policy versions. Running work does
   not silently adopt the latest contract.
4. Native definitions must match the deployed module. Generate the initial
   catalog manifest alongside `Schema.ts` from your declaration source, and
   verify a catalog/module digest during activation.
5. Describing a native table does not authorize altering its actual structure.
6. Policies use a small, versioned, bounded expression language: role checks,
   ownership, equality, membership, numeric bounds, status checks, and approval
   requirements. No `eval`, arbitrary TypeScript, SQL fragments, or unrestricted
   recursion.
7. `handlerKey` selects an installed implementation. Declaring a new operation
   does not create executable behavior.
8. Fail closed on unknown types, unsupported evaluator versions, dangling
   references, invalid transitions, or mismatched module hashes.
9. Logical uniqueness, references, indexes, and document updates commit
   together.
10. Definitions distinguish durable state, append-only evidence, derived
    projections, and leased resources. These are behavior contracts, not merely
    descriptive labels.

The interpreter and bootstrap trust rules necessarily remain code. Making
permissions data-driven does not remove the need for a trusted enforcement
mechanism.

## 8. How two agents cannot claim one browser

The exclusivity unit is the browser session, not an individual tab, unless you
deliberately implement stronger context isolation. Tabs can share
authentication, profile state, and browser-level behavior.

Use one persistent `resource_lease` row keyed by the browser resource ID. Never
delete its epoch history by deleting and recreating the current lease row.

Conceptually:

```
claim_task_and_resources(taskId, browserId, agentInstanceId, requestId):

  authenticate sender and verify ownership of agentInstanceId
  evaluate task and browser admission policies
  now := database context timestamp

  read task lease and browser lease
  require each is released, revoked, or expiresAt <= now
  require task is runnable

  increment each lease epoch
  assign both leases to this agent and new task attempt
  set bounded server-derived expirations

  insert task_attempt
  set task.currentAttemptId and task.status
  insert lease_history, execution_event, request_receipt
  commit
```

The claim is one reducer transaction—not “read availability in a client, then
write ownership.” SpacetimeDB’s transactional execution and key constraints
provide the database-side foundation for this
exclusion.[spacetimedb](https://spacetimedb.com/docs/databases/transactions-atomicity/)

Every renewal, command dispatch, and result commit verifies:

```
sender owns holderAgentId
attemptId matches
lease epoch matches
lease has not expired
resource generation matches
task attempt remains active
```

Important consequences:

- An expired lease cannot be resurrected by renewal.
- A new claim increments the epoch.
- A reconnect does not restore ownership automatically.
- Disconnect does not need to release ownership; expiry is the recovery
  mechanism.
- Timer cleanup is not a correctness prerequisite. Every use checks expiry.
- Revocation increments the epoch immediately.

## Database exclusion is not browser fencing

The lease prevents two valid owners in the database. It cannot stop a paused old
process from waking up and sending CDP commands directly.

Require a single browser-owning gateway/worker to enforce generations, lease
epochs, and command ordering. Agents must not possess independent unrestricted
CDP connections. The gateway rejects stale tokens and drains or quarantines
outstanding old-generation work before handing the browser to a new owner.

Even this cannot undo a click or form submission that already happened. Treat
external effects as a recoverable command protocol:

1. Persist intent.
2. Admit and dispatch under a valid lease.
3. Execute outside the transaction.
4. Record the outcome.
5. Reconcile ambiguous outcomes before retrying.

Do not promise exactly-once browser effects. In particular, a crash after
submitting a web-chat prompt but before recording its external turn ID must
trigger reconciliation, not blind resubmission. Procedures likewise cannot make
multiple transactions and external effects one atomic operation.spacetimedb+1

## 10. Runtime schema modification and physical deployment

For logical changes, authorized reducers can create a draft, validate it, and
atomically switch `schema_head`. Runtime workers can discover the new contracts
through subscriptions without reading TypeScript.

For physical changes, use this separate flow:

1. Store a proposed migration and exact target manifest.
2. Authorize it using current roles and an approval bound to the plan hash.
3. A trusted external controller generates declarations, builds the module,
   tests it, and publishes the exact approved artifact.
4. The controller records the resulting module digest and migration outcome.
5. Activate the corresponding catalog only after compatibility checks.

Compatible changes may migrate automatically; more complicated changes can
require successor tables and incremental migration. Publishing an updated
module—not editing catalog rows—is the documented mechanism for physical schema
evolution.[spacetimedb](https://spacetimedb.com/docs/intro/faq)

There is a hard trust limit: the publisher can replace the authorization
reducers themselves. Database roles can govern the cooperative deployment
workflow, but they cannot constrain someone who independently controls the
database owner credentials or host. Keep publisher credentials outside the
agent’s authority; the database owner can bypass ordinary table visibility
restrictions.[spacetimedb](https://spacetimedb.com/docs/1.12.0/databases)

Also, publication and the preceding authorization transaction are not one atomic
operation. Use a deployment lock/quiescence protocol and explicit recovery
states.

Expose indexed, caller-authorized views such as:

- `my_tasks`, `task_detail`, `task_events`.
- `available_browser_sessions`, `browser_tabs`, `latest_page_snapshot`.

- `capability_catalog`, `schema_release_detail`, `effective_permissions`.
- `my_ui_state`.

Procedural views need indexed access rather than unrestricted `.iter()` scans,
which is why scope, identity, parent IDs, and lookup keys are indexed throughout
this design.[spacetimedb](https://spacetimedb.com/docs/functions/views)

“Everything locally answerable” should mean that current state, declared
contracts, authority, outcomes, and evidence provenance can be read from this
database. It must not mean that a database row is the actual browser connection
or that an old page snapshot is current truth. Every external observation
carries capture time and generation.

## 12. What I would refuse to put in the database

- Live CDP sockets, process handles, JavaScript execution contexts, closures, or
  browser object instances.
- Plaintext cookies, bearer tokens, private identity keys, browser-profile
  secrets, or unrestricted credential-bearing endpoints. Store secret
  references.
- Arbitrary executable code disguised as policy, schema defaults, or skill
  manifests.
- Hidden model chain-of-thought. Store supplied answers and visible product
  traces.
- Unbounded screenshots, video, network dumps, duplicate DOMs, or large package
  binaries by default. Store bounded extracts and content-addressed artifact
  metadata.
- Unsupported “verified” flags without a pinned specification, environment,
  tool, and evidence.
- Browser exactly-once guarantees that the external system cannot provide.
- Parallel, contradictory sources of ownership such as `browser.claimedBy`,
  `agent.currentBrowser`, and `lease.owner`. The lease is authoritative; other
  fields are projections.
- Raw secrets or operational state inside the existing public `event.payload`.

## Post identity is stable, but article position is not

The selector explicitly acknowledges that the page recycles articles as you
scroll; it provides ID-based lookup for that reason
(`Maintain/Source/Browser/Selector/Post.ts:8–10, 94–96`). Initial resolution by
`--id` uses that protection: `withTarget()` passes the requested ID into
`snapshot()` (`Maintain/Source/Transport/CDP/Post.ts:63–64, 137`).

But the first mutation does not preserve that identity. `toggle()` sends only
the article index and control, and `settle()` also polls by index. The selector
chooses by ID only when the request includes one; otherwise it selects the
current article at that index
(`Maintain/Source/Browser/Session/Post.ts:112–136`;
`Maintain/Source/Browser/Selector/Post.ts:204–205`).

The inference is a time-of-check/time-of-use failure: if article A occupied
index 0 during the initial read, but article B occupies it when the click
executes, the click can affect B. The returned post identity still comes from
the original context, A (`Maintain/Source/Transport/CDP/Post.ts:196–203`).

This can be silent. The code checks that an article and button were found, not
that the clicked article has the original ID. A later read by A’s ID does not
undo the click on B.

To settle this, I would exercise the exported selector expression against a
controlled page: snapshot A, reorder or replace the articles before the toggle
expression, then record which button receives the click. This needs no
live-account mutation.

## A URL is treated as a locator, not an identity constraint

For `--url`, validation merely checks whether `idFromHref()` finds
`/status/<digits>` anywhere in the string. It does not validate the URL’s host
or scheme (`Maintain/Source/Model/Post/Post.ts:24–26`;
`Maintain/Source/Transport/CDP/Post.ts:69–74`).

More importantly, `wanted` is populated only for the `id` target variant. A URL
target therefore waits for any article and snapshots index 0, rather than
selecting the ID embedded in the URL
(`Maintain/Source/Transport/CDP/Post.ts:63–64, 116–137`).

What I am reading: there is no requested-ID check on the URL path through
`withTarget()`.

What I am inferring: a redirect, an unexpected page layout, or a thread where
another article appears first can produce a successful reading of another post.
The same path is used for like/bookmark operations, so the assumption matters
beyond extraction. Whether the current X detail page actually presents that
ordering requires a run.

I would test a controlled URL containing `/status/123` whose page contains
articles 999 and 123 in that order. The expected invariant is that the resolved
post is 123, regardless of article order.

## “An article exists” is treated as “the post is ready”

`withTarget()` waits up to 25 seconds for
`document.querySelectorAll("article").length > 0`. Absence produces a named
`no-post` failure; presence proceeds directly to the snapshot
(`Maintain/Source/Transport/CDP/Post.ts:116–150`).

That proves the existence of an article, not that its text, author, controls, or
media have finished rendering. The selector supplies empty text, an empty author
handle, null time, and empty media when the corresponding nodes are absent
(`Maintain/Source/Browser/Selector/Post.ts:104–138`).

Consequently:

- No articles is loud: `no-post`.
- A partially rendered article can be silent: a structurally successful reading
  with missing fields.
- With an ID target, an unrelated article can satisfy readiness before the
  desired post arrives; the immediate ID snapshot can then fail loudly even
  though the desired post would appear later.

The last case is an inference about asynchronous rendering, not an observed
live-page failure. I would run a fixture that inserts an unrelated article
immediately, the target article later, and its media later still; then compare
when the transport returns with when the fixture becomes complete.

## Extraction assumes the DOM preserves specific meanings

The reading depends on relative permalinks matching `/handle/status/digits`,
specific `data-testid` values, and English metric keywords. These are not merely
cosmetic selectors: they determine identity, control state, and numeric values
(`Maintain/Source/Browser/Selector/Post.ts:55–79, 83–90, 104–123, 140–157, 208–235`).

Some consequences are directly visible:

- Unreadable counts become zero rather than unknown.
- Missing controls have `present: false` and `active: false`.
- Missing text becomes `""`.
- A video without `src` can be represented using its poster URL while retaining
  `kind: "video"`.

Those behaviors are implemented at
`Maintain/Source/Browser/Selector/Post.ts:55–65, 119–132, 208–235`.

The inference is that a selector change, localized count format, or delayed
rendering can turn “not observed” into apparently valid data. For example,
parsing removes commas before interpreting the number, so a decimal-comma label
can be misread; the views selector also specifically searches English
“Views”/“views” (`Maintain/Source/Browser/Selector/Post.ts:58, 143`).

These are predominantly silent failures. I would evaluate the actual exported
`expression()` against DOM fixtures with missing controls, localized labels,
absolute rather than relative permalinks, and video posters without playable
sources. The test should distinguish unknown from zero/false/empty, rather than
merely assert that extraction returns an object.

## Background visibility is handled asymmetrically

There is a deliberate workaround for lazy media: `reveal()` enables focus
emulation and an active lifecycle state. But it is called only when the target
was addressed by ID and the tool opened a temporary tab
(`Maintain/Source/Transport/CDP/Post.ts:33–45, 63–64, 129–135`).

A URL target can also open a temporary tab, but does not take that branch. An ID
target that reuses an already-open background tab does not take it either
(`Maintain/Source/Transport/CDP/Post.ts:82–85, 97–111`).

The source itself explains that hidden detail tabs can otherwise report empty
media. What remains an inference is how often current browser/page behavior
exposes this discrepancy. In addition, both reveal commands swallow errors, so a
failed workaround has no warning
(`Maintain/Source/Transport/CDP/Post.ts:38–44`).

I would compare the same photo post through `read({id})` and `read({url})`,
first with no existing detail tab and then with a pre-existing background tab.
That would establish whether the representations diverge in the current
environment.

## “Network idle” means no recent event—not no outstanding request

The session updates one timestamp on request start, finish, or failure.
`waitForIdle()` returns true once that timestamp has been quiet long enough; it
does not maintain a set of in-flight requests
(`Maintain/Source/Browser/Session/Session.ts:75–90, 139–147`).

Therefore a request can start, remain pending longer than the quiet interval,
and still satisfy this implementation’s idle condition. That follows from the
algorithm; it does not require speculation about X.

The practical consequence is inferred: a reload or tab close can occur while a
mutation request remains outstanding. If the Network domain is unavailable, the
session silently degrades further to a timed quiet window
(`Maintain/Source/Browser/Session/Session.ts:81–89`).

I would attach to a local fixture that starts a fetch whose server response is
deliberately held open, then measure `waitForIdle({quiet: 600})`. If it resolves
while the request remains pending, it proves the semantic gap.

## A completed click is not consistently a verified write

The mutation path contains useful safeguards: no click if the observed state
already matches, named failures for missing article/button, and reload-based
verification for temporary tabs
(`Maintain/Source/Browser/Session/Post.ts:112–133, 64–97`).

But it also contains several weaker assumptions:

- The result of the six-second `settle()` is discarded.
- If the post/control cannot be read afterward, `now` defaults to the desired
  state.
- Reload-based verification can return null; that leaves the earlier value in
  place.
- Existing user tabs are not reloaded for persistence verification.

These behaviors are explicit in
`Maintain/Source/Browser/Session/Post.ts:135–146`; the persistence fallback is
at `:69–97`.

Thus `now: true` need not mean “the server accepted and retained the change.” In
the missing-control branch it can mean “the caller wanted true and verification
no longer had a readable control.”

A server rejection that remains visible as the opposite state can be returned as
`now != want`, which is observable but not thrown as an error. A rejection
masked by optimistic state or missing verification can be silent at the
transport boundary. I have not traced the CLI’s treatment of that returned
mismatch in this pass.

I would test `toggle()` with a fake Session that reports a successful click,
never settles, and then loses the control. A second test would make the
temporary-tab reload fail. Neither requires clicking a real account.

## CDP failures are usually loud; polling deliberately blurs their cause

The underlying client rejects protocol errors, times out individual commands,
and rejects pending calls when the socket closes. Those become `cdp-error` or
`cdp-unreachable`, rather than indefinite pending operations
(`Maintain/Source/Browser/CDP/Client.ts:107–136, 139–179`).

Page evaluation also turns JavaScript exceptions into named `cdp-error` failures
(`Maintain/Source/Browser/CDP/Command/Evaluate.ts:30–42`).

However, `Session.waitFor()` catches every evaluation error and keeps
polling—not only the transient execution-context destruction mentioned in its
comment (`Maintain/Source/Browser/Session/Session.ts:149–163`).

That is a reasonable navigation-recovery strategy with a diagnostic cost: a
permanently invalid expression or dead connection can emerge as a generic
readiness timeout instead of its original cause. In the post path this becomes
“the page showed no posts”; other direct evaluations remain loud.

I would inject a permanent evaluation error and compare it with a transient
context-destruction error. The important assertion is not just that both
terminate, but whether the permanent error’s cause remains available to the
caller.

## Cleanup assumes attachment succeeded and close failures are harmless

`withTarget()` encloses attachment in its `try/finally`, so it attempts to close
a temporary tab even if attachment fails
(`Maintain/Source/Transport/CDP/Post.ts:68, 115, 162–172`).

`withPage()` attaches before entering its cleanup `try`. If it has opened a tab
and attachment rejects, that cleanup block is never reached
(`Maintain/Source/Browser/Session/Page.ts:57–85, 113–123`).

That is a source-confirmed cleanup gap. Its practical manifestation would be a
leaked tab after an attachment failure. Both cleanup paths also suppress close
errors on the assumption that the tab is already gone; an unreachable browser is
another possible explanation, so cleanup failure can be silent.

I would inject an attachment failure after successful tab creation and assert
that the close operation is attempted. This should be a controlled test, not a
run that deliberately leaves tabs in your browser.

## The architectural distinction to preserve

The transport correctly separates “live browser” from “shadow archive”: the view
transport refuses browser operations and mutations, and requires an ID for
archived reads (`Maintain/Source/Transport/View/Transport.ts:59–95`). Transport
selection also rejects unsupported declared requirements before command
execution (`Maintain/Source/Transport/Select.ts:47–58`).

The weakness is not that the subsystem lacks failures. It is that its strongest
failure signals concern connection and element availability, while its weakest
signals concern the meaning of a successful result. The most valuable tests
would therefore adversarially change identity, delay completeness, and reject
persistence—not just disconnect CDP.

# Required behaviour

The port is the shape a consumer writes, not the shape a provider offers. Declare
it in `src/Port.ts`:

```ts
export type Target = { url: string; ws: string };

export interface Browse {
	/** Open a page, returning the handle the session is bound to. */
	open(url: string, options: { timeoutMs: number; signal: AbortSignal }): Promise<Target>;
	/** Evaluate an expression in the page and return its value. */
	evaluate<T>(target: Target, expression: string, options: { timeoutMs: number; signal: AbortSignal }): Promise<T>;
	/** Release everything this session holds. Never throws. */
	dispose(target: Target): Promise<void>;
}
```

Rules the implementation and its test must both honour:

1. **Cancellation is the caller's.** Every operation takes an `AbortSignal` and
   rejects with the port's `timeout` error when it fires or the deadline passes -
   not with a generic `Error`.
2. **`dispose` never throws** and is safe to call twice. A dispose failure is
   reported, not propagated.
3. **A dead transport is a `browser-unavailable`**, distinguishable from a page
   that is merely not ready. Do not let a connection error surface as a timeout.
4. **Ownership is explicit.** `open` either attaches to an existing target or
   launches one, and the caller must be able to tell which, because only a
   launched target may be closed.

Implement against a fake in `Test/Browse.test.ts`: a fake transport that answers
`evaluate`, can be made to hang, can be made to fail, and can be closed twice.

export type Budget = { deadline: number; operations: number };

export interface Runtime {
	/** Admit a run, or refuse it with the reason. */
	admit(request: { capability: string; state: StateClass; generation: number }): Promise<{ ok: true } | { ok: false; reason: string }>;

   deadline and operation count come from the caller every time.
4. **`release` never throws** and is safe twice.

# Design

What this workspace is for, and the decisions that shape it. Written after a
research pass over the real inventory of the consuming repository, so the
boundaries here are argued from that surface rather than from a taxonomy.

Provenance convention: each section below opens with **Source:** naming the
capture in `.hermes/research/` that states the claim, or **Decision: this
repository** where no capture does. `docs/Research.md` indexes the captures.

## The problem

**Decision: this repository** - the 125 module / 13 skill / 17 script inventory,
and the four capabilities it implies, are measurements of the consuming
repository. No capture records them. `delo-e9b61659.md` is a research pass over
one subsystem of that repository, which supports the premise in the opening
paragraph rather than the numbers.

A repository of 125 TypeScript modules, 13 documentation skills and 17 reference
scripts has capabilities that are useful outside it - browser control, chat
research, design measurement, skill validation - but each is entangled with the
product it was written for. The goal is to lift those capabilities out without
lifting the product with them.

## The four layers

**Source:** `.hermes/research/rust-4ffde6d5.md` - the layer table, "the arrows
point inward", "a capability importing an adapter is a defect" and "ports are
owned by their consumers, not by a central crate" are stated there, and §1
answers the Cargo consequence: no shared ports crate, two consumer-owned
`Browse` traits. **Decision: this repository** - the reason given for the defect
(a capability untestable without its provider) and the rule that a shared
contract is extracted only when two consumers genuinely share semantics and
evolution policy.

| Layer            | Owns                                       | May import                   |
| ---------------- | ------------------------------------------ | ---------------------------- |
| Capability       | a use case and the port it needs           | its own port and data shapes |
| Adapter          | one port against one mechanism or provider | that port, that provider     |
| Workflow         | a composition of capabilities              | capabilities only            |
| Composition root | the binding of ports to adapters           | everything                   |

The arrows point inward. Nothing imports a root, and a capability that imports
an adapter is a defect: it makes the capability untestable without the provider,
which is the property this workspace exists to keep.

**Ports belong to their consumers, not to a centre.** A shared ports package
becomes the dependency magnet at forty modules - one workflow's requirement then
forces edits through unrelated modules. Extract a shared contract only when two
consumers genuinely share semantics and evolution policy.

**A port states more than data shapes.** Domain errors, ordering, pagination,
cancellation, idempotency, consistency and resource ownership belong in the
port, or two adapters type-check and still behave differently.

## The manifest

**Source:** `.hermes/research/rust-4ffde6d5.md` - the four state classes and
their admission role, the hashed `artefact` entry point and the cold/warm `cost`
are stated there, and §3 answers how they survive into Rust: a checked-in
`module.json`, serde validation in CI, read by a harness that never starts the
application. **Decision: this repository** - the observation each class implies:
a liveness probe for `holds-transport`, a freshness check for `holds-identity`,
a reset proof for `holds-workspace`. The capture says what a harness must
revalidate at admission, not which observation proves it.

Every package carries `module.json`. The field that does the most work is
`state`:

- `stateless` - nothing to check at admission.
- `holds-transport` - a liveness probe, because an open socket can be dead.
- `holds-identity` - a freshness check, because a token expires while pooled.
- `holds-workspace` - a reset proof, because a dirty workspace answers wrongly.

`artefact` carries a hashed entry point, so a harness loads a built file and
never builds at launch. `cost` carries cold and warm figures, so a pre-warm
planner has something to budget against rather than guessing.

## The lifecycle

**Decision: this repository** - the seven-step order, and the claim that
preparation and admission are different acts. The captures supply the vocabulary
(admission, leases, budgets, cancellation, cleanup) but not this sequence.
`delo-e9b61659.md` supplies the failure the paragraph below is about: a
completed click accepted as a write, and an article's existence accepted as
readiness (`:40-54`, `:96-116`).

prepare reusable infrastructure → lease an isolated context → bind identity and
task → validate credentials, connections and versions → execute within a fresh
budget → persist only authorised artefacts → destroy or prove a reset.

Preparation improves latency. Admission establishes correctness. Merging them is
how a warm pool starts returning wrong answers quickly.

## Why the packages are these five

**Source:** `.hermes/research/rust-4ffde6d5.md` - the five packages with their
kinds, state classes and errors, and §1's decision that all five stay separate
crates rather than becoming modules of one. **Decision: this repository** - the
build order, and each reason for building that package when it is built. The
capture says which crates exist, not which one to write first.

Chosen in build order, each justified by a consumer that already exists:

1. **browser-cdp** - the only execution substrate shared by chat research,
   design verification, scraping and browser maintenance. Build it first because
   everything else needs it.
2. **chat-driver** - a real reusable behaviour with identifiable failure
   semantics: turn correlation, mode and model verification, completion
   deadlines, trace extraction.
3. **design-verification** - separates a measurement engine from the particular
   design being measured.
4. **skill-contract** - gives a large skill corpus one validation path without
   pretending every skill is executable infrastructure.
5. **runtime** - admission, leases, budgets, cancellation, cleanup. Build it
   last, after the four above expose real lifecycle needs; otherwise it is a
   harness designed around imagined commonality.

## What is deliberately not here

**Decision: this repository** - the exclusion list. No capture discusses store,
model, command, vote, ingest, board synchronisation or gateway maintenance as a
capability domain; the words appear only in other senses (a probabilistic model,
a cost model, a CLI command).

Domain policy and write authority stay with their consumers until a second
workflow demonstrates a stable contract: store, model, command, vote, ingest,
board synchronisation, gateway maintenance. A generic name is not a port, and a
package called `model` or `context` becomes foundational by default unless
somebody refuses it.

## The two failures this design expects

**Source:** `.hermes/research/rust-4ffde6d5.md` §2 for the ownership half -
`Ownership::{Attached, Launched}`, attach and launch as separate operations,
disposal of an attached session that must not terminate the browser, and a
disconnected session that invalidates warm readiness. **Decision: this
repository** - the five-surface inventory, the global warm browser scenario, and
the extraction failure. No capture examines either.

**Competing browser owners.** Five surfaces already touch browser execution with
different ownership needs - a driver, a verifier, a scraper, a relaunch tool, an
audit. A global warm browser can close a session someone else owns, reuse the
wrong account, or couple an isolated audit to interactive state. The lease model
has to make ownership explicit before any of it is shared.

**Source-relative assumptions surviving extraction.** Bundling into `Target/`
does not make `Reference/`, routes, tokens, configuration or sibling paths
portable. The test is installing packed artefacts into a fixture with the
original repository unavailable.

## The warning metric

**Decision: this repository** - the metric itself. Unrelated coordination per
change is this document's proposal; nothing in the captures measures it or names
it.

After two years the bad version of this design has hundreds of package files,
duplicated manifests, permissive graph exceptions, warm contexts nobody can
safely evict, and consumers pinned to matching release trains. The useful signal
is not package count: it is **unrelated coordination per change** - how many
packages must be edited together for one workflow's requirement.

# Observability: the database is the surface

An earlier pass argued that SpacetimeDB is a poor fit for control flow and that
orchestration belongs in Rust. That was too broad, and this document records the
correction because it changes the architecture.

## The rule, as a decision procedure

For any new piece of behaviour, ask one question: **does it touch the outside world?**

- Pushing JavaScript into a page, opening a socket, spawning a process, reading a
  file, waiting on a network reply - that is an effect, and it lives in Rust, in a
  crate that owns the transport.
- Everything that is a decision, a counter, a phase, a retry, a budget, a lease, a
  claim, a cancellation request, an operator command, or a record of any of those -
  that is an internality, and it lives in a table, changed by a reducer.

The test is not "is it important" or "is it fast". It is whether the thing has an
observable effect outside the database. A retry count is internal. The HTTP call it
guards is external. A "we are now on phase three" is internal. The DOM evaluation
that phase three performs is external.

When a piece of work straddles the line - a scrape is a decision plus a page read plus
a write - split it: the reducer moves the phase and grants the lease, the Rust effect
runner performs the read, and a second reducer records the result. Two transactions,
one truth, and the gap between them is visible as a lease that is held.

## The corrected boundary

The line is not control flow versus data. It is **effects versus state**.

| Lives in SpacetimeDB | Lives in Rust |
| --- | --- |
| workflow transitions between phases | the browser connection and its socket |
| admission decisions | DOM evaluation and input dispatch |
| leases, fences, expiry | process and filesystem operations |
| budgets and their consumption | the work an effect actually performs |
| cancellation requests | honouring a cancellation mid-operation |
| operator commands | the binary's own supervision |

Rust does not need a second authoritative workflow engine whose decisions are copied
into the database afterwards. That copy is a second state machine, and the
reconciliation between the two is the bug you spend the next year fixing.

## Why a reducer is the right primitive

A reducer is one transaction. That means a single call can:

1. move an invocation into a new phase,
2. reserve its budget,
3. grant its resource lease,
4. record the transition and its cause,
5. and make the resulting state visible to every subscriber,

and if any step fails, none of it happened. A log line cannot do the fourth step, and
an out-of-band write cannot do the fifth. This is the difference between a dashboard
that describes the system and a dashboard that *is* the system's state.

## What this buys over OTEL

- **Subscriptions are a live view, not a query loop.** A panel sees a change because
  it was pushed, not because it polled.
- **The record and the state are the same bytes.** There is no window in which the
  trace says one thing and the state another.
- **Replay is re-reading.** A generation's rows are what happened; a subscription
  replayed from the start is the history.
- **Atomicity is the transaction, not a convention.** A partial transition is not a
  thing that can be recorded.

## What OTEL still does better

- **Cross-process and cross-vendor traces.** A database only observes what writes to
  it; a process that dies before its first write leaves no row. OTEL's collector sees
  the process, the crash and the resource pressure around it.
- **Cardinality at volume.** Metrics with unbounded label sets are cheap to emit and
  expensive to store as rows.
- **Ecosystem.** Existing dashboards, alerting and retention policies.

The honest position: SpacetimeDB for the state machine and the operational surface;
OTEL for what happens *around* the process, if it is wanted at all.

## The dashboard

One surface, exported from inside the binary, able to attach to any deployment:

- **Transport**: a subscription over the same database the workers write to. Not a
  second API, because a second API is a second truth.
- **Identity**: the panel connects as a client with its own identity and a role that
  grants read on the operational tables and no write. The role is data, so it can be
  revoked without a redeploy.
- **Federation**, in ascending order of effort: one shared control database for every
  enrolled binary; several panel connections each labelled with its environment; or a
  designed replication layer. SpacetimeDB's inter-database communication is
  outbox-based - a message row delivered as a reducer call - which is evidence for
  that pattern, not proof of global transactions across databases.
- **Staleness is visible.** A disconnected panel or a worker whose heartbeat has
  expired must render as unknown, never as healthy-because-of-its-last-row. The
  cheapest way to guarantee that is a heartbeat table plus a view that computes
  freshness from the database's own clock.

## What this rules out

- A second scheduler in Rust that decides what the database then records.
- A dashboard that polls a REST projection of the tables.
- Any state that exists only in a process and is "synced later".

# Thread 1 - hidden properties and disclosure

Source: `.hermes/research/research-575cb198.md`. Nothing in this workspace talks
to SpacetimeDB yet, so every experiment below is a probe against a local
instance rather than a test in this tree. `spacetime sql <database> "<query>"`
is the shape of the check; `<database>` has to exist first.

## Q1 - What does a failed transaction roll back?

**Asked:** the thread's follow-up list (`research-575cb198.md:145`).

**Settle it with:** a reducer that inserts a row, then panics; query the table
afterwards. A surviving row means the insert is not part of the rollback.

```sh
spacetime sql < database > "SELECT * FROM Observation"
```

Repeat with a reducer that schedules a reducer and then fails - whether the
schedule survives is a separate answer.

**Blocks:** the `observe()` rule of charging and revealing in one transaction
(`:135`). If rollback is partial, "pay and reveal is all-or-nothing" is false
and the reducer has to be restructured.

## Q2 - What is the exact concurrency model across modules?

**Asked:** the thread's follow-up list (`:146`).

**Settle it with:** two reducers in two different databases, each recording a
start and end timestamp; run them concurrently and read the interleaving. Then
repeat inside one database, where the thread says execution is serial (`:11`).

**Blocks:** the nonleakage argument, which assumes an observer cannot learn from
interleaving it did not cause.

## Q3 - How are scheduled reducers sequenced internally?

**Asked:** the thread's follow-up list (`:147`).

**Settle it with:** schedule two reducers for the same timestamp and record
invocation order across repeated runs. The thread already warns that no dispatch
order is guaranteed (`:14`), so the experiment's job is to find out whether that
is merely undocumented or actually unstable.

**Blocks:** any probe-expiry or budget-refill design, both of which want a
deterministic clock.

## Q4 - What are the formal guarantees of client subscriptions?

**Asked:** the thread's follow-up list (`:148`).

**Settle it with:** two clients, identities A and B, each subscribing to every
public table and every view, ignoring the intended UI; then compare the two
transcripts field by field (`:127`, `:132`).

**Blocks:** the nonleakage target itself (`:140`). A guarantee of "same visible
values" is weaker than the transcript equality the thread specifies, and only a
subscription guarantee can close the difference.

## Q5 - How does state replication affect determinism?

**Asked:** the thread's follow-up list (`:149`).

**Settle it with:** two replicas fed the same transaction stream; compare the
observable transcripts they serve. Divergence means the paired-world test cannot
be run against a single node and generalised.

**Blocks:** the paired-world experiment below (Q8), and with it the whole formal
leakage claim.

## Q6 - Does one probe action strictly dominate all the others?

**Asked:** the thread's own check (`:122`).

**Settle it with:** enumerate the legal probes for one capability, compute what
each reveals per unit of cost, and check whether one action is weakly better in
every belief state. If it is, the hidden-information mechanic is degenerate and
the disclosure policy has failed its own fairness test.

**Blocks:** the disclosure policy. A policy that permits a dominant probe has
published the secret with extra steps.

## Q7 - Can two racing probes exceed a one-use allowance?

**Asked:** the thread's own check (`:135`).

**Settle it with:** two clients probing a capability with an allowance of one,
in the same transaction window; count the `Observation` rows written. The
reducer authenticates and charges before revealing, so the count must be one.

**Blocks:** the `observe()` reducer contract. This is the only place where the
thread claims atomicity buys correctness, so it is the one that has to be run.

## Q8 - Does a planted secret canary appear anywhere a client can see?

**Asked:** the thread's own check (`:139`).

**Settle it with:** plant a canary value in a private property, run a full
probe, then read every client-visible row, trace, failure response, result-size
field and exported artefact, and grep for the canary. Then run the paired worlds
of `:140` - two databases differing only in the secret - and diff the complete
observable transcripts.

**Blocks:** the trace-projection rule and the claim that the schema provides
nonleakage. The thread is explicit that timing, size, contention and operator
access are outside it (`:141`); the experiment's job is to show those are the
_only_ channels left.

# What blocks the most work

**Q18, Rust/TypeScript parity.** Everything in `docs/Rust.md` past the crate
layout waits on it, and the thread's own recommendation is conditional on it. A
failed parity run does not invalidate the port; it converts the sidecar from
migration bridge to permanent component, which changes the crate count.

**Q19 and Q20, manifest mapping and the arrows check.** These are the two halves
of the answer that was cut off. Without them the manifest is a JSON file nobody
has mapped and the layering rule is a convention, which is exactly the state the
Rust port exists to leave behind.

**Q9 and Q10, transport identity.** If a transport can address a post by
position, the browser port's signature is wrong, and it is wrong in a way that
every capability above it inherits. This is the cheapest of the blocking
questions to settle - a fixture page and no live account - which makes it the

# Reflection and Coherence

How a system that can rewrite its own capability definitions stays honest about
what it is running. The formal apparatus is worth using only up to a point; this
note says where that point is.

## The property worth naming

**Decision: this repository** - the term, the definition and the three things a
safety claim may name. No capture uses "manifest-enforcement coherence". The
observation-boundary half of the third item is the subject of
`.hermes/research/research-575cb198.md`, which argues what an observation must
name and record; no section here draws on it yet.

**Manifest-enforcement coherence:** the capability manifest a runtime reports is
the manifest it actually enforces.

Not "the manifest is correct" - correctness is a separate, unprovable-in-general
claim. Coherence is checkable: read what the system says it is, then exercise
the system and compare. Every safety claim in this workspace should name one of
three things: a state invariant, an observation boundary, or an equivalence
theorem. A claim that names none of them is prose.

## The trust boundary comes first

**Decision: this repository** - the argument that a self-check has no
independent basis, and the choice of the composition root plus the process that
launches it as the party outside the mutation authority. **Source:**
`.hermes/research/rust-4ffde6d5.md` §3 for the mechanism the section relies on:
a source descriptor embedded in the crate and an external release descriptor
binding it to the delivered bytes, so a manifest read from the mutable store is
not the only copy that exists.

If the mutation path can rewrite the reporter, the enforcer and the verifier, an
internal self-check has no independent basis for detecting coordinated
deception - it is checking the story against itself.

So one component must sit **outside** the mutation authority, and something
external must attest to its identity. In this system that is the composition
root plus the process that launches it: the root holds the manifest it bound,
and the runtime can only report what it was given. A manifest read from the
mutable store alone is not evidence.

## The obligation, reduced to induction

**Decision: this repository** - the induction shape, the four-part mutation
obligation (authorisation, validation, migration, publication) and the
transition table as the cheap version that is still honest. No capture states
any of them.

Coherence is an invariant, so the proof shape is an induction over transitions:

- establish `Inv(s0)` - the manifest bound at start-up is the manifest enforced
  at start-up;
- show `Inv(s) ∧ s →ℓ s' ⇒ Inv(s')` for every state-changing transition:
  admission, invocation, mutation, recovery, lease reuse, revocation.

For a mutation specifically, four things must jointly establish the invariant
for the new generation: authorisation, validation, migration, publication. If
any one of them is missing, the new generation is coherent only by accident.

Tests check instances of this; a general claim needs a proof or exhaustive
checking of a finite model. The cheap version that is still honest: a transition
table listing every state-changing operation beside the invariant it must
preserve, and a conformance run per generation.

## Where the apparatus collapses

**Decision: this repository** - the boundary at declarative records under a
fixed interpreter, and the reduction to ordinary versioning. No capture
discusses reflection; this is why `research-575cb198.md` cannot be read as the
source of this section despite the name the brief gave it.

If reflection only reads and replaces **declarative records under a fixed
interpreter** - which is what this workspace is - then none of the higher-order
machinery is needed. The requirement is ordinary versioning:

- a versioned contract,
- a validator,
- a compiler or installer,
- a migration `m: Vn → Vn+1`,
- and a test that the migration composed with enforcement preserves the
  contract.

No reflective tower is required merely because the records describe executable
capabilities. The check is: demonstrate that every admitted record is
interpreted by the same trusted semantics.

## Where it stops being ordinary migration

**Decision: this repository** - the three changes that defeat record validation
(interpreter, authorisation semantics, the manifest language itself), and the
"new policy approved versus old policy preserved" distinction.

If an authorised update can change the **interpreter**, the **authorisation
semantics**, or the **meaning of the manifest language**, record-shape
validation is not enough. Then you need either a semantics-preservation or
refinement obligation between interpreter versions, or an explicit declaration
that the policy changed.

The distinction to enforce in every upgrade: _new policy approved_ versus _old
policy preserved_. Conflating them is how a rewrite gets described as a
refactor.

## What gets built

**Decision: this repository** - all four items. Nothing in the captures proposes
a `reflect` operation, a generation-scoped conformance run, a transition table
or the semantics-change refusal.

1. A `reflect` operation that returns the enforced manifest **and its generation
   id**, answered from the composition root rather than from the mutable store.
2. A generation-scoped conformance run: the contract test suite executes against
   the generation, and its result is recorded with that id.
3. A transition table in the runtime package listing each state-changing
   operation beside the invariant it preserves.
4. A refusal: a mutation that would change interpreter semantics is rejected
   unless the caller states which policy changed, so the audit trail records the
   difference between preservation and replacement.

Bisimulation is used where it belongs - comparing executions across two
implementations of the same port, or across two interpreter versions. Congruence
is the obligation when claiming those comparisons survive inspection and
mutation contexts. Neither is needed to say "this generation enforces what it
reports".

**Source:** `.hermes/research/rust-4ffde6d5.md` §2 - the acceptance suite the
capture requires before the TypeScript sidecar is removed (attach/launch,
navigation races, evaluation timeout, disconnect, disposal, shared-browser
ownership) is one such comparison, named concretely rather than gestured at.
**Decision: this repository** - the bisimulation and congruence framing, and the
claim that neither is needed for enforcement coherence.

## Thread 1 - hidden properties and disclosure

**Source:** [`research-575cb198.md`](../.hermes/research/research-575cb198.md),
154 lines.

**Asked:** (1) is SpacetimeDB's execution model reactor-style, and what is
actually documented about concurrency and ordering; (2) what formal frame fits
capabilities whose properties are hidden at first and revealed by traces or by
other actors - POMDP, epistemic logic, hidden Markov model - and what Netrunner
teaches about disclosure that is fair, legible and not degenerate; (3) how to
represent those properties in a relational database whose subscriptions push
rows to clients without leaking through the subscription itself, with the table
design, the reducer rule, and where the leak is unavoidable.

**Settled** (`:11-142`):

- Each database is a single-threaded actor: transactions in one database execute
  serially, different databases concurrently. "Reactor" is not the documented
  word; the architecture article says "actor" (`:11`, `:17`).
- Transaction guarantees are not scheduling guarantees. There is no global FIFO
  across clients and no guaranteed dispatch order for overdue scheduled
  reducers, so ordering has to be made explicit with sequence or version
  preconditions (`:14`).
- An action-conditioned POMDP is the frame for discovery, with an epistemic MDP
  as the close special case when properties are fixed; the belief update is
  Bayesian filtering, and a stored confidence number is not a calibrated model
  (`:21`, `:24`, `:112`).
- Epistemic logic answers a different question - what each actor knows, and what
  they know about others' knowledge - and probabilities are degrees of belief
  rather than knowledge (`:115`).
- Latent truth, authorised evidence and actor belief are three records with
  three authorities; another actor's statement is testimony, not a mutation of
  truth (`:118`).
- Netrunner's lesson is rule-governed access, not face-down storage: every
  reveal must name the rule authorising its target, audience and extent,
  repeated probing must be bounded, a persistent property must not be rerolled
  on inspection, and "hidden again" is not "unknown again" (`:119`, `:122`,
  `:123`).
- Never publish a secret-bearing row and conceal it client-side: subscriptions
  synchronise authorised rows into the client cache, private tables are
  inaccessible, and a caller-aware view can project by `ctx.sender` (`:127`,
  `:132`).
- A schema (`CapabilityPublic` / `CapabilitySecret` / `Observation`), an
  `observe(target, action)` reducer that charges and reveals in one transaction,
  recorded disclosure preferred over a live join to current truth, traces
  treated as a subscription surface, and a nonleakage target stated as
  observable- transcript equality rather than equal visible values (`:131-141`).

**Left open:** the five follow-ups the thread lists (`:145-149`) - rollback,
cross-module concurrency, scheduled-reducer sequencing, subscription guarantees,
replication determinism - plus the checks embedded in its findings (dominance,
racing probes, canary, paired worlds). All eight are `docs/Open-Questions.md`
Q1-Q8.

## Thread 2 - the transport audit

**Source:** [`delo-e9b61659.md`](../.hermes/research/delo-e9b61659.md), 155
lines.

**Asked:** read `Maintain/Source/Transport` in the consuming repository, name
the invariants it assumes about browser and page, where each breaks, which
breaks are silent, with file and line for every claim, saying where it infers
and what it would run.

**Settled** (`:12-144`) - nine findings, in the thread's order:

1. Post identity is stable, article _position_ is not; `toggle()` sends an
   index, so a reorder between read and click is a time-of-check/time-of-use
   failure that still returns the original identity (`:16-26`).
2. A URL is a locator, not an identity constraint: `wanted` is populated only
   for the id variant, so a URL target snapshots index 0 (`:28-38`).
3. "An article exists" is treated as "the post is ready"; a partial render is
   silent, and an unrelated article can satisfy readiness (`:40-54`).
4. Extraction assumes the DOM preserves meaning - relative permalinks,
   `data-testid`, English metric keywords - and turns "not observed" into `0`,
   `false` or `""` (`:56-74`).
5. Background visibility is handled asymmetrically: `reveal()` runs only for an
   id target in a temporary tab, and swallows its own errors (`:76-84`).
6. "Network idle" is a quiet timestamp, not an empty in-flight set (`:86-94`).
7. A completed click is not a verified write: the settle result is discarded,
   `now` defaults to the desired state when the control is unreadable, reload
   verification can return null, and existing tabs are never reloaded
   (`:96-116`).
8. CDP failures are loud, but `waitFor()` catches every evaluation error and
   keeps polling, so a permanent error surfaces as a generic readiness timeout
   (`:118-128`).
9. Cleanup assumes attachment succeeded: `withPage()` attaches before its
   cleanup try, so an attachment failure leaks the tab, and close errors are
   suppressed (`:130-138`).

The sentence worth keeping is the thread's own conclusion: the strongest signals
concern connection and element availability, the weakest concern the _meaning_
of a successful result (`:144`).

**Left open:** every finding is a source reading plus an inference - the thread
says so at `:14` - and none was run. It names the experiment for each and five
follow-ups (`:146-150`). Nine experiments, `docs/Open-Questions.md` Q9-Q17.

# Scope, and the parity with TimeLine

Agentic is not TimeLine's toolkit. It is a local agent and swarm: it gathers links,
browses, researches, and organises that work in SpacetimeDB. TimeLine is a product -
an archive, a site, a decision corpus and the tooling that maintains them.

They share a vocabulary (capabilities, ports, leases, evidence) and one direction of
travel: **research done here can be fed into the DELO project as findings.** Nothing
travels the other way, and nothing here is required for TimeLine to build.

## What Agentic is

| Capability | What it does | State class |
| --- | --- | --- |
| `browser-cdp` | attach to or launch a browser, evaluate in a page, dispose | holds-transport |
| `chat-driver` | ask a chat product, verify mode and model, read the answer and its trace | holds-transport |
| `design-verification` | measure a built page and produce evidence, with no design baked in | stateless |
| `skill-contract` | discover, validate and digest a corpus of skill documents | stateless |
| `runtime` | admission, leases with a fence, per-run budgets, cleanup | holds-workspace |
| `spacetime-catalog` | the database rules: capabilities, generations, leases, probes, trace | stateless |

## What TimeLine is

| Surface | What it does |
| --- | --- |
| the archive | posts, authors, observations, media, membership - written only through `observe` |
| the site | a prerendered Astro surface that reads the archive and never scrapes |
| the RFC corpus | 76 decision records, their frontmatter contract and the issue board that publishes them |
| design verifiers | three probes that measure the built workspace, showcase and spread against a written spec |
| the tool layer | 125 TypeScript modules: CLI, transport, store, model, command |
| the gates | `pnpm run check`: RFC frontmatter, skills, compatibility, state, tokens, position, tests |

## Parity: what one has that the other does not

| Concern | Agentic | TimeLine |
| --- | --- | --- |
| A browser session | yes - attach or launch, with ownership recorded | yes, but only to verify its own pages |
| Asking a chat product | yes - mode and model verified, trace captured | no |
| Research corpus | yes - threads captured as records with frontmatter and a catalogue | no |
| A published site | no | yes |
| An archive of records | no | yes |
| A decision corpus with a contract | no | yes - 76 records, gated |
| Issue-board synchronisation | no | yes - `rfc_board.py`, read-only by default |
| Leases and admission | yes - the runtime's whole purpose | no |
| Design measurement | yes - the engine, with no design in it | yes - the engine plus the site's own numbers |
| Skill validation | yes - generic corpus rules | yes - its own 13 skills and their contract |
| Persistence | SpacetimeDB, as the scheduler and the record | files plus SpacetimeDB for the live view |

## The one direction of travel

A research finding from Agentic becomes a card in the DELO board, and the card is
written in TimeLine's terms: a file, a claim, a command that settles it. That is the
whole interface. Agentic does not import TimeLine's code, does not read its database,
and is not a dependency of its build.

## What this rules out

- Agentic does not grow a copy of the site, the archive or the RFC board.
- TimeLine does not grow a copy of the runtime, the chat driver or the research
  corpus.
- A capability that both need - measurement, skill validation - lives in Agentic as
  the engine and in TimeLine as the site's own expectations, which is why the two
  versions of `design-verification` are not duplicates.

# SpacetimeDB Catalog

The agentic database model as rules and data: capabilities, generations, leases
with a fence, probes, beliefs and trace. It lives here rather than in the
consuming repository because none of it needs SpacetimeDB to be true, and a
module that imported the server SDK to hold it could not be tested without a
published database.

This is an adoption note. It says what the catalog is, which invariant each rule
enforces, and exactly what TimeLine copies to adopt it in one step.

## What this package is

- **`src/Tables.ts` is the catalog** - six tables described as data: name,
  columns with a type and a key, and whether a table is append-only. A module's
  `Schema.ts` renders its `table(...)` declarations from this, and a reader
  diffs two versions of it to see what a generation changed.
- **`src/Context.ts` is the seam** - the whole of what a rule is given: who is
  calling, what time it is, and a way to read, insert, update and delete one row
  by key. Five members, and no more, because everything else on a module's `ctx`
  is a way to escape the rules rather than to state them.
- **`src/Reducers.ts` is the behaviour** - `bind`, `claim`, `renew`, `probe` and
  `recordTrace` as pure functions over `Context`. Each answers a result rather
  than throwing, so a refusal is a value a caller handles and a test asserts.
- **`Test/Fake.ts` is the instrument** - an in-memory context that records every
  write, refuses to rewrite a table the catalog marks append-only, and checks
  every row against the catalog's columns and types.

The seam is the point. `spacetimedb/server` is not a dependency of this
workspace, and adding one to test rules would be backwards: the rules are what
must be true of the module, so they have to be checkable before a module exists.

## The six tables

- **`capability`** - keyed by `<name>@<version>`, append-only. What a manifest
  may name. Rows are declared by the module's `init`, from its own package
  manifests; there is no reducer that declares one, because a declaration is a
  deployment.
- **`generation`** - keyed by a sequence number as a decimal string,
  append-only. What was bound, by whom, at what time, and which capabilities the
  manifest named. A generation is superseded by the existence of its successor,
  never by a column someone rewrote.
- **`lease`** - keyed by the resource id, current state. Who holds what, under
  which generation, at which fence, until when. Never deleted: deleting the row
  would erase the fence, and a fence that restarts at one is a fence a late
  worker can satisfy.
- **`probe`** - keyed by `<property>:<version>:<audience>`, append-only. What a
  specific audience learned about a specific property version, and when. It has
  no column for the private value, and no rule is ever given it.
- **`belief`** - keyed by `<audience>:<property>:<version>`, disposable. What an
  audience inferred from the observations it holds. A belief grants no access,
  which is why it may carry an interpretation while `probe` may not carry the
  truth.
- **`trace`** - keyed by `<at>:<kind>:<subject>`, append-only. What happened,
  keyed by content so a replay appends nothing. Evidence, not authority.

## The four invariants

### Coherence - one bound generation, and no writes under an old one

`bind` appends a generation and refuses a manifest that names a capability the
catalog does not declare. Every other rule refuses to write under a generation
that is not the latest bound one.

The mechanism is worth stating, because it looks like a trick and is a
consequence. Generation ids are dense sequence numbers, and `bind` computes a
successor's id as its parent's sequence plus one. So "is this generation the
latest?" is answered by one key lookup - does `sequence + 1` exist - in a
context that finds by key and cannot scan. The same property is what leaves the
predecessor byte-identical: a bind inserts a row and updates nothing, and the
catalog marks the table append-only so a context cannot quietly rewrite it.

A bind names its predecessor and that predecessor must still be the latest, so
two binds of the same parent are a compare-and-swap rather than a fork: in a
serialised database the second sees the first one's successor and is refused.

Proved by:
`bind appends a successor and leaves the previous generation untouched`,
`bind refuses a manifest that names a capability the catalog does not declare, and writes nothing`,
`bind refuses a predecessor that is already superseded, or that was never bound`,
and
`a claim or a probe under a superseded generation is refused, and writes nothing`.

### Fence - a late worker cannot write

`claim` increments the fence. `renew` and `probe` refuse a fence that is not the
lease's current one, and refuse without writing, so a worker whose lease expired
cannot write as though it still held it.

Expiry is a comparison, not a stored transition: a row that says `held` but
whose deadline has passed is expired, because no timer has to have run for that
to be true. An expired lease is re-claimed, which increments the fence - renewal
extends a lease and never resurrects one. The row is never deleted, so the fence
never restarts at one.

Proved by:
`the fence: a claim past expiry increments it, and the first fence can no longer renew`,
`a claim while a lease is live is lease-held, and writes nothing`,
`an expired lease cannot be renewed back to life`, and
`a probe carrying a stale fence is refused, and writes nothing`.

### Disclosure - one row, to one audience, at one version

`probe` writes exactly one audience-scoped row and returns only that row. The
private value is never an argument, so it is never a column: a reader cannot
select what the database was never told.

The row id is `<property>:<version>:<audience>`, which makes the reveal its own
receipt and its own allowance. A second probe by the same audience at the same
version finds the row and is refused as `budget-exhausted`, so no second row has
to record a charge, and two racing probes cannot both spend a one-use
allowance - the database serialises them and the loser finds the winner's row.

The audience is `ctx.sender`, never an argument: a caller-supplied audience
would be a caller-supplied identity, and the recipient of a reveal is not
something the caller names.

Proved by:
`a probe writes exactly one row, carrying the audience and the property version, not the private value`,
`a second probe by the same audience at the same property version is budget-exhausted, and writes nothing`,
and `the allowance is per audience, so another holder gets its own observation`.

### Time - one clock, and it is the context's

Every timestamp a row carries is `ctx.now`. No function reads a wall clock, and
the lags between times are differences the caller computes, because a reducer
that read `Date.now()` could not be replayed and a test could not decide what
time it is.

Proved by:
`every timestamp a row carries is ctx.now, so a fake clock decides the whole transcript`.

## Adopting it in TimeLine, in one step

One deliberate step, five parts. Everything but the fourth is a copy.

1. **Copy the rules and the catalog, unchanged.** The exact list is below. They
   compile as they are; nothing in them names TimeLine.
2. **Render `Schema.ts` from `Tables.ts`.** Each table becomes one `table(...)`
   declaration, each column one `t.*` call. The mapping is mechanical: `string`
   is `t.string()`, `number` is `t.u64()`, `boolean` is `t.bool()`, `json` is
   `t.string()` holding the array's text, `optional-string` is
   `t.string().optional()`, `optional-number` is `t.u64().optional()`, a
   `primary` key takes `.primaryKey()` and an `index` key `.index()`.
3. **Write the adapter that satisfies `Context`.** This is the only new code
   TimeLine writes, and it is the five members below.
4. **Seed `capability` in `init`** from the package manifests TimeLine already
   has, so the first `bind` has something to check against.
5. **Register the five rules as reducers**, with their arguments as the `t.*`
   declarations the module already uses for `observe` and `record`.

### The declarations to copy

From `src/Tables.ts`: `ColumnTypes`, `ColumnType`, `Column`, `Table`,
`TableName`, `Tables`, `TableNames`, `PrimaryKey`.

From `src/Context.ts`: `Value`, `Row`, `Context`.

From `src/Reducers.ts`: `Reason`, `Refusal`, `Result`, `ProbeResult`, `Bind`,
`Claim`, `Renew`, `Probe`, `Trace`, `bind`, `claim`, `renew`, `probe`,
`recordTrace`, and `Latest` - the coherence predicate, which a view that lists
live generations needs for the same reason a rule does.

That is all of both files. Nothing is copied selectively, because a rule that
travelled without its table description would be a rule checked against a
different catalog than the module renders.

### The adapter, which is the only code TimeLine writes

- **`sender`** - `ctx.sender`, the module's `Identity`, rendered to a string
  once at the boundary. Every rule compares that rendering, so a rule never
  compares an identity to a caller-supplied argument.
- **`now`** - `ctx.timestamp` in whatever integer unit the module uses. The
  rules only add, compare and store it, so the unit is TimeLine's choice; a
  `ttl` passed to `claim` or `renew` has to be in that same unit. A `u64` column
  is a `bigint` in the module's TypeScript, so the adapter converts at the
  boundary and the rules stay on `number`.
- **`find(table, id)`** - `ctx.db[table].id.find(id)`.
- **`insert(table, row)`** - `ctx.db[table].insert(row)`.
- **`update(table, row)`** - `ctx.db[table].id.update(row)`.
- **`delete(table, id)`** - `ctx.db[table].id.delete(id)`. No rule in
  `Reducers.ts` deletes a row; the member exists because the context has it.

### The one thing that cannot be copied

The module's own `spacetimedb` import: `schema()`, `table()`, `reducer()`,
`ctx.db`, `t.*`, `init` and the client callbacks. Those exist only inside a
published module - there is no `ctx.db` before a module is compiled and no
`table()` without the server package - so a rule that named them could not be
tested without a database, and would pin the rules to one server version.

That is the whole of what stays behind in TimeLine: `Schema.ts`, the `Context`
adapter, `init`, and the five `spacetimedb.reducer(...)` registrations. The
rules, the catalog and the interface travel unchanged, and the tests that prove
them travel with them.

## What is deliberately absent

- **No reducer declares a capability.** A declaration is a deployment, and a
  reducer that could add one would let a caller widen the set its own manifest
  is checked against.
- **No belief writer.** Deriving a belief is the actor's own act on observations
  it already holds; the table is described because its shape is the third thing
  the disclosure rule needs, and deleting a belief loses nothing.
- **No `release` reducer.** These five rules never end a lease early; expiry is
  the recovery mechanism and every write checks it. `lease.state` is read by
  every fenced write, so a module that wants early release adds `release` and
  the rules already treat `released` as claimable.
- **No scan.** Every question a rule asks is one key lookup, which is why
  generation ids are dense sequence numbers and why a probe's key is its
  receipt. A rule that needed `iter()` would be a rule that cannot be stated
  here.
- **No wall clock, no credentials, no private values.** The catalog describes
  what is stored; a secret the rules are never given is not merely filtered from
  a reader, it never enters the database.

# Agentic

Capabilities, adapters and the runtime that leases them. Nothing here knows
about a particular product; the product consumes this workspace and supplies its
own composition root.

## The four layers

| Layer            | Owns                                       | May import                   |
| ---------------- | ------------------------------------------ | ---------------------------- |
| Capability       | a use case and the port it needs           | its own port and data shapes |
| Adapter          | one port against one mechanism or provider | that port, that provider     |
| Workflow         | a composition of capabilities              | capabilities only            |
| Composition root | the binding of ports to adapters           | everything                   |

The arrows point inward and nothing imports a root. A capability that imports an
adapter is a defect rather than a shortcut: it makes the capability untestable
without the provider, which is the one property this workspace exists to keep.

## The manifest

Every package carries `module.json`. The field that does the most work is
`state`, because it decides what a harness may warm and what it must revalidate
at admission:

- `stateless` - nothing to check.
- `holds-transport` - a liveness probe, because an open socket can be dead.
- `holds-identity` - a freshness check, because a token expires while pooled.
- `holds-workspace` - a reset proof, because a dirty workspace answers wrongly.

## The lifecycle

prepare reusable infrastructure → lease an isolated context → bind identity and
task → validate credentials, connections and versions → execute within a fresh
budget → persist only authorised artefacts → destroy or prove a reset.

Preparation improves latency. Admission establishes correctness. They are
different acts, and merging them is how a warm pool starts returning wrong
answers quickly.

## The packages

| Package                        | Kind       | Why it exists                                                                      |
| ------------------------------ | ---------- | ---------------------------------------------------------------------------------- |
| `@agentic/browser-cdp`         | adapter    | the shared execution substrate: attach versus launch, bounded evaluation, disposal |
| `@agentic/chat-driver`         | capability | ask a chat product, correlate turns, verify mode and model, read answer and trace  |
| `@agentic/design-verification` | capability | measure a built page and produce evidence; expectations stay with the site         |
| `@agentic/skill-contract`      | capability | discover, validate and package a skill corpus                                      |
| `@agentic/runtime`             | runtime    | admission, leases, budgets, cancellation, cleanup, telemetry                       |

## Deliberately not here

Domain policy and write authority stay with their consumers until a second
workflow demonstrates a stable contract: store, model, command, vote, ingest,
board synchronisation, gateway maintenance. A generic name is not a port.
