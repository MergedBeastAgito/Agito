//! `agentic.verify.design`, exercised against a scripted transport.
//!
//! No browser and no network: the transport answers what the test told it to,
//! and a deadline passes because the test advanced its own clock.

use std::{
	sync::{Arc, Mutex},
	time::{Duration, Instant},
};

use AgenticDesignVerification::{
	Browse,
	Code,
	Create,
	Error,
	Failure,
	Invariant,
	Measurement,
	Outcome,
	Verdict,
	Verifier,
	Verify,
	Want,
};
use AgenticRuntime::{BlockOn, Call, Clock, FutureBox, Never, Pace, Signal};
use serde_json::{Value, json};

fn Span(Milliseconds:u64) -> Duration { Duration::from_millis(Milliseconds) }

/// A clock the test owns, and a wait that advances it.
struct TestPace {
	Now:Mutex<Instant>,
	Fire:Mutex<Option<Arc<Signal>>>,
}

impl TestPace {
	fn New() -> Arc<Self> { Arc::new(Self { Now:Mutex::new(Instant::now()), Fire:Mutex::new(None) }) }

	fn Firing(self: &Arc<Self>, Signal:Arc<Signal>) -> Arc<Self> {
		*self.Fire.lock().unwrap() = Some(Signal);
		self.clone()
	}
}

impl Clock for TestPace {
	fn Now(&self) -> Instant { *self.Now.lock().unwrap() }
}

impl Pace for TestPace {
	fn Sleep(&self, Span:Duration) -> FutureBox<'_, ()> {
		// A pace that is about to fire a signal models a caller giving up, not
		// time passing: it advances a millisecond, fires, and never
		// finishes its wait - so the cancellation and not the deadline
		// is what stops the call.
		if let Some(Signal) = self.Fire.lock().unwrap().take() {
			*self.Now.lock().unwrap() += Duration::from_millis(1);
			Signal.Fire();
			return Box::pin(std::future::pending());
		}
		*self.Now.lock().unwrap() += Span.max(Duration::from_millis(1));
		Box::pin(std::future::ready(()))
	}
}

/// A transport that answers what the test scripted.
struct Script {
	/// The wire answer: one object per selector.
	Answer:Vec<Value>,
	/// A failure instead of an answer.
	Fail:Option<String>,
	/// Never answer at all.
	Hang:bool,
}

struct FakeTransport {
	Script:Mutex<Script>,
	Asked:Mutex<Vec<Vec<String>>>,
}

impl FakeTransport {
	fn New(Answer:Vec<Value>) -> Arc<Self> {
		Arc::new(Self {
			Script:Mutex::new(Script { Answer, Fail:None, Hang:false }),
			Asked:Mutex::new(Vec::new()),
		})
	}

	fn Failing(Message:&str) -> Arc<Self> {
		let Fake = Self::New(vec![]);
		Fake.Script.lock().unwrap().Fail = Some(Message.to_string());
		Fake
	}

	fn Hanging() -> Arc<Self> {
		let Fake = Self::New(vec![]);
		Fake.Script.lock().unwrap().Hang = true;
		Fake
	}

	fn Asked(&self) -> Vec<Vec<String>> { self.Asked.lock().unwrap().clone() }
}

impl Browse for FakeTransport {
	fn Measure<'a>(&'a self, Selectors:&'a [String], _Call:Call) -> FutureBox<'a, Result<Vec<Value>, Failure>> {
		self.Asked.lock().unwrap().push(Selectors.to_vec());
		let Script = self.Script.lock().unwrap();
		let Fail = Script.Fail.clone();
		let Answer = Script.Answer.clone();
		let Hang = Script.Hang;
		Box::pin(async move {
			if Hang {
				std::future::pending::<()>().await;
			}
			if let Some(Message) = Fail {
				return Err(Failure::Of(Message));
			}
			Ok(Answer)
		})
	}
}

fn Built(Transport:&Arc<FakeTransport>, Pace:&Arc<TestPace>) -> Verifier { Create(Transport.clone(), Pace.clone()) }

fn Measured(
	Verifier:&impl Verify,
	Pace:&TestPace,
	Selectors:&[String],
	Budget:Duration,
) -> Result<Vec<Measurement>, Error> {
	let Call = Call::Within(Budget, Pace, Arc::new(Never));
	BlockOn(Verifier.Measure(Selectors, Call))
}

fn Selectors(Names:&[&str]) -> Vec<String> { Names.iter().map(|Name| Name.to_string()).collect() }

/// The selectors a test asks about, none of them a design of this package.
fn Asked() -> Vec<String> { Selectors(&["[data-rail]", "[data-drawer]"]) }

fn Present(Selector:&str, Value:Value) -> Value { json!({ "selector": Selector, "present": true, "value": Value }) }

fn Absent(Selector:&str) -> Value { json!({ "selector": Selector, "present": false, "value": null }) }

fn Readings() -> Vec<Measurement> {
	vec![
		Measurement::Present("[data-rail]", json!(320)),
		Measurement::Present("[data-drawer]", json!(320)),
	]
}

#[test]
fn measure_asks_for_every_selector_once_and_answers_in_the_order_it_was_given() {
	let Pace = TestPace::New();
	let Transport = FakeTransport::New(vec![Absent("[data-drawer]"), Present("[data-rail]", json!(320))]);
	let Verifier = Built(&Transport, &Pace);
	let Measurements = Measured(&Verifier, &Pace, &Asked(), Span(2_000)).expect("measured");
	assert_eq!(Transport.Asked(), vec![Asked()], "one request, once");
	assert_eq!(Measurements.len(), 2);
	assert_eq!(Measurements[0].Selector, "[data-rail]");
	assert_eq!(Measurements[1].Selector, "[data-drawer]");
}

# Rust workspace

The TypeScript toolkit in `Package/` rebuilt as seven crates, keeping the port shapes and the
failure classification the TypeScript suites already enforce. The TypeScript packages stay as they
are: they are the specification, not a predecessor to delete.

The layout follows the house style of `github.com/PlayForm/Run`: one directory per crate under
`Crate/`, `Source/Library.rs` as the library root, `Fn/` for operations and `Struct/` for data and
contracts, PascalCase module names, and one `Target/` at the workspace root.

---

## Crates

**`agentic-runtime`** - `Crate/Runtime/`, library `AgenticRuntime`. The execution core: `Call`,
`Cancellation`, `Clock`, `Pace`, the `Bound`/`Race` combinators, the std `BlockOn`, and the policy
engine (admission, leases with a fence, per-run budgets, release that never fails). Dependencies:
none. An optional `tokio` feature adds a `Pace` implementation and is off by default.

**`agentic-skill-contract`** - `Crate/SkillContract/`. Corpus discovery, validation and a stable
digest. Dependencies: `agentic-runtime`, `serde`, `serde_json`, `sha2`.

**`agentic-chat-driver`** - `Crate/ChatDriver/`. `agentic.chat.ask` over a page it declares itself.
Dependencies: `agentic-runtime`, `serde_json`. No browser, no executor.

**`agentic-browser-cdp`** - `Crate/BrowserCDP/`. The concrete substrate: attach versus launch,
bounded evaluation, disposal that closes only what it launched. Dependencies: `agentic-runtime`,
`serde_json`, `tokio`, `tokio-util`, `futures-util`, `chromiumoxide`. The only crate that names
`chromiumoxide`.

**`agentic-design-verification`** - `Crate/DesignVerification/`. Measurement and evidence, holding
no design of its own. Dependencies: `agentic-runtime`, `serde_json`.

**`agentic-workflow`** - `Crate/Workflow/`. The bridge from the substrate to the two consumer-owned
ports, and one composition that uses four capabilities. Dependencies: `agentic-runtime`,
`agentic-chat-driver`, `agentic-design-verification`, `agentic-skill-contract`,
`agentic-browser-cdp`, `serde_json`.

**`agentic`** - `Crate/Agentic/`. Wiring, the manifest table and the executable. Dependencies: all
six above, `clap`, `tokio`, `serde_json`.

---

## The two rules the dependency graph enforces

**A capability's required surface is runtime-neutral.** `agentic-chat-driver`,
`agentic-skill-contract` and `agentic-design-verification` do not name `tokio` in their dependency
lists, and nothing they pull in resolves it. Time arrives as a `Call` (a deadline and a
cancellation) and a `Pace` (a clock and a wait) supplied by the composition root.

**No capability imports an adapter.** `agentic-chat-driver` declares its own `Page` trait and
`agentic-design-verification` its own `Browse`; `agentic-workflow` owns the bindings that implement
them over `agentic-browser-cdp`'s `Browse`. There is no shared ports crate, and the translations are
deliberately not factored into a "universal browser request" type.

Both are checkable, and the checks are the first two commands below.

--- FILE: Crate/Runtime/Source/Fn/BlockOn.rs ---
//! Drive a future to completion on this thread.
//!
//! A capability's port is a boxed future, and a test that exercises one has no
//! executor - that is the point of the port being runtime-neutral. This is the
//! smallest honest answer: poll until ready, yielding between polls.
//!
//! It is not an executor and does not pretend to be one: nothing here parks or
//! schedules, so it is correct for futures that are driven by something else
//! (a signal fired on another thread, a fake clock advanced by a fake wait) and
//! wrong for a future that only ever resolves from a real timer. A test that
//! wants time to pass advances a clock instead of waiting for one.

use std::{
	future::Future,
	pin::pin,
	task::{Context, Waker},
};

/// Poll `Work` until it produces a value.
pub fn BlockOn<Work:Future>(Work:Work) -> Work::Output {
	let mut Work = pin!(Work);
	let Waker = Waker::noop();
	let mut Context = Context::from_waker(Waker);
	loop {
		if let std::task::Poll::Ready(Value) = Work.as_mut().poll(&mut Context) {
			return Value;
		}
		std::thread::yield_now();
	}
}

--- FILE: Crate/Runtime/Source/Fn/Bound.rs ---
//! Bound one operation by a deadline and the caller's cancellation.
//!
//! This is the combinator every port's guarantee rests on, and it is written
//! against three traits - a boxed future, a `Pace` and a `Cancellation` - so no
//! capability has to know which executor is running it.
//!
//! The two bounds are separate on purpose. A caller's cancellation and a passed
//! deadline are both "no answer", but they are not the same fact: one is the
//! caller giving up and the other is the page not finishing, and a consumer
//! that collapsed them without keeping `detail` could not tell a slow page from
//! its own impatience.

use crate::{
	Fn::Race::{Either, Race},
	Struct::Call::{Call, Expired, FutureBox, Pace},
};

/// Resolve with `Work`'s value, or with why it was cut short.
///
/// A signal that has already fired rejects before `Work` is polled at all: a
/// caller that has given up must not cause one more page operation to start.
pub fn Bound<'a, T:'a>(Work:FutureBox<'a, T>, Call:&'a Call, Pace:&'a dyn Pace) -> FutureBox<'a, Result<T, Expired>> {
	Box::pin(async move {
		if Call.Cancellation.IsCancelled() {
			return Err(Expired::Cancelled);
		}
		let Guard = async {
			let Wait = Pace.Sleep(Call.Remaining(Pace));
			let Stop = Call.Cancellation.Cancelled();
			match Race(Wait, Stop).await {
				Either::Left(()) => Expired::Timeout,
				Either::Right(()) => Expired::Cancelled,
			}
		};
		match Race(Work, Guard).await {
			Either::Left(Value) => Ok(Value),
			Either::Right(Why) => Err(Why),
		}
	})
}

--- FILE: Crate/Runtime/Source/Fn/Race.rs ---
//! Race two futures, first one to finish wins.
//!
//! The primitive the deadline is built from. It is written here rather than
//! taken from a combinator crate because the runtime core is std-only: a
//! `select` is thirty lines of `poll`, and a dependency that exists to save
//! thirty lines is a dependency every capability inherits.

use std::{future::Future, pin::pin};

/// Whichever side finished first.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Either<Left, Right> {
	/// The left side.
	Left(Left),
	/// The right side.
	Right(Right),
}

impl<Left, Right> Either<Left, Right> {
	/// Whether the left side won.
	pub const fn IsLeft(&self) -> bool { matches!(self, Either::Left(_)) }
}

/// Resolve with whichever of `Left` and `Right` resolves first.
///
/// Both sides are polled in order on every wake, so a side that is already
/// ready wins without the other being polled at all. The loser is dropped,
/// which is the only cancellation Rust gives a future for free - and the reason
/// the port says a caller's cancellation stops *its wait* rather than closing a
/// page.
pub async fn Race<Left, Right>(Left:Left, Right:Right) -> Either<Left::Output, Right::Output>
where
	Left: Future,
	Right: Future, {
	let mut Left = pin!(Left);
	let mut Right = pin!(Right);
	std::future::poll_fn(|Context| {
		if let std::task::Poll::Ready(Value) = Left.as_mut().poll(Context) {
			return std::task::Poll::Ready(Either::Left(Value));
		}
		if let std::task::Poll::Ready(Value) = Right.as_mut().poll(Context) {
			return std::task::Poll::Ready(Either::Right(Value));
		}
		std::task::Poll::Pending
	})
	.await
}

//! Filling a script's placeholders.
//!
//! The expressions this workspace sends to a page are JavaScript files, not
//! string literals assembled by `format!`. A file cannot interpolate, so a value
//! the caller supplies is written where it belongs as a `{{name}}` placeholder
//! and replaced at call time. That is the whole of the mechanism, and it is here
//! rather than in either crate that uses it because both a capability and the
//! workflow that composes it need it, and two copies of a substitution rule is
//! two chances for the rule to differ.
//!
//! What the placeholder must not be is silent. A script whose caller forgets a
//! parameter keeps `{{busy}}` in the expression, and a page that receives it
//! throws a `SyntaxError` - a failure with a reason, not a selector that quietly
//! matches nothing. [`Unfilled`] is the same check without a page: it names the
//! placeholder a string still carries, so a test can assert that every script is
//! filled completely and that the check itself bites.

/// Replace every `{{name}}` in `Template` with the value `Parameters` gives it.
///
/// A parameter the template does not name is not an error - it is simply not
/// used - because a caller may build one parameter set for several scripts. A
/// placeholder no parameter names is left in place, deliberately: [`Unfilled`]
/// is what reports it.
pub fn Fill(Template:&str, Parameters:&[(&str, &str)]) -> String {
	let mut Text = Template.to_string();
	for (Name, Value) in Parameters {
		Text = Text.replace(&format!("{{{{{Name}}}}}"), Value);
	}
	Text
}

/// The name of the first placeholder `Text` still carries, if it carries one.
pub fn Unfilled(Text:&str) -> Option<String> {
	let Start = Text.find("{{")?;
	let Rest = &Text[Start + 2..];
	let End = Rest.find("}}")?;
	Some(Rest[..End].to_string())
}

//! The seam between the runner and whoever actually owns the lease.
//!
//! `Documentation/Observability.md` puts every decision, counter, lease and record in a
//! table changed by a reducer, and puts only the effect itself in Rust. This
//! trait is that sentence as a type: the runner holds an [`Authority`] and asks
//! it, and it holds no lease state of its own to disagree with.
//!
//! Two implementations exist in the shape of things, and only one of them is in
//! this crate:
//!
//! - **A reducer adapter**, which is the authority in production: `Claim` is the
//!   `claim` reducer, `Renew` is `renew`, `Admit` reads the lease row the
//!   reducer already wrote, and `Present` is the `record` reducer. It is not
//!   here because it needs a SpacetimeDB client, which belongs to an adapter
//!   crate.
//! - **[`crate::Struct::StandIn::StandIn`]**, which is the local stand-in: the
//!   same six checks the reducer makes, over a local table, so the port can be
//!   exercised with no database at all. It is a test double. It is named as one,
//!   it is not wired into a production composition root, and the mirror test
//!   pins it to the reducer so it cannot quietly become a second authority.
//!
//! That is the difference the review asked to have made explicit: this crate
//! *runs* an effect and *presents* a ticket. It decides nothing. The thing that
//! decides is a value substituted at construction, and substituting a database
//! for the stand-in changes no line of the runner.

use std::time::Duration;

use crate::Struct::{
	Call::FutureBox,
	Error::Error,
	Port::{Budget, Destination, Lease, Ticket},
};

/// One admitted effect: the ticket to present, and the accounting the admission
/// opened.
///
/// The guard is opaque on purpose. The runner holds it for exactly as long as
/// the work runs and never reads it, so the count of unsettled local futures is
/// the authority's business and cannot be inspected, cached or reconstructed by
/// the runner. Dropping it settles the future whether the work finished, failed
/// or was abandoned - which is the only way an abandoned future gets counted
/// correctly, since Rust has no `finally`.
pub struct Admitted {
	/// The authority the effect presents when it lands.
	pub Ticket:Ticket,
	/// Dropping this settles the local future this admission opened.
	///
	/// Never read: the runner holds it for exactly as long as the work runs, and
	/// the count of unsettled futures is the authority's business.
	#[expect(dead_code, reason = "held only for its Drop; the runner must not read it")]
	pub(crate) Settle:Box<dyn Send>,
}

/// Everything the runner needs from whoever owns the leases.
///
/// `Claim`, `Renew` and `Release` are the verbs a reducer already has; `Admit` is
/// the one that is not a reducer, because it is the optimisation rather than the
/// record - it refuses doomed work early, charges the run's allowance and mints
/// the ticket. A reducer adapter implements it by reading the lease row and
/// returning the ticket that row implies.
pub trait Authority: Destination {
	/// The generation this authority records under.
	fn Generation(&self) -> u64;
	/// Claim a resource for one run, with that run's budget, or fail because one
	/// is held.
	fn Claim<'a>(
		&'a self,
		Resource:&'a str,
		Ttl:Duration,
		Budget:Option<Budget>,
	) -> FutureBox<'a, Result<Lease, Error>>;
	/// Extend a lease, only for the holder of the current fence, and only
	/// forwards.
	fn Renew<'a>(&'a self, Lease:Lease, Ttl:Duration) -> FutureBox<'a, Result<Lease, Error>>;
	/// Give the authority back. Never fails, and is safe twice.
	fn Release<'a>(&'a self, Lease:Lease) -> FutureBox<'a, ()>;
	/// The admission step: refuse a lease whose authority ended, charge the run's
	/// allowance, count the local future, and mint the ticket it will present.
	fn Admit<'a>(&'a self, Lease:Lease) -> FutureBox<'a, Result<Admitted, Error>>;
}

//! What every operation is bounded by, and the facilities needed to apply a
//! bound without naming an executor.
//!
//! A deadline is an explicit parameter, not a global: [`Call`] carries it,
//! every port method takes one, and a provider that wants a different bound is
//! handed a different `Call` rather than reading a process-wide setting. The
//! caller's cancellation rides beside it, so "I gave up" and "the deadline
//! passed" stay two facts even where a port collapses them into one code.

use std::{
	future::Future,
	pin::Pin,
	sync::{
		Arc,
		Mutex,
		MutexGuard,
		atomic::{AtomicBool, Ordering},
	},
	task::{Poll, Waker},
	time::{Duration, Instant},
};

/// One boxed future.
///
/// Every port method returns one, which is what keeps `Arc<dyn Port>` possible:
/// a port with an `async fn` in it is not object-safe, and a port that is not
/// object-safe cannot be substituted by a fake in a test.
pub type FutureBox<'a, T> = Pin<Box<dyn Future<Output = T> + Send + 'a>>;

/// The caller's cancellation.
///
/// Two readings of one fact: a flag, for a provider that checks before it
/// starts work, and a future, for one that must stop in the middle. An
/// implementation must be race-safe and must resolve immediately when already
/// cancelled - a future that only resolves on the *next* fire would make an
/// already-abandoned call wait for an event that has happened.
pub trait Cancellation: Send + Sync {
	/// Whether the signal has fired.
	fn IsCancelled(&self) -> bool;
	/// Resolves once the signal has fired, or at once if it already has.
	fn Cancelled(&self) -> FutureBox<'_, ()>;
}

/// The only clock a deadline is compared against.
///
/// Supplied rather than reached for, so a test can expire a lease without
/// sleeping and a caller can express a deadline on the same authority the
/// runtime compares it against.
pub trait Clock: Send + Sync {
	/// The current instant, in whatever epoch the caller and the runtime
	/// agreed.
	fn Now(&self) -> Instant;
}

/// The clock, plus the ability to wait.
///
/// A capability paces itself between reads; the wait is supplied so that the
/// pacing does not decide which executor is running. A test supplies a wait
/// that advances its own clock, which is how a poll loop becomes deterministic
/// instead of slow.
pub trait Pace: Clock {
	/// Resolve after `Span`, or at once when it is zero.
	fn Sleep(&self, Span:Duration) -> FutureBox<'_, ()>;
}

/// What one operation is bounded by.
#[derive(Clone)]
pub struct Call {
	/// The instant the operation may no longer continue.
	pub Deadline:Instant,
	/// The caller's cancellation.
	pub Cancellation:Arc<dyn Cancellation>,
}

impl Call {
	/// A call whose deadline is `Span` from `Clock`'s reading now.
	pub fn Within(Span:Duration, Clock:&dyn Clock, Cancellation:Arc<dyn Cancellation>) -> Self {
		Self { Deadline:Clock.Now() + Span, Cancellation }
	}

	/// How long is left, saturating at zero rather than panicking on a clock
	/// that has already passed the deadline.
	pub fn Remaining(&self, Clock:&dyn Clock) -> Duration { self.Deadline.saturating_duration_since(Clock.Now()) }

	/// Whether the deadline has passed. The interval is half-open, so the
	/// instant itself is already outside it.
	pub fn Expired(&self, Clock:&dyn Clock) -> bool { Clock.Now() >= self.Deadline }

	/// Whether the caller has given up.
	pub fn Cancelled(&self) -> bool { self.Cancellation.IsCancelled() }
}

impl std::fmt::Debug for Call {
	fn fmt(&self, Formatter:&mut std::fmt::Formatter<'_>) -> std::fmt::Result {
		Formatter
			.debug_struct("Call")
			.field("Deadline", &self.Deadline)
			.field("Cancelled", &self.Cancellation.IsCancelled())
			.finish()
	}
}

/// Why a bounded operation stopped before it produced a value.
///
/// Two reasons and not one: a consumer that maps both onto its own single code
/// still needs `detail` to say which happened, and a provider that reported
/// "timeout" for its own cancellation would be blaming the page for the caller.
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Expired {
	/// The deadline passed.
	Timeout,
	/// The caller's signal fired.
	Cancelled,
}

impl Expired {
	/// The phrase a failure detail uses.
	pub const fn Name(self) -> &'static str {
		match self {
			Expired::Timeout => "the deadline passed",
			Expired::Cancelled => "the caller cancelled",
		}
	}
}

/// A cancellation that never fires.
///
/// The default for a call nobody can cancel, and the honest way to say "this
/// operation has a deadline and no signal" instead of inventing one.
#[derive(Debug, Default)]
pub struct Never;

impl Cancellation for Never {
	fn IsCancelled(&self) -> bool { false }

	fn Cancelled(&self) -> FutureBox<'_, ()> { Box::pin(std::future::pending()) }
}

/// A cancellation a caller fires when it gives up.
///
/// The waiters are woken rather than polled, so a cancelled call resolves on
/// the thread that fired the signal instead of on the next tick of a loop. A
/// waker that arrives after the fire is answered by the second flag read, which
/// is why the read happens on both sides of the registration.
#[derive(Debug, Default)]
pub struct Signal {
	Fired:AtomicBool,
	Waiters:Mutex<Vec<Waker>>,
}

impl Signal {
	/// A signal nobody has fired.
	pub fn New() -> Arc<Self> { Arc::new(Self::default()) }

	/// Fire it, waking everyone waiting. Firing twice is a no-op.
	pub fn Fire(&self) {
		self.Fired.store(true, Ordering::SeqCst);
		let Waiters = std::mem::take(&mut *Self::Lock(&self.Waiters));
		for Waker in Waiters {
			Waker.wake();
		}
	}

	/// A lock whose poisoning is not a failure: a panicked waiter must not make
	/// cancellation itself unavailable.
	fn Lock<T>(Guard:&Mutex<T>) -> MutexGuard<'_, T> { Guard.lock().unwrap_or_else(|Poison| Poison.into_inner()) }
}

impl Cancellation for Signal {
	fn IsCancelled(&self) -> bool { self.Fired.load(Ordering::SeqCst) }

	fn Cancelled(&self) -> FutureBox<'_, ()> {
		Box::pin(std::future::poll_fn(move |Context| {
			if self.Fired.load(Ordering::SeqCst) {
				return Poll::Ready(());
			}
			Self::Lock(&self.Waiters).push(Context.waker().clone());
			if self.Fired.load(Ordering::SeqCst) {
				return Poll::Ready(());
			}
			Poll::Pending
		}))
	}
}

#[derive(Debug, Default)]
pub struct Wall;

impl Clock for Wall {
	fn Now(&self) -> Instant { Instant::now() }
}

//! The three failure modes `module.json` declares, and no others.
//!
//! One enum, one variant per code, and `detail` kept apart from the message in
//! every one of them: a consumer narrows on the variant, a human reads the
//! message, and a caller that needs to know what was actually observed reads
//! `detail` instead of parsing a sentence.

use std::fmt;

/// The codes, in the order the manifest names them.
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub enum Code {
	/// The request may not proceed: it is malformed, or its authority ended.
	AdmissionRefused,
	/// The resource is held under a newer fence, or a local future it admitted
	/// has not settled.
	LeaseHeld,
	/// The run's own allowance is spent: its deadline passed or its count ran
	/// out.
	BudgetExhausted,
}

impl Code {
	/// The string the manifest uses.
	pub const fn Name(self) -> &'static str {
		match self {
			Code::AdmissionRefused => "admission-refused",
			Code::LeaseHeld => "lease-held",
			Code::BudgetExhausted => "budget-exhausted",
		}
	}

	/// Every code, so a harness can compare the manifest against the type.
	pub const fn All() -> [Code; 3] { [Code::AdmissionRefused, Code::LeaseHeld, Code::BudgetExhausted] }

	/// The code a manifest string names, if it names one of the three.
	pub fn Parse(Value:&str) -> Option<Code> { Code::All().into_iter().find(|Code| Code.Name() == Value) }
}

/// The one error type `claim`, `renew` and `withLease` reject with.
///
/// `admit` does not return one - its port answers a refusal - so a caller has
/// two shapes to handle and only one of them is an error. That asymmetry is the
/// port's: admission answers a question, while a claim or a write either
/// happened or did not.
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
	/// The request may not proceed.
	AdmissionRefused {
		/// The sentence a human reads.
		Message:String,
		/// What was observed, kept apart from the sentence.
		Detail:String,
	},
	/// The resource is held, or a local future it admitted has not settled.
	LeaseHeld {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
	/// The run's allowance is spent.
	BudgetExhausted {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
}

impl Error {
	/// A refusal with nothing observed to add.
	pub fn Refused(Message:impl Into<String>) -> Self {
		Error::AdmissionRefused { Message:Message.into(), Detail:String::new() }
	}

	/// A held resource with nothing observed to add.
	pub fn Held(Message:impl Into<String>) -> Self { Error::LeaseHeld { Message:Message.into(), Detail:String::new() } }

	/// A spent allowance with nothing observed to add.
	pub fn Spent(Message:impl Into<String>) -> Self {
		Error::BudgetExhausted { Message:Message.into(), Detail:String::new() }
	}

	/// The same failure, with what was observed attached.
	pub fn WithDetail(self, Detail:impl Into<String>) -> Self {
		let Detail = Detail.into();
		match self {
			Error::AdmissionRefused { Message, .. } => Error::AdmissionRefused { Message, Detail },
			Error::LeaseHeld { Message, .. } => Error::LeaseHeld { Message, Detail },
			Error::BudgetExhausted { Message, .. } => Error::BudgetExhausted { Message, Detail },
		}
	}

	/// The code this failure carries.
	pub const fn Code(&self) -> Code {
		match self {
			Error::AdmissionRefused { .. } => Code::AdmissionRefused,
			Error::LeaseHeld { .. } => Code::LeaseHeld,
			Error::BudgetExhausted { .. } => Code::BudgetExhausted,
		}
	}

	/// The sentence a human reads.
	pub fn Message(&self) -> &str {
		match self {
			Error::AdmissionRefused { Message, .. }
			| Error::LeaseHeld { Message, .. }
			| Error::BudgetExhausted { Message, .. } => Message,
		}
	}

	/// What was observed, which may be empty.
	pub fn Detail(&self) -> &str {
		match self {
			Error::AdmissionRefused { Detail, .. }
			| Error::LeaseHeld { Detail, .. }
			| Error::BudgetExhausted { Detail, .. } => Detail,
		}
	}
}

impl fmt::Display for Error {
	fn fmt(&self, Formatter:&mut fmt::Formatter<'_>) -> fmt::Result {
		write!(Formatter, "{}: {}", self.Code().Name(), self.Message())
	}
}

//! The port `agentic.runtime` promises, written the way a consumer asks for it.
//!
//! The port states more than data shapes, because two runtimes that agreed on
//! the shapes alone would still disagree about when authority ends:
//!
//! - **An effect carries its authority.** `WithLease` hands the work a
//!   [`Ticket`]: the lease, the fence, the generation and the seal it was
//!   admitted under. An effect is not a bare closure, because a closure has
//!   nothing to present when it arrives late.
//! - **The destination is the guarantee; the admission is the optimisation.**
//!   `WithLease` checks the lease so doomed work does not start, but that check
//!   ran before the effect did any work and cannot say anything about the
//!   instant it writes. The effect presents its ticket to the place that records
//!   it - the reducer, or the local stand-in for one - and that place refuses a
//!   stale fence, a stale generation, a released lease or an expired record. A
//!   design where only the caller checks is a design where an effect admitted
//!   while a lease was live can still land after the lease ended.
//! - **The numbers are not the identity.** A ticket a caller can reconstruct
//!   from what it observed is a ticket any caller can present. The reducer's
//!   identity is `ctx.sender`, which its transport authenticates; this port's
//!   equivalent is a [`Seal`], which the authority records and a caller cannot
//!   derive. Matching generation, resource, lease and fence authorise nothing on
//!   their own.
//! - **A refusal is not an undo.** Rejecting an effect's record does not retract
//!   its external action. A write that landed in a page is still in the page
//!   when the record of it is refused, so the destination records it as
//!   [`Ambiguous`] and quarantines the resource instead of pretending nothing
//!   happened.
//! - **A fence is a number, and expiry is a clock.** A monotonically increasing
//!   fence alone does not prove a lease is live - an expired worker can arrive
//!   before any newer token has reached the resource - so every check compares
//!   the presented fence against the record *and* the record's expiry against
//!   the clock. Either one failing is a refusal.
//! - **A held resource is refused, not queued.** A second claim while the first
//!   is live is `lease-held`. A queue would turn a scheduling decision into a
//!   promise the runtime cannot keep.
//! - **A budget belongs to one run.** The deadline and the operation count are
//!   the caller's, supplied when the run claims its resource. Nothing is
//!   pooled.
//! - **Release never throws, and is safe twice.** Cleanup runs on the failure
//!   path by definition, so it cannot be the step that turns a handled failure
//!   into an unhandled one.
//!
//! Two shapes can carry these rules, and they are not interchangeable. A **local
//! stand-in** keeps the lease table in this process and decides here; it is a
//! test double, and in production it is a second authority however correct its
//! fencing. A **thin effect runner** keeps no lease state at all: it takes a
//! ticket the authority issued, runs the work, and presents the ticket back. This
//! crate is the runner. [`crate::Struct::StandIn`] is the double, it is named as
//! one, and [`crate::Struct::Authority::Authority`] is the seam a reducer adapter
//! replaces it through.
//!
//! One port shape changed in the port: `claim(resource, ttl, budget?)` and
//! `now()`/`reflect()` are on this one trait instead of a separate extension
//! trait, because Rust has no method overloading and an extension trait that
//! repeats `claim` with a different arity cannot exist.

/// The rendering of one admitted caller's identity.
///
/// The reducer's identity is `ctx.sender`: its transport authenticates the
/// caller, so a rule compares that value and never accepts it as an argument. A
/// single process has no transport, so the rendering here is 128 bits drawn from
/// a key the operating system seeds per process. A caller can observe every
/// number a ticket carries - generation, resource, lease id, fence, expiry - and
/// still not present a seal, because those numbers are not the identity. That is
/// the whole answer to "a caller that guesses the right numbers is not the caller
/// that was admitted".
///
/// It is not a cryptographic commitment and this doc does not claim one. It is
/// unguessable, which is the property the check needs, and it is redacted from
/// `Debug` because a ticket is a bearer token and a logged ticket is a leaked
/// one.
#[derive(Clone, Copy, PartialEq, Eq, Hash)]
pub struct Seal([u8; 16]);

impl Seal {
	/// The seal a test presents when it has the numbers and no identity.
	pub const Empty:Seal = Seal([0; 16]);

	/// Mint a seal no other claim, in this process or after a restart, will
	/// produce.
	///
	/// Two halves from a hasher the standard library keys from the operating
	/// system's entropy, mixed with a per-process counter so two seals minted in
	/// the same nanosecond still differ. A restart produces new keys, so a seal
	/// from before a restart is not accepted by the authority after it - which is
	/// the honest answer to "can an identifier be reused across a restart": it
	/// cannot, because the authority that would recognise it did not survive.
	pub(crate) fn Mint() -> Seal {
		use std::collections::hash_map::RandomState;
		static Counter:AtomicU64 = AtomicU64::new(0);
		let State = RandomState::new();
		let Draw = Counter.fetch_add(1, Ordering::Relaxed);
		let mut Bytes = [0u8; 16];
		for (Index, Half) in Bytes.chunks_mut(8).enumerate() {
			let mut Hasher = State.build_hasher();
			Hasher.write_u64(Draw);
			Hasher.write_u64(Index as u64);
			Half.copy_from_slice(&Hasher.finish().to_le_bytes());
		}
		Seal(Bytes)
	}

	/// Read a seal back from the text an adapter stored.
	///
	/// This is the adapter's door, not the caller's: it rebuilds a seal that was
	/// issued, and confers nothing, because the authority accepts only the seals
	/// it recorded.
	pub fn Parse(Text:&str) -> Option<Seal> {
		if Text.len() != 32 {
			return None;
		}
		let mut Bytes = [0u8; 16];
		for (Index, Byte) in Bytes.iter_mut().enumerate() {
			*Byte = u8::from_str_radix(Text.get(Index * 2..Index * 2 + 2)?, 16).ok()?;
		}
		Some(Seal(Bytes))
	}

	/// The text an adapter stores. Never a log line.
	pub fn Hex(&self) -> String {
		use std::fmt::Write;
		let mut Text = String::with_capacity(32);
		for Byte in self.0 {
			let _ = write!(Text, "{Byte:02x}");
		}
		Text
	}
}

impl fmt::Debug for Seal {
	fn fmt(&self, Formatter:&mut fmt::Formatter<'_>) -> fmt::Result { Formatter.write_str("Seal(<redacted>)") }
}

/// Where an effect is recorded: the one place that decides whether it may be.
///
/// One verb, because a destination that offered a second way to write would be a
/// destination with a second rule. It is a future rather than a call because
/// reaching the authority is not free: the production implementation is a
/// reducer call over a connection, and a trait that forced it to block would
/// have decided which transport the database gets.
///
/// The production implementation calls the reducer. `StandIn` implements it over
/// a local table so the port can be exercised without a database, and it is a
/// test double rather than the authority: a build with no database has no
/// decision to make and says so by having no implementation wired. The reads a
/// stand-in needs - which effects landed, which are ambiguous, which resources
/// are quarantined - are that stand-in's own surface, because a database exposes
/// them as rows and a subscription, not as a method on the port.
pub trait Destination: Send + Sync {
	/// Record one effect at the place it lands, or refuse it.
	///
	/// A refusal is a value. A panic would turn a lost lease into a lost
	/// process, and a lost lease is the expected outcome of a slow worker.
	///
	/// A refusal is also not an undo: an effect that ran before it presented may
	/// have changed the world, so the destination records it as [`Ambiguous`]
	/// and quarantines the resource rather than reporting nothing happened.
	fn Present<'a>(&'a self, Ticket:Ticket, Name:&'a str) -> FutureBox<'a, Result<Recorded, Error>>;
}

/// The port this crate offers.
///
/// `WithLease` is generic over the write's return type, which is why this trait
/// is not object-safe: a boxed `dyn Runtime` would have to erase `T`, and
/// erasing it is what a caller writing `withLease::<Row>` is asking not to
/// happen. The composition root holds the concrete engine instead.
pub trait Runtime: Send + Sync {
	/// Admit a run, or refuse it with the reason.
	fn Admit<'a>(&'a self, Request:Request) -> FutureBox<'a, Admission>;
	/// Claim a resource for one run, with that run's budget, or fail because
	/// one is held. Omitting the budget means the run is unbounded.
	fn Claim<'a>(
		&'a self,
		Resource:&'a str,
		Ttl:Duration,
		Budget:Option<Budget>,
	) -> FutureBox<'a, Result<Lease, Error>>;
	/// Extend a lease, only for the holder of the current fence, and only
	/// forwards.
	fn Renew<'a>(&'a self, Lease:Lease, Ttl:Duration) -> FutureBox<'a, Result<Lease, Error>>;
	/// Admit one effect under a lease, and hand it the authority it was
	/// admitted under.
	///
	/// The check here is an optimisation: work whose authority has already
	/// ended is refused before it starts, and the run is charged for the
	/// attempt. It is not the guarantee. A lease can expire while the work is
	/// running, and this call cannot revoke the effect at that point because
	/// this call is not where the effect lands. The work receives a [`Ticket`]
	/// and must present it to the [`Destination`] - the reducer, or the local
	/// stand-in for one - which refuses a stale fence, a stale generation, a
	/// released lease or an expired record.
	///
	/// An effect whose record is refused after it ran is not retracted: the
	/// destination records it as [`Ambiguous`] and quarantines the resource.
	fn WithLease<'a, T, Write>(&'a self, Lease:Lease, Write:Write) -> FutureBox<'a, Result<T, Error>>
	where
		T: Send + 'a,
		Write: FnOnce(Ticket) -> FutureBox<'a, T> + Send + 'a;
	/// Release, always. Never fails.
	fn Release<'a>(&'a self, Lease:Lease) -> FutureBox<'a, ()>;
	/// The runtime's clock, so a deadline can be expressed on it.
	fn Now(&self) -> Instant;
	/// The manifest this runtime enforces, and the generation it is bound to.
	fn Reflect(&self) -> Binding;
	/// The clock itself, for a caller that must express a deadline on it.
	fn Clock(&self) -> &dyn Clock;
}

//! The runner: admit a run, run an effect, present its ticket, clean up without
//! throwing.
//!
//! Everything here is delegation. The runner holds one generation, one
//! registration list, one clock and one [`Authority`], and every verb it offers
//! is the authority's verb with the manifest check in front of it. It keeps no
//! lease, no fence and no record, so it cannot become the second authority the
//! review warned about: there is nothing in this file to disagree with.
//!
//! Four mechanisms carry the rules the port states, and only the first is the
//! runner's:
//!
//! 1. **The manifest it reports is the manifest it enforces.** `Admit` answers
//!    from the binding it was constructed with, and nothing else, which is what
//!    makes that sentence checkable rather than asserted.
//! 2. **A record per resource, and a fence counter that outlives it.** The
//!    authority's. The counter is what was ever issued, so deleting a record on
//!    release cannot make an old number current again.
//! 3. **An effect carries the authority it was admitted under.** `WithLease` asks
//!    the authority to admit the effect, and hands the work the [`Ticket`] it
//!    minted. The admission is an optimisation: it refuses doomed work early and
//!    charges the budget. The guarantee is at the destination
//!    ([`crate::Struct::Port::Destination`]), which re-reads the record when the
//!    effect lands and refuses a ticket whose fence, generation, holder or expiry
//!    is no longer current. A check in this file cannot see the instant an effect
//!    writes; a check where it lands can.
//! 4. **Unsettled local futures are counted, and a takeover waits for them.**
//!    Also the authority's, and this file cannot read the count. That is the
//!    point: the count is accounting, not fencing, and a runner that could read
//!    it would be a runner that could decide.

/// Bind a runner to one generation, one registration list and one authority.
///
/// A duplicate registration is refused here rather than resolved, because two
/// declarations of one name is a composition root disagreeing with itself and
/// picking one silently is how a pre-warm harness prepares the wrong thing.
///
/// The authority is an argument and not a default. A composition root that has a
/// database passes the adapter that calls it; one that does not passes
/// [`crate::Struct::StandIn::StandIn`] and is, by the name it had to type, running
/// without the authority.
pub fn Create<A: Authority>(Options:Options, Authority:A) -> Result<Runner<A>, String> {
	let mut Capabilities = BTreeMap::new();
	for Registration in Options.Capabilities {
		if Capabilities.contains_key(&Registration.Name) {
			return Err(format!("{} is registered twice", Registration.Name));
		}
		Capabilities.insert(Registration.Name.clone(), Registration);
	}
	Ok(Runner {
		Generation:Options.Generation,
		Capabilities,
		Clock:Options.Clock,
		Authority,
	})
}

impl<A: Authority> Runner<A> {
	/// The authority this runner delegates to.
	pub fn Authority(&self) -> &A { &self.Authority }
}

impl<A: Authority + Clone> Runner<A> {
	/// A handle to the destination an effect presents its ticket to.
	///
	/// Cloning a handle is not cloning an authority: both handles name the same
	/// table, which is why a destination handed to another thread still refuses
	/// what this one would. Hand this to whatever performs an effect. An effect
	/// that never presents its ticket leaves no record, and an unrecorded effect
	/// is invisible to the state: the runtime cannot make the presentation happen,
	/// but it can make the absence observable.
	pub fn Destination(&self) -> A { self.Authority.clone() }
}

impl<A: Authority> Runtime for Runner<A> {
	fn Admit<'a>(&'a self, Request:Request) -> FutureBox<'a, Admission> {
		Box::pin(async move {
			if Request.Generation != self.Generation {
				return Admission::Refused(format!(
					"admission-refused: generation {} is not bound; this runtime enforces generation {}",
					Request.Generation, self.Generation
				));
			}
			let Some(Registered) = self.Capabilities.get(&Request.Capability) else {
				return Admission::Refused(format!(
					"admission-refused: {} is not registered with this runtime",
					Request.Capability
				));
			};
			if Registered.State != Request.State {
				return Admission::Refused(format!(
					"admission-refused: {} declares {}, not {}",
					Registered.Name,
					Registered.State.Name(),
					Request.State.Name()
				));
			}
			Admission::Admitted
		})
	}

	fn Claim<'a>(
		&'a self,
		Resource:&'a str,
		Ttl:Duration,
		Budget:Option<Budget>,
	) -> FutureBox<'a, Result<Lease, Error>> {
		self.Authority.Claim(Resource, Ttl, Budget)
	}

	fn Renew<'a>(&'a self, Lease:Lease, Ttl:Duration) -> FutureBox<'a, Result<Lease, Error>> {
		self.Authority.Renew(Lease, Ttl)
	}

	fn WithLease<'a, T, Write>(&'a self, Lease:Lease, Write:Write) -> FutureBox<'a, Result<T, Error>>
	where
		T: Send + 'a,
		Write: FnOnce(Ticket) -> FutureBox<'a, T> + Send + 'a, {
		Box::pin(async move {
			// One delegation, then the work, then the settlement. The runner
			// never inspects the admission: the ticket is the effect's, and the
			// accounting guard is the authority's.
			let Admitted:Admitted = self.Authority.Admit(Lease).await?;
			let Ticket = Admitted.Ticket.clone();
			let Value = Write(Ticket).await;
			drop(Admitted);
			Ok(Value)
		})
	}

	fn Release<'a>(&'a self, Lease:Lease) -> FutureBox<'a, ()> { self.Authority.Release(Lease) }

	fn Now(&self) -> Instant { self.Clock.Now() }

	fn Reflect(&self) -> Binding {
		Binding {
			Generation:self.Generation,
			Capabilities:self.Capabilities.values().cloned().collect(),
		}
	}

	fn Clock(&self) -> &dyn Clock { self.Clock.as_ref() }
}

//! The local stand-in for the reducer: the same checks, over a local table.
//!
//! This module is a **test double**. It exists so the port can be exercised with
//! no database, no socket and no clock it did not choose; it is not the authority
//! in production, and the shape of the crate says so: the runner holds an
//! `Authority` and knows nothing else, so a composition root that wants the real
//! authority substitutes a reducer adapter and this type is never constructed.
//!
//! What it is not: a wrapper around [`crate::Runtime::WithLease`]. `WithLease`
//! admits an effect when the work starts; this refuses a record when the work
//! ends. Between those two moments the lease can expire, be released, or be taken
//! over by a newer fence, and the admission that already happened cannot say
//! anything about any of it. Only the party the record is presented to can. That
//! is why the check lives here, and why moving it back into the caller is a
//! regression a test catches rather than a refactor.
//!
//! # The decision, and its mirror
//!
//! `Package/spacetime-catalog/Source/Reducers.ts` makes the same decision in
//! `record`, over the same columns, and `Test/Destination.test.ts` there drives
//! the same transcript to the same refusal. The correspondence is exact, column
//! for column:
//!
//! | reducer                                   | stand-in                                  |
//! | ----------------------------------------- | ----------------------------------------- |
//! | `Latest(ctx, generation) !== null`        | `Ticket.Generation == self.Generation`     |
//! | `ctx.find("lease", resource) !== null`    | `Inner.Runs.get(resource)`                 |
//! | `lease["fence"] === args.fence`           | `Run.Fence == Ticket.Fence`                |
//! | `lease["holder"] === ctx.sender`          | `Run.Seal == Ticket.Seal`                  |
//! | `lease["state"] !== "released"`           | `!Run.Released`                            |
//! | `lease["state"] !== "quarantined"`        | `!Run.Quarantined`                         |
//! | `IsLive(lease, ctx.now)`                  | `Now < Run.ExpiresAt`                      |
//! | refusal after identity → ambiguity + quarantine | the same, in `Quarantine`            |
//!
//! # The handover rule
//!
//! An effect that was admitted under an epoch and is refused when it lands is
//! recorded as [`Ambiguous`] against the resource, and the resource is
//! quarantined: the write is not counted as a success, and no new holder is
//! assigned the resource until a resolver states whether the external action
//! landed or was absent. Rejecting an effect's result does not undo its effect,
//! so the refusal is never allowed to be the only trace of it.
//!
//! # The honest limit, and where the rule puts the rest
//!
//! A refusal is *ambiguous* only when the ticket's identity matches the current
//! epoch - the authority can then say "this effect was admitted and its authority
//! has ended". A ticket whose fence has already been superseded names an epoch
//! the row no longer describes, so it is refused without an ambiguity entry. That
//! is not a gap in the rule, it is where the rule puts the decision: the claim
//! that superseded the epoch is the accounting point for it, and the reconciler
//! that granted the new fence is the party that knew the old one had ended. This
//! stand-in refuses such a takeover while a local future is unsettled, which is
//! the same accounting done eagerly instead of on inspection.
//!
//! A local check also cannot un-send bytes that already left the process. What
//! the destination guarantees is that a stale effect is never *recorded* as a
//! success, and that one which may have landed is recorded as unknown rather than
//! as nothing.
//!
//! # What a restart does, and why that is the argument for the database
//!
//! The table lives in this process, so a restart discards it: the fence counter
//! starts again, and a seal minted before the restart is not recognised after it.
//! A fence that restarts at one is a fence a late worker can satisfy, and that is
//! exactly why this type is not the authority. In the database the fence is a row
//! that no process owns, so it survives every restart, and the seal is a column of
//! that row rather than a value in memory. The stand-in's restart behaviour is not
//! a defect to fix here - it is the measurement that says the table belongs to the
//! reducer.

/// Distinguishes one stand-in's leases from another's in the same process.
///
/// Lease ids are built from this token, so a lease issued by one stand-in reads
/// differently from one issued by a second that happens to have counted the same
/// fence for the same resource name. The id is for reading, not for deciding: the
/// decision is the seal, because an id is a string a caller can construct and a
/// seal is not.
static Instances:AtomicU64 = AtomicU64::new(0);

/// Accounts for one admitted local future, however it ends.
///
/// The row survives a release, so settling only decrements: the epoch stays
/// described, which is what lets a late effect from a released epoch be
/// attributed to the holder that had it. `Drop` is what makes the count true for
/// an abandoned future as well as a completed one - there is no `finally` in
/// Rust, and a cancellation that leaked the count would hold a resource until the
/// process ended.
struct Settling {
	Inner:Arc<Mutex<Inner>>,
	Resource:String,
	Fence:u64,
}

impl Drop for Settling {
	fn drop(&mut self) {
		let mut Inner = State::Lock(&self.Inner);
		if let Some(Run) = Inner.Runs.get_mut(&self.Resource)
			&& Run.Fence == self.Fence
		{
			Run.Unsettled = Run.Unsettled.saturating_sub(1);
		}
	}
}

/// The local stand-in: one generation, one clock, one table.
///
/// Cloning one is cloning a handle to the same table, so a destination handed to
/// another thread is still the table the lease verbs are fencing against.
#[derive(Clone)]
pub struct StandIn {
	Generation:u64,
	Clock:Arc<dyn Clock>,
	Token:String,
	Inner:Arc<Mutex<Inner>>,
}

impl StandIn {
	/// A stand-in recording one generation, on the clock it is given.
	pub fn New(Generation:u64, Clock:Arc<dyn Clock>) -> StandIn {
		let Instance = Instances.fetch_add(1, Ordering::SeqCst) + 1;
		StandIn {
			Generation,
			Clock,
			Token:format!("r{Instance}"),
			Inner:Arc::new(Mutex::new(Inner::default())),
		}
	}

	/// Every effect this destination accepted, in the order it accepted them.
	///
	/// A refused effect is absent: it was never recorded, so as far as anything
	/// reading the state is concerned it did not happen. What the world saw is in
	/// [`StandIn::Ambiguities`].
	pub fn Records(&self) -> Vec<Recorded> { State::Lock(&self.Inner).Records.clone() }

	/// Every effect that was admitted and refused, and whose external action is
	/// therefore unknown.
	pub fn Ambiguities(&self) -> Vec<Ambiguous> { State::Lock(&self.Inner).Ambiguities.clone() }

	/// Every resource that is not to be reassigned until its ambiguity is
	/// resolved, sorted so a caller need not know the table's order.
	pub fn Quarantined(&self) -> Vec<String> {
		let Inner = State::Lock(&self.Inner);
		let mut Names:Vec<String> = Inner
			.Runs
			.iter()
			.filter(|(_, Run)| Run.Quarantined)
			.map(|(Name, _)| Name.clone())
			.collect();
		Names.sort();
		Names
	}

	/// State what happened to the effects that made a resource ambiguous, and lift
	/// the quarantine.
	///
	/// The epoch is marked released rather than deleted, so the fence survives and
	/// the next claim advances past it. The ambiguity entries keep the answer, and
	/// stay: the record of an effect that landed under an authority that had ended
	/// is the evidence a later reader needs, and a resolution that erased it would
	/// be the retraction the rule forbids.
	pub fn Resolve(&self, Resource:&str, Resolution:Resolution) -> Result<(), Error> {
		let mut Inner = State::Lock(&self.Inner);
		let Some(Run) = Inner.Runs.get_mut(Resource) else {
			return Err(Error::Refused(format!("no lease is live for {Resource}")));
		};
		if !Run.Quarantined {
			return Err(Error::Refused(format!("{Resource} is not quarantined")));
		}
		Run.Quarantined = false;
		Run.Released = true;
		for Ambiguity in Inner.Ambiguities.iter_mut().filter(|Ambiguity| Ambiguity.Resource == Resource) {
			Ambiguity.Resolution = Some(Resolution);
		}
		Ok(())
	}

	/// The caller's check, evaluated where an effect starts.
	///
	/// Five checks, ordered so the reason is the true one:
	///
	/// 1. **No record** - nothing is held under that name any more.
	/// 2. **Fence mismatch** - the caller's authority was superseded.
	///    `lease-held`: somebody else holds it, and the caller's copy is history.
	///    Checked before the seal, because a superseded caller would fail both and
	///    "you were superseded" is the fact it needs.
	/// 3. **Seal mismatch** - the fence is current but the caller is not the
	///    holder that was admitted. This is the check a guessable identifier
	///    cannot pass.
	/// 4. **Released** - the holder gave the authority back.
	/// 5. **Expired** - the fence matches, the caller is the holder, and authority
	///    ended anyway.
	///
	/// This is the caller's check, and it answers a question about *now*: may this
	/// work start? It is deliberately not the destination's check, which answers a
	/// different question - may this effect be recorded? - at a later moment, and
	/// answers it with an ambiguity when the answer is no. Sharing one function
	/// between the two would reintroduce exactly the defect the destination exists
	/// to close, so [`Destination::Present`] has its own.
	fn Held<'a>(Inner:&'a mut Inner, Resource:&str, Holder:Seal, Fence:u64, Now:Instant) -> Result<&'a mut Run, Error> {
		let Some(Run) = Inner.Runs.get_mut(Resource) else {
			return Err(Error::Refused(format!("no lease is live for {Resource}")));
		};
		if Run.Fence != Fence {
			return Err(Error::Held(format!("{Resource} is held under fence {}, not fence {Fence}", Run.Fence)));
		}
		if Run.Seal != Holder {
			return Err(Error::Refused(format!(
				"the lease presented for {Resource} was not the one issued: the caller is not the holder that was admitted"
			)));
		}
		if Run.Released {
			return Err(Error::Refused(format!("the lease on {Resource} was released")));
		}
		if Now >= Run.ExpiresAt {
			return Err(Error::Refused(format!(
				"the lease on {Resource} expired at {:?}, and the clock reads {Now:?}",
				Run.ExpiresAt
			)));
		}
		Ok(Run)
	}

	/// Spend one operation of the run's allowance.
	///
	/// Charged before the effect, not after: an effect that fails was still
	/// attempted, and a count that only moved on success would let a run that fails
	/// in a loop spend nothing. The charge happens under the same lock as the fence
	/// check, so two effects cannot both read the last operation.
	fn Charge(Run:&mut Run, Now:Instant) -> Result<(), Error> {
		let Some(Budget) = Run.Budget.as_mut() else {
			return Ok(());
		};
		if let Some(Deadline) = Budget.Deadline
			&& Now >= Deadline
		{
			return Err(Error::Spent(format!("the run's deadline passed at {Deadline:?}, and the clock reads {Now:?}")));
		}
		if Budget.Operations == 0 {
			return Err(Error::Spent("the run has spent every operation it was granted"));
		}
		Budget.Operations -= 1;
		Ok(())
	}

	/// Record an effect as ambiguous and quarantine its resource.
	///
	/// Called only after the ticket's identity matched the current epoch, because
	/// that match is what makes the refusal mean "this effect was admitted": a
	/// guessed ticket is refused without an ambiguity, since nothing was ever
	/// authorised for it to have landed.
	fn Quarantine(Inner:&mut Inner, Ticket:&Ticket, Name:&str, Now:Instant, Reason:String) -> Error {
		Inner.Ambiguities.push(Ambiguous {
			Name:Name.to_string(),
			Resource:Ticket.Resource().to_string(),
			Generation:Ticket.Generation(),
			Fence:Ticket.Fence(),
			LeaseId:Ticket.LeaseId().to_string(),
			At:Now,
			Reason:Reason.clone(),
			Resolution:None,
		});
		if let Some(Run) = Inner.Runs.get_mut(Ticket.Resource()) {
			Run.Quarantined = true;
		}
		Error::Refused(format!(
			"{Reason}. The effect may have landed, so it is recorded as ambiguous and {} is quarantined until a resolver states whether it did.",
			Ticket.Resource()
		))
	}
}

impl Authority for StandIn {
	fn Generation(&self) -> u64 { self.Generation }

	fn Claim<'a>(
		&'a self,
		Resource:&'a str,
		Ttl:Duration,
		Budget:Option<Budget>,
	) -> FutureBox<'a, Result<Lease, Error>> {
		Box::pin(async move {
			if Resource.is_empty() {
				return Err(Error::Refused("a claim names the resource it wants"));
			}
			if Ttl.is_zero() {
				return Err(Error::Refused(format!(
					"a lease ttl must be a positive, finite number of milliseconds; got {}",
					Ttl.as_millis()
				)));
			}
			let Now = self.Clock.Now();
			let ExpiresAt = Now.checked_add(Ttl).ok_or_else(|| {
				Error::Refused(format!("a lease ttl of {}ms has no instant to expire at", Ttl.as_millis()))
			})?;
			let mut Inner = State::Lock(&self.Inner);
			if let Some(Current) = Inner.Runs.get(Resource) {
				if Current.Unsettled > 0 {
					return Err(Error::Held(format!(
						"{Resource} has an unsettled local future under fence {}; a takeover waits for it to settle",
						Current.Fence
					)));
				}
				if Current.Quarantined {
					return Err(Error::Held(format!(
						"{Resource} is quarantined: an effect admitted under fence {} was refused when it landed, and no holder is assigned until a resolver states whether it landed",
						Current.Fence
					)));
				}
				if !Current.Released && Now < Current.ExpiresAt {
					return Err(Error::Held(format!(
						"{Resource} is held under fence {} until {:?}",
						Current.Fence, Current.ExpiresAt
					)));
				}
			}
			let Fence = Inner.Fences.get(Resource).copied().unwrap_or(0) + 1;
			Inner.Fences.insert(Resource.to_string(), Fence);
			let Lease = Lease {
				Id:format!("{}#{Resource}#{Fence}", self.Token),
				Resource:Resource.to_string(),
				Fence,
				ExpiresAt,
				Seal:Seal::Mint(),
			};
			Inner.Runs.insert(
				Resource.to_string(),
				Run {
					Seal:Lease.Seal,
					Generation:self.Generation,
					Fence,
					ExpiresAt,
					Budget,
					Unsettled:0,
					Released:false,
					Quarantined:false,
				},
			);
			Ok(Lease)
		})
	}

	fn Renew<'a>(&'a self, Lease:Lease, Ttl:Duration) -> FutureBox<'a, Result<Lease, Error>> {
		Box::pin(async move {
			if Ttl.is_zero() {
				return Err(Error::Refused(format!(
					"a renewal ttl must be a positive, finite number of milliseconds; got {}",
					Ttl.as_millis()
				)));
			}
			let Now = self.Clock.Now();
			let mut Inner = State::Lock(&self.Inner);
			let Run = Self::Held(&mut Inner, &Lease.Resource, Lease.Seal, Lease.Fence, Now)?;
			let ExpiresAt = Now.checked_add(Ttl).ok_or_else(|| {
				Error::Refused(format!("a renewal ttl of {}ms has no instant to expire at", Ttl.as_millis()))
			})?;
			if ExpiresAt <= Run.ExpiresAt {
				return Err(Error::Refused(format!(
					"a renewal must extend the lease: {ExpiresAt:?} is not later than {:?}. Release it instead.",
					Run.ExpiresAt
				)));
			}
			Run.ExpiresAt = ExpiresAt;
			Ok(Lease { ExpiresAt, ..Lease })
		})
	}

	fn Release<'a>(&'a self, Lease:Lease) -> FutureBox<'a, ()> {
		Box::pin(async move {
			let mut Inner = State::Lock(&self.Inner);
			// A superseded lease is not this caller's to end: the resource now
			// belongs to the fence that took it over, and releasing here would
			// hand a stranger's authority back. The row survives either way,
			// because it is the only thing that still knows who held the epoch.
			if let Some(Run) = Inner.Runs.get_mut(&Lease.Resource)
				&& Run.Fence == Lease.Fence
				&& Run.Seal == Lease.Seal
			{
				Run.Released = true;
			}
		})
	}

	fn Admit<'a>(&'a self, Lease:Lease) -> FutureBox<'a, Result<Admitted, Error>> {
		Box::pin(async move {
			// The admission step: check, charge, count, and mint the authority
			// the effect will carry. All four happen under one lock, so the
			// fence in the ticket is the fence the charge was taken under.
			let Now = self.Clock.Now();
			let mut Inner = State::Lock(&self.Inner);
			let Ticket = {
				let Run = Self::Held(&mut Inner, &Lease.Resource, Lease.Seal, Lease.Fence, Now)?;
				Self::Charge(Run, Now)?;
				Run.Unsettled += 1;
				// The record's expiry, not the copy the caller presented: a lease
				// renewed after it was handed out is still the same grant, and the
				// caller's copy is the older number.
				Ticket::Mint(self.Generation, &Lease, Run.ExpiresAt)
			};
			let Settle:Box<dyn Send> = Box::new(Settling {
				Inner:self.Inner.clone(),
				Resource:Lease.Resource.clone(),
				Fence:Lease.Fence,
			});
			Ok(Admitted { Ticket, Settle })
		})
	}
}

impl Destination for StandIn {
	fn Present<'a>(&'a self, Ticket:Ticket, Name:&'a str) -> FutureBox<'a, Result<Recorded, Error>> {
		Box::pin(async move {
			let Now = self.Clock.Now();
			let mut Inner = State::Lock(&self.Inner);
			// The generation first: a ticket minted under another generation
			// carries numbers this authority never counted, so nothing about it
			// can be attributed here.
			if Ticket.Generation() != self.Generation {
				return Err(Error::Refused(format!(
					"the effect was admitted under generation {}, and this destination records generation {}",
					Ticket.Generation(),
					self.Generation
				)));
			}
			let Some((Granted, Seal, Fence, Released, Quarantined, ExpiresAt)) = Inner
				.Runs
				.get(Ticket.Resource())
				.map(|Run| (Run.Generation, Run.Seal, Run.Fence, Run.Released, Run.Quarantined, Run.ExpiresAt))
			else {
				return Err(Error::Refused(format!("no lease is live for {}", Ticket.Resource())));
			};
			// The reducer's `lease["generation"] !== generation`: a lease never
			// adopts a later generation, so a ticket naming one it was not granted
			// under is not this epoch's ticket.
			if Granted != Ticket.Generation() {
				return Err(Error::Refused(format!(
					"the lease on {} was granted under generation {Granted}, not {}",
					Ticket.Resource(),
					Ticket.Generation()
				)));
			}
			// The fence before the seal, for the same reason the caller's check
			// does: a superseded ticket fails both, and "the epoch you were in has
			// ended" is the fact it needs. A superseded epoch is where the
			// takeover, not this destination, does the accounting.
			if Fence != Ticket.Fence() {
				return Err(Error::Held(format!(
					"{} is held under fence {Fence}, not fence {}",
					Ticket.Resource(),
					Ticket.Fence()
				)));
			}
			// The identity: the numbers matching is not the caller matching.
			if Seal != Ticket.Seal() {
				return Err(Error::Refused(format!(
					"the ticket presented for fence {Fence} of {} was not the one issued: the numbers match and the identity does not",
					Ticket.Resource()
				)));
			}
			// From here the ticket names the current epoch, so it is one this
			// authority issued and the effect it carries was authorised. Any
			// refusal now means an authorised effect may have landed, and that is
			// what makes it ambiguous rather than merely refused.
			if Quarantined {
				return Err(Self::Quarantine(
					&mut Inner,
					&Ticket,
					Name,
					Now,
					format!("{} is already quarantined by an earlier effect", Ticket.Resource()),
				));
			}
			if Released {
				return Err(Self::Quarantine(
					&mut Inner,
					&Ticket,
					Name,
					Now,
					format!("the lease on {} was released while the effect was in flight", Ticket.Resource()),
				));
			}
			if Now >= ExpiresAt {
				return Err(Self::Quarantine(
					&mut Inner,
					&Ticket,
					Name,
					Now,
					format!(
						"the effect outlived its authority: the lease on {} ended at {ExpiresAt:?}, the ticket was minted for {:?}, and the clock reads {Now:?}",
						Ticket.Resource(),
						Ticket.ExpiresAt()
					),
				));
			}
			let Receipt = Recorded {
				Name:Name.to_string(),
				Resource:Ticket.Resource().to_string(),
				Fence,
				Generation:self.Generation,
				At:Now,
			};
			Inner.Records.push(Receipt.clone());
			Ok(Receipt)
		})
	}
}

//! The one table the local stand-in keeps.
//!
//! Two verbs mutate it: the authority issues, renews and releases leases, and
//! the destination accepts or refuses a record. Both take the same lock over the
//! same maps, which is what makes "the manifest it reports is the manifest it
//! enforces" a property of the bytes rather than a convention - there is no
//! second copy of a fence to drift, and no window in which the lease engine and
//! the recorder disagree about which fence is current.
//!
//! It is crate-private on purpose. The table is an implementation of one
//! authority, not a shape a consumer may reach into: a consumer holds a
//! [`crate::Struct::StandIn::StandIn`] and presents tickets to it, which is the
//! only way to change what is recorded. A reducer adapter keeps no such table
//! here at all - its table is the database's, and that is the whole difference
//! between the stand-in and the authority.

use std::{
	collections::HashMap,
	sync::{Mutex, MutexGuard},
	time::Instant,
};

use crate::Struct::{
	Port::{Ambiguous, Budget, Recorded, Seal},
};

/// What the authority knows about one live grant.
pub(crate) struct Run {
	/// The holder's identity. This is the reducer's `lease.holder` column: the
	/// only caller a fenced write is accepted from, rendered as something a
	/// caller cannot guess instead of a name a caller could.
	pub(crate) Seal:Seal,
	/// The generation the lease was granted under. A lease never adopts a later
	/// one.
	pub(crate) Generation:u64,
	/// The fence the resource is at while this record is the live one.
	pub(crate) Fence:u64,
	/// The record's own expiry. A presented lease's copy is never trusted.
	pub(crate) ExpiresAt:Instant,
	/// This run's allowance, owned by the authority once the claim is admitted.
	pub(crate) Budget:Option<Budget>,
	/// Local futures admitted under this record that have not settled yet.
	///
	/// This is an accounting count, not a fence. It says a future this process
	/// admitted is still running. It does not say the worker behind that future
	/// still has authority, and it says nothing at all about work that has
	/// already left the process: dropping a future decrements this, and
	/// dropping a future does not stop a browser-side operation.
	pub(crate) Unsettled:u64,
	/// Set when the holder released. The row survives, because the row is the
	/// only place that still knows who held this epoch - and a late effect from
	/// a released epoch has to be attributable to somebody.
	pub(crate) Released:bool,
	/// Set when an effect from this epoch was refused after it ran. A
	/// quarantined resource is not reassigned until a resolver says whether the
	/// effect landed.
	pub(crate) Quarantined:bool,
}

/// The table every verb mutates, behind one lock.
#[derive(Default)]
pub(crate) struct Inner {
	/// What is live now, by resource name.
	pub(crate) Runs:HashMap<String, Run>,
	/// The highest fence ever issued, by resource name. Never reset, so a
	/// released row cannot make an old number current again.
	pub(crate) Fences:HashMap<String, u64>,
	/// Every effect the destination accepted, in the order it accepted them.
	///
	/// A refused effect leaves nothing here: the record is the effect's
	/// existence in the state, so an effect the destination refused did not
	/// happen as far as anything reading this table is concerned. What a
	/// refusal leaves is in `Ambiguities`, because the world is not the state
	/// and a write that landed is not retracted by a record that did not.
	pub(crate) Records:Vec<Recorded>,
	/// Every effect that was admitted and refused, in the order it was refused.
	pub(crate) Ambiguities:Vec<Ambiguous>,
}

/// A lock whose poisoning is not a failure: a panicked caller must not make the
/// whole runtime unusable.
pub(crate) fn Lock(Guard:&Mutex<Inner>) -> MutexGuard<'_, Inner> {
	Guard.lock().unwrap_or_else(|Poison| Poison.into_inner())
}

//! `agentic.runtime`, exercised against a clock the test owns.
//!
//! No tokio, no browser, no network and no sleeping: a lease expires because the
//! test advances a clock, which is the whole reason the clock is a parameter of
//! the port rather than a call to the system's.
//!
//! Three requirements meet in this file and they are not the same requirement:
//!
//! - **Authority** is decided where an effect lands. `Present` re-reads the table
//!   at the instant the effect arrives and refuses a stale fence, a stale
//!   generation, a released lease or an expired record. The test that proves it is
//!   `an_effect_that_outlives_its_lease_is_refused_where_it_lands_and_recorded_ambiguous`,
//!   which is the counterexample from `Documentation/Review.md` executed with the write
//!   before the record rather than after it.
//! - **Identity** is not a number. `a_ticket_whose_numbers_match_but_whose_seal_was_never_issued_is_refused_and_records_no_ambiguity`
//!   presents a ticket with the right generation, resource, lease id, fence and
//!   expiry, and is refused, because the destination compares the holder it
//!   recorded and not the numbers it can be told.
//! - **Accounting** is decided by the unsettled-future count, which says a local
//!   future has not settled. `a_takeover_waits_for_an_unsettled_local_future`
//!   encodes that and only that. It does not say authority survives a settled
//!   future, and it is not the guarantee.
//!
//! The three run against `StandIn`, which is the local stand-in for the reducer.
//! That is the point of the split: the runner under test holds no lease state, so
//! every assertion about a fence is an assertion about the double, and the double
//! is pinned to the reducer by the mirror test in `Package/spacetime-catalog`.

/// A clock that only moves when the test moves it.
#[derive(Debug)]
struct TestClock {
	Now:Mutex<Instant>,
}

impl TestClock {
	fn New() -> Arc<Self> { Arc::new(Self { Now:Mutex::new(Instant::now()) }) }

	fn Advance(&self, Span:Duration) {
		let mut Now = self.Now.lock().unwrap();
		*Now += Span;
	}
}

impl Clock for TestClock {
	fn Now(&self) -> Instant { *self.Now.lock().unwrap() }
}

/// A runner bound to generation 7, over a stand-in on the test's clock.
///
/// The generation is passed twice on purpose: once to the options the runner
/// enforces and once to the stand-in that records. They are two objects, and a
/// test that could not tell them apart could not tell a second authority from
/// the one it is supposed to be.
fn Engine(Clock:Arc<TestClock>) -> Runner<StandIn> {
	let Options = Options::New(7)
		.With("chat", StateClass::HoldsTransport)
		.With("verify", StateClass::Stateless)
		.On(Clock.clone());
	let Authority = StandIn::New(Options.Generation, Clock);
	Create(Options, Authority).expect("the runtime binds")
}

fn Claim(System:&Runner<StandIn>, Resource:&str, Ttl:Duration) -> Result<Lease, Error> {
	BlockOn(System.Claim(Resource, Ttl, None))
}

fn Claimed(System:&Runner<StandIn>, Resource:&str, Ttl:Duration) -> Lease {
	Claim(System, Resource, Ttl).expect("the claim is admitted")
}

/// Present a ticket to the destination, which is the only place it can be
/// refused for being late.
fn Present(Place:&StandIn, Ticket:Ticket, Name:&str) -> Result<Recorded, Error> {
	BlockOn(Place.Present(Ticket, Name))
}

fn Span(Milliseconds:u64) -> Duration { Duration::from_millis(Milliseconds) }

/// Poll a future once, without an executor, so a test can observe the state
/// between "admitted" and "settled" instead of only after it finishes.
fn PollOnce<Work:std::future::Future + ?Sized>(Work:std::pin::Pin<&mut Work>) -> bool {
	let Waker = std::task::Waker::noop();
	let mut Context = std::task::Context::from_waker(Waker);
	matches!(Work.poll(&mut Context), std::task::Poll::Ready(_))
}

/// A latch an effect waits on, so a test can hold it between "admitted" and
/// "the write" without an executor.
///
/// `BlockOn` re-polls in a loop, so a latch opened without waking anyone is
/// still observed on the next poll.
#[derive(Clone)]
struct Gate(Arc<AtomicBool>);

impl Gate {
	fn New() -> Self { Self(Arc::new(AtomicBool::new(false))) }

	fn Open(&self) { self.0.store(true, Ordering::SeqCst); }

	fn Wait(&self) -> Waiting { Waiting(self.0.clone()) }
}

/// The future half of a [`Gate`].
struct Waiting(Arc<AtomicBool>);

impl std::future::Future for Waiting {
	type Output = ();

	fn poll(
		self:std::pin::Pin<&mut Self>,
		_Context:&mut std::task::Context<'_>,
	) -> std::task::Poll<()> {
		if self.0.load(Ordering::SeqCst) {
			std::task::Poll::Ready(())
		} else {
			std::task::Poll::Pending
		}
	}
}

#[test]
fn an_effect_that_outlives_its_lease_is_refused_where_it_lands_and_recorded_ambiguous() {
	// `Documentation/Review.md`, executed step for step, with the write where the review
	// put it - before the record, not after it:
	//
	//   lease is live
	//   WithLease admits the callback
	//   callback waits
	//   lease expires
	//   callback performs the external write      <- the bytes are in the page now
	//   the ticket is presented                   <- only now can anything refuse
	//   WithLease returns success                 <- what must no longer happen
	//
	// The admission happened while the lease was live and cannot be un-made, and
	// the write cannot be un-sent. The only party that can notice is the one the
	// record is presented to, and the only honest answer it can give is "this
	// effect was authorised, it ran, and I will not count it" - an ambiguity, not
	// a silence.
	let Clock = TestClock::New();
	let Engine = Engine(Clock.clone());
	let Place = Engine.Destination();
	let Lease = Claimed(&Engine, "page", Span(100));
	let Latch = Gate::New();
	let Wrote = Arc::new(AtomicBool::new(false));
	let mut Effect = Engine.WithLease(Lease.clone(), |Ticket| {
		let Wait = Latch.Wait();
		let Wrote = Wrote.clone();
		let Place = Place.clone();
		Box::pin(async move {
			Wait.await;
			// The external action, first and unconditional. Nothing here consults
			// the lease, because a real effect does not: by the time it writes, the
			// only copy of its authority is the ticket it is about to present.
			Wrote.store(true, Ordering::SeqCst);
			Place.Present(Ticket, "write").await
		})
	});
	assert!(!PollOnce(Effect.as_mut()), "the effect is admitted and waiting");
	Clock.Advance(Span(101));
	Latch.Open();
	let Outcome = BlockOn(Effect);
	assert!(Wrote.load(Ordering::SeqCst), "the effect must have written before it presented");
	// The destination's verdict is checked first, because it is the one that
	// matters: whatever the caller reports afterwards, an effect that left a
	// record behind landed.
	assert!(Place.Records().is_empty(), "a refused effect left a record behind");
	let Error = Outcome
		.expect("the effect was admitted")
		.expect_err("the destination recorded a write it should have refused");
	assert_eq!(Error.Code(), Code::AdmissionRefused);
	assert!(Error.Message().contains("outlived"), "{}", Error.Message());
	// The state did not claim the effect, and the world is not the state: the
	// write is recorded as unknown rather than as nothing.
	let Ambiguous = Place.Ambiguities();
	assert_eq!(Ambiguous.len(), 1, "the external action was not recorded as ambiguous");
	assert_eq!(Ambiguous[0].Name, "write");
	assert_eq!(Ambiguous[0].Resource, "page");
	assert_eq!(Ambiguous[0].Fence, Lease.Fence);
	assert_eq!(Ambiguous[0].Generation, 7);
	assert_eq!(Ambiguous[0].Resolution, None, "the outcome is unknown, not guessed");
	// The resource is not handed to anybody else while the outcome is unknown.
	assert_eq!(Place.Quarantined(), vec!["page".to_string()]);
	let Takeover = Claim(&Engine, "page", Span(1_000)).unwrap_err();
	assert_eq!(Takeover.Code(), Code::LeaseHeld);
	assert!(Takeover.Message().contains("quarantined"), "{}", Takeover.Message());
	// A resolver states what happened, and only then is reassignment possible.
	Place.Resolve("page", Resolution::Landed).expect("the resource is quarantined");
	assert!(Place.Quarantined().is_empty());
	assert_eq!(Place.Ambiguities()[0].Resolution, Some(Resolution::Landed));
	assert!(Place.Records().is_empty(), "a resolution is not a record of authority");
	assert_eq!(Claimed(&Engine, "page", Span(1_000)).Fence, Lease.Fence + 1);
}

#[test]
fn an_effect_that_presents_before_it_writes_never_writes_and_is_still_unknown() {
	// The other ordering, and the reason the port says "present first if you can":
	// an effect that records before it acts cannot act once its authority ended,
	// because the record is refused before the external call is made.
	//
	// It is still recorded as ambiguous, and that is not a defect in the rule: the
	// destination cannot know whether the caller wrote before or after presenting,
	// so the honest record is "unknown". A resolver turns it into `Absent`, and
	// then nothing claims the effect at all.
	let Clock = TestClock::New();
	let Engine = Engine(Clock.clone());
	let Place = Engine.Destination();
	let Lease = Claimed(&Engine, "page", Span(100));
	let Latch = Gate::New();
	let Wrote = Arc::new(AtomicBool::new(false));
	let mut Effect = Engine.WithLease(Lease.clone(), |Ticket| {
		let Wait = Latch.Wait();
		let Wrote = Wrote.clone();
		let Place = Place.clone();
		Box::pin(async move {
			Wait.await;
			Place.Present(Ticket, "write").await?;
			Wrote.store(true, Ordering::SeqCst);
			Ok::<(), Error>(())
		})
	});
	assert!(!PollOnce(Effect.as_mut()), "the effect is admitted and waiting");
	Clock.Advance(Span(101));
	Latch.Open();
	let Outcome = BlockOn(Effect).expect("the effect was admitted");
	assert!(Outcome.is_err(), "the record was accepted after the lease ended");
	assert!(!Wrote.load(Ordering::SeqCst), "the write happened after the record was refused");
	assert!(Place.Records().is_empty());
	assert_eq!(Place.Ambiguities().len(), 1, "an effect that may have acted is unknown, not absent");
	Place.Resolve("page", Resolution::Absent).expect("the resource is quarantined");
	assert_eq!(Place.Ambiguities()[0].Resolution, Some(Resolution::Absent));
	assert!(Place.Records().is_empty(), "an absent effect is never recorded");
	assert_eq!(Claimed(&Engine, "page", Span(1_000)).Fence, Lease.Fence + 1);
}

#[test]
fn an_effect_whose_lease_was_released_while_it_ran_is_refused_and_recorded_ambiguous() {
	// Release is the other way authority ends, and it has the same consequence:
	// the effect was admitted, the holder gave the authority back, and the write
	// arrives afterwards. The row survives the release precisely so this case can
	// still be attributed to somebody.
	let Clock = TestClock::New();
	let Engine = Engine(Clock.clone());
	let Place = Engine.Destination();
	let Lease = Claimed(&Engine, "page", Span(1_000));
	let Kept:Arc<Mutex<Option<Ticket>>> = Arc::new(Mutex::new(None));
	BlockOn(Engine.WithLease(Lease.clone(), {
		let Kept = Kept.clone();
		move |Ticket| {
			Kept.lock().unwrap().replace(Ticket);
			Box::pin(async {})
		}
	}))
	.expect("the effect was admitted");
	BlockOn(Engine.Release(Lease.clone()));
	let Late = Kept.lock().unwrap().take().expect("the effect kept its ticket");
	let Error = Present(&Place, Late, "late").expect_err("the destination refused it");
	assert_eq!(Error.Code(), Code::AdmissionRefused);
	assert!(Error.Message().contains("released"), "{}", Error.Message());
	assert!(Place.Records().is_empty());
	assert_eq!(Place.Ambiguities().len(), 1);
	assert_eq!(Place.Quarantined(), vec!["page".to_string()]);
}

#[test]

	// the authority wrote, carrying the seal that row holds, is the same authority
	// as the one the runner handed out. That is what makes the production path - a
	// ticket read back out of the database - the same path this test exercises.
	let Engine = Engine(TestClock::New());
	let Place = Engine.Destination();
	let Lease = Claimed(&Engine, "page", Span(1_000));

	let Receipt = Present(&Place, Restored, "restored").expect("the row the authority issued");

#[test]
fn a_takeover_waits_for_an_unsettled_local_future() {
	// This encodes the accounting rule and nothing more: a future this process
	// admitted has not settled, so a local takeover does not begin under it.
	// It is not the authority rule - that one is
	// `an_effect_that_outlives_its_lease_is_refused_where_it_lands_and_recorded_ambiguous`,
	// which lets the future settle and refuses the write anyway.
	let Clock = TestClock::New();
	let Engine = Engine(Clock.clone());
	let Lease = Claimed(&Engine, "page", Span(100));
	// The effect is admitted, and the lease expires while it is still running.
	let mut Write = Engine.WithLease(Lease.clone(), |_Ticket| Box::pin(std::future::pending::<i32>()));
	assert!(!PollOnce(Write.as_mut()), "the effect is admitted and waiting");
	Clock.Advance(Span(200));
	let Running = Engine.Claim("page", Span(1_000), None);
	let Error = BlockOn(Running).unwrap_err();
	assert_eq!(Error.Code(), Code::LeaseHeld);
	assert!(Error.Message().contains("unsettled local future"), "{}", Error.Message());
	// Dropping the abandoned effect settles it, which is what frees the
	// resource. Dropping a future is not proof that the work behind it stopped;
	// it is proof that this process is no longer counting it.
	drop(Write);
	assert_eq!(Claimed(&Engine, "page", Span(1_000)).Fence, 2);
}

#[test]
fn a_bound_call_stops_its_wait_when_the_caller_cancels() {
	// The one place the combinator itself is exercised: a page that never
	// answers is cut short by the caller, and the answer is a cancellation
	// rather than a timeout, so a consumer can tell its own impatience from
	// a slow page.
	struct Fixed(Instant);
	impl Clock for Fixed {
		fn Now(&self) -> Instant { self.0 }
	}
	struct Waiting;
	impl AgenticRuntime::Pace for Waiting {
		fn Sleep(&self, Span:Duration) -> AgenticRuntime::FutureBox<'_, ()> {
			// A wait that never ends: only the signal can stop this call.
			let _ = Span;
			Box::pin(std::future::pending())
		}
	}
	impl Clock for Waiting {
		fn Now(&self) -> Instant { Instant::now() }
	}

	let Signal = AgenticRuntime::Signal::New();
	let Call = Call::Within(Span(60_000), &Fixed(Instant::now()), Signal.clone());
	let Pace = Waiting;
	let Silent = Box::pin(std::future::pending::<u8>());
	let Bounded = AgenticRuntime::Bound(Silent, &Call, &Pace);
	Signal.Fire();
	assert_eq!(BlockOn(Bounded), Err(AgenticRuntime::Expired::Cancelled));
	assert!(Call.Cancelled());
	assert!(Signal.IsCancelled());
}

//! The corpus digest.
//!
//! A pure function of the records it is handed: no file is read, so a
//! modification time cannot enter it, and `Directory` is excluded, so the same
//! corpus digests the same in two checkouts. The canonical form is the JSON the
//! TypeScript implementation hashes, `[[name, description, references], ...]`,
//! which is why the digest of a given corpus is the same value across the two
//! trees and the contract test can assert a recorded constant.

use sha2::{Digest as _, Sha256};

use crate::Struct::Skill::Skill;

/// The corpus digest, prefixed with the algorithm that produced it.
pub fn Digest(Skills:&[Skill]) -> String {
	let Canonical:Vec<serde_json::Value> = Skills
		.iter()
		.map(|Skill| serde_json::json!([Skill.Name, Skill.Description, Skill.References]))
		.collect();
	let Bytes = serde_json::to_string(&Canonical).unwrap_or_else(|_| "[]".to_string());
	let Sum = Sha256::digest(Bytes.as_bytes());
	// Hex by hand: the digest type's own `LowerHex` is behind a feature this
	// crate does not need, and a format string is not a reason to take one.
	let Hex:String = Sum.iter().map(|Byte| format!("{Byte:02x}")).collect();
	format!("sha256:{Hex}")
}

//! Finding the skills under a corpus root, and reading one of them.
//!
//! Code-unit order, not a locale-aware collation: a sweep must not change
//! meaning with the machine's locale, and `String`'s `Ord` is exactly the byte
//! order the TypeScript implementation compares with.

use std::path::Path;

use crate::{
	Fn::{
		Entries::{BodyFile, CorpusIndexes, Entries, IndexFile, IsDirectory, IsFile, ReferenceDirectory, Text},
		Frontmatter::Split,
	},
	Struct::Skill::Skill,
};

/// Every skill directory under `Root`, sorted by name.
pub fn Discover(Root:&Path) -> Vec<Skill> {
	let mut Skills = Vec::new();
	for Entry in Entries(Root) {
		if !IsDirectory(&Entry, &Root.join(&Entry.Name)) {
			continue;
		}
		Skills.push(Read(Root, &Entry.Name));
	}
	Skills.sort_by(|Left, Right| Left.Name.cmp(&Right.Name));
	Skills
}

/// One skill's record: the directory name is the skill's name, because that is
/// the address a reader types and an index cites.
pub fn Read(Root:&Path, Name:&str) -> Skill {
	let Directory = Root.join(Name);
	let Body = Text(&Directory.join(BodyFile)).unwrap_or_default();
	let Description = Split(&Body)
		.ok()
		.and_then(|Data| Data.get("description").map(|Value| Value.trim().to_string()))
		.unwrap_or_default();
	Skill {
		Name:Name.to_string(),
		Directory:Directory.to_string_lossy().into_owned(),
		Description,
		References:References(&Directory),
	}
}

/// The file names directly under `Reference/`, excluding the index itself.
pub fn References(Directory:&Path) -> Vec<String> {
	let mut Names:Vec<String> = Entries(&Directory.join(ReferenceDirectory))
		.iter()
		.filter(|Entry| IsFile(Entry))
		.map(|Entry| Entry.Name.clone())
		.filter(|Name| Name != IndexFile)
		.collect();
	Names.sort();
	Names
}

/// The first index a corpus root carries, and the name it was found under.
pub fn CorpusIndex(Directory:&Path) -> Option<(String, String)> {
	for Name in CorpusIndexes {
		if let Some(Text) = Text(&Directory.join(Name)) {
			return Some((Name.to_string(), Text));
		}
	}
	None
}

//! Reading a directory, and the names this contract gives the files it reads.
//!
//! A root that cannot be read is an empty corpus rather than an error, because
//! a sweep that dies on one bad path cannot report the rest. A caller that must
//! tell a typo from an empty corpus stats the root itself - the port does not
//! pretend to know which of the two a missing directory was.

use std::{fs, path::Path};

/// The file a skill's frontmatter lives in.
pub const BodyFile:&str = "SKILL.md";

/// The index a skill's `Reference/` carries.
pub const IndexFile:&str = "Index.md";

/// The directory a skill's worked instances live in.
pub const ReferenceDirectory:&str = "Reference";

/// The corpus root's index, by the two names a corpus actually gives it.
///
/// `Index.md` is the port's own name, and the one it prefers when both exist.
/// `AGENTS.md` is the name a repository already gives the file its tools
/// discover, and a corpus that grows inside such a repository should not have
/// to rename it to be read - one file with two accepted names is cheaper than a
/// corpus that is uncited in its entirety. A skill's own `Reference/Index.md`
/// has one name only: there the card names the file.
pub const CorpusIndexes:[&str; 2] = [IndexFile, "AGENTS.md"];

/// One entry of a directory, as much of it as this contract reads.
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Entry {
	/// The entry's name, without its directory.
	pub Name:String,
	/// Whether the entry is a directory without following a link.
	pub Directory:bool,
	/// Whether the entry is a symbolic link.
	pub Symlink:bool,
	/// Whether the entry is a regular file without following a link.
	pub File:bool,
}

/// The entries of a directory, or none.
pub fn Entries(Path:&Path) -> Vec<Entry> {
	let Ok(Read) = fs::read_dir(Path) else {
		return Vec::new();
	};
	let mut Entries = Vec::new();
	for Entry in Read.flatten() {
		let Ok(Kind) = Entry.file_type() else {
			continue;
		};
		let Name = Entry.file_name().to_string_lossy().into_owned();
		Entries.push(Entry { Name, Directory:Kind.is_dir(), Symlink:Kind.is_symlink(), File:Kind.is_file() });
	}
	Entries.sort_by(|Left, Right| Left.Name.cmp(&Right.Name));
	Entries
}

/// A file's text, or none. An unreadable file reads as absent for the reason
/// above.
pub fn Text(Path:&Path) -> Option<String> { fs::read_to_string(Path).ok() }

/// Whether an entry is a directory, following a symlink one level.
pub fn IsDirectory(Entry:&Entry, Path:&Path) -> bool {
	if Entry.Directory {
		return true;
	}
	if !Entry.Symlink {
		return false;
	}
	fs::metadata(Path).map(|Meta| Meta.is_dir()).unwrap_or(false)
}

/// Whether an entry is a file or a link to one.
pub fn IsFile(Entry:&Entry) -> bool { Entry.File || Entry.Symlink }

--- FILE: Crate/SkillContract/Source/Fn/Frontmatter.rs ---
//! The frontmatter block and the fields this port promises from it.
//!
//! A YAML parser would be a dependency, and the two fields a loader reads do
//! not justify one. What is read: top-level `key: value` pairs, a value quoted
//! or plain, and a `|` or `>` block scalar, which is how a long description is
//! usually written. What is not read: nested mappings and sequences. Those are
//! skipped rather than refused, because a skill may carry metadata this port
//! does not promise and a validator that rejects unfamiliar keys is a validator
//! people turn off.

use std::collections::BTreeMap;

/// The frontmatter block and the fields this port promises from it.
///
/// `Ok` is the mapping the block described, and may be empty. `Err` is the
/// reason the block did not describe one, worded so `validate` can paste it
/// into a finding without rewriting it.
pub type Frontmatter = Result<BTreeMap<String, String>, String>;

/// The YAML block at the top of a file. A file without one has nothing to read.
pub fn Split(Text:&str) -> Frontmatter {
	let Normalised = Text.replace("\r\n", "\n");
	let Lines:Vec<&str> = Normalised.split('\n').collect();
	if Lines.first().map(|Line| Line.trim()) != Some("---") {
		return Err("there is no frontmatter block".to_string());
	}
	let Some(End) = Lines.iter().skip(1).position(|Line| *Line == "---") else {
		return Err("the frontmatter block is never closed".to_string());
	};
	Parse(&Lines[1..End + 1])
}

/// A `key: value` pair, or why the line is not one.
fn Pair(Line:&str) -> Option<(&str, &str)> {
	let Colon = Line.find(':')?;
	let Key = &Line[..Colon];
	if Key.is_empty() {
		return None;
	}
	if !Key
		.chars()
		.all(|Character| Character.is_ascii_alphanumeric() || matches!(Character, '_' | '.' | '-'))
	{
		return None;
	}
	Some((Key, &Line[Colon + 1..]))
}

/// Whether a value opens a block scalar, and whether it keeps its line breaks.
fn Block(Value:&str) -> Option<bool> {
	let First = Value.chars().next()?;
	if First != '|' && First != '>' {
		return None;
	}
	let Rest = &Value[First.len_utf8()..];
	if !Rest.is_empty() && Rest != "+" && Rest != "-" {
		return None;
	}
	Some(First == '|')
}

/// The number of leading whitespace bytes, which is the block's own
/// indentation.
fn Indent(Line:&str) -> usize { Line.len() - Line.trim_start().len() }

/// A `|` or `>` block scalar with the block's own indentation removed.
///
/// A folded scalar joins its lines into one, a literal one keeps the breaks.
/// Both drop blank lines, because a description is a sentence rather than a
/// document and the port reads exactly one field this way.
fn Scalar(Lines:&[&str], Literal:bool) -> String {
	let Indented:Vec<&str> = Lines.iter().copied().filter(|Line| !Line.trim().is_empty()).collect();
	let Some(Indent) = Indented.iter().map(|Line| Indent(Line)).min() else {
		return String::new();
	};
	let Stripped:Vec<&str> = Lines.iter().map(|Line| &Line[Indent.min(Line.len())..]).collect();
	if Literal {
		Stripped.join("\n").trim().to_string()
	} else {
		Stripped
			.iter()
			.filter(|Line| !Line.trim().is_empty())
			.copied()
			.collect::<Vec<&str>>()
			.join(" ")
			.trim()
			.to_string()
	}
}

/// A value with one matching pair of surrounding quotes removed.
fn Unquote(Value:&str) -> String {
	let Bytes = Value.as_bytes();
	if Bytes.len() >= 2 {
		let First = Bytes[0];
		if (First == b'"' || First == b'\'') && Bytes[Bytes.len() - 1] == First {
			return Value[1..Value.len() - 1].to_string();
		}
	}
	Value.to_string()
}

/// The pairs of a frontmatter block.
fn Parse(Lines:&[&str]) -> Frontmatter {
	let mut Data = BTreeMap::new();
	let mut Index = 0;
	while Index < Lines.len() {
		let Line = Lines[Index];
		Index += 1;
		let Trimmed = Line.trim();
		if Trimmed.is_empty() || Trimmed.starts_with('#') {
			continue;
		}
		let Some((Key, Value)) = Pair(Line) else {
			if Line.starts_with(char::is_whitespace) {
				continue;
			}
			return Err(format!(
				"{} is not a key: value pair",
				serde_json::to_string(Line).unwrap_or_else(|_| format!("{Line:?}"))
			));
		};
		let Value = Value.trim();
		if let Some(Literal) = Block(Value) {
			let mut Collected:Vec<&str> = Vec::new();
			while Index < Lines.len() {
				let Next = Lines[Index];
				if !Next.trim().is_empty() && !Next.starts_with(char::is_whitespace) {
					break;
				}
				Collected.push(Next);
				Index += 1;
			}
			Data.insert(Key.to_string(), Scalar(&Collected, Literal));
			continue;
		}
		Data.insert(Key.to_string(), Unquote(Value));
	}
	Ok(Data)
}

//! Whether a text names a token, as a whole name.
//!
//! A substring test is not enough: `Notes.md` is a substring of `Notes.md.bak`,
//! so an index could appear to name a file it never names. The boundaries are
//! the characters a file or directory name is built from, which is what the
//! TypeScript implementation spells as `(^|[^\w.-])token(?![\w.-])` - and this
//! is that rule, written without a regular-expression dependency so the crate
//! keeps its dependency list to a digest and a serialiser.

/// Whether a character may continue a file or directory name.
fn IsName(Character:char) -> bool { Character.is_alphanumeric() || matches!(Character, '_' | '.' | '-') }

/// Whether `Text` names `Token` between two name boundaries.
pub fn Names(Text:&str, Token:&str) -> bool {
	if Token.is_empty() {
		return false;
	}
	for (At, _) in Text.char_indices() {
		if !Text[At..].starts_with(Token) {
			continue;
		}
		let Before = Text[..At].chars().next_back();
		let After = Text[At + Token.len()..].chars().next();
		if !Before.is_some_and(IsName) && !After.is_some_and(IsName) {
			return true;
		}
	}
	false
}

//! Every rule a skill breaks.
//!
//! Three checks, in the order the manifest names them, and each one reports
//! rather than returns early: a skill can be unparsable and uncited at once,
//! and a sweep that stopped at the first finding would hide half the corpus's
//! problems.

use std::path::Path;

use crate::{
	Fn::{
		Discover::{CorpusIndex, References},
		Entries::{BodyFile, CorpusIndexes, IndexFile, ReferenceDirectory, Text},
		Frontmatter::Split,
		Names::Names,
	},
	Struct::{
		Error::Rule,
		Skill::{Finding, Skill},
	},
};

/// Every rule `Skill` breaks. An empty vector is a valid skill.
pub fn Validate(Skill:&Skill) -> Vec<Finding> {
	let mut Findings = Vec::new();
	let Directory = Path::new(&Skill.Directory);

	match Text(&Directory.join(BodyFile)) {
		None => {
			Findings.push(Finding::Of(
				Skill,
				Rule::Unparsable,
				format!("{} carries no {BodyFile}", Skill.Directory),
			))
		},
		Some(Body) => {
			match Split(&Body) {
				Err(Reason) => {
					Findings.push(Finding::Of(
						Skill,
						Rule::Unparsable,
						format!("the frontmatter does not parse: {Reason}"),
					))
				},
				Ok(Data) => {
					let Description = Data.get("description").map(|Value| Value.trim()).unwrap_or_default();
					if Description.is_empty() {
						Findings.push(Finding::Of(Skill, Rule::Unparsable, "the frontmatter states no description"));
					}
					let Name = Data.get("name").map(|Value| Value.trim()).unwrap_or_default();
					if !Name.is_empty() && Name.to_lowercase() != Skill.Name.to_lowercase() {
						Findings.push(Finding::Of(
							Skill,
							Rule::Unparsable,
							format!(
								"the frontmatter names {} but the directory is {}",
								Quote(Name),
								Quote(&Skill.Name)
							),
						));
					}
				},
			}
		},
	}

	let Index = Text(&Directory.join(ReferenceDirectory).join(IndexFile));
	for Reference in References(Directory) {
		match &Index {
			None => {
				Findings.push(Finding::Of(
					Skill,
					Rule::UnlistedReference,
					format!(
						"{ReferenceDirectory}/{Reference} is not named in {ReferenceDirectory}/{IndexFile}, because \
						 there is no {ReferenceDirectory}/{IndexFile}"
					),
				))
			},
			Some(Index) if !Names(Index, &Reference) => {
				Findings.push(Finding::Of(
					Skill,
					Rule::UnlistedReference,
					format!("{ReferenceDirectory}/{Reference} is not named in {ReferenceDirectory}/{IndexFile}"),
				))
			},
			Some(_) => {},
		}
	}

	let Corpus = Directory.parent().and_then(CorpusIndex);
	match Corpus {
		None => {
			Findings.push(Finding::Of(
				Skill,
				Rule::Uncited,
				format!(
					"the corpus root carries no {}, so nothing points at {}",
					CorpusIndexes.join(" or "),
					Skill.Name
				),
			))
		},
		Some((Name, Text)) if !Names(&Text, &Skill.Name) => {
			Findings.push(Finding::Of(
				Skill,
				Rule::Uncited,
				format!("{Name} does not name {}/", Skill.Name),
			))
		},
		Some(_) => {},
	}

	Findings
}

/// A name as a quoted literal, so a finding cannot be misread.
fn Quote(Value:&str) -> String { serde_json::to_string(Value).unwrap_or_else(|_| format!("{Value:?}")) }

//! `agentic.skill.validate`: discover a corpus, report what a skill breaks,
//! digest the result.
//!
//! A corpus is a directory of skill directories. A skill is a directory
//! carrying a `SKILL.md` and, when it has worked instances, a `Reference/`
//! tree. A skill's `Reference/` carries an `Index.md` and the corpus root
//! carries an index, and each index names its siblings: the root's names the
//! skills, a skill's names its files.
//!
//! Everything here reads the filesystem and nothing here caches it. There is no
//! module state, so two calls with the same arguments answer the same findings
//! and a call after the tree changed answers the new ones.
//!
//! The port's three codes are [`Struct::Error::Rule`]s rather than exceptions:
//! `validate` reports, it does not throw, so one malformed skill cannot abort a
//! sweep over a corpus of good ones. A rule carries `detail` for the same
//! reason an error does - the sentence is for a reader, the observation is for
//! whoever has to fix it.

//! The implementation of the port in `Port.rs`: a stateless contract over a
//! directory of skill directories.
//!
//! There is no state beyond the constants in `Fn/`, so two calls with the same
//! arguments answer the same findings and a call after the tree changed answers
//! the new ones. `digest` is the one exception, and it is the opposite of a
//! cache: it is a pure function of the records it is handed, so it can be
//! computed for a corpus that is not on this machine at all.

use std::path::Path;

use AgenticRuntime::FutureBox;

use crate::{
	Fn::{Digest::Digest, Discover::Discover, Validate::Validate},
	Struct::{
		Port::SkillContract,
		Skill::{Finding, Skill},
	},
};

/// A stateless contract. It holds nothing, which is the point.
#[derive(Debug, Default)]
pub struct Contract;

/// Build one. A contract has no configuration, so this takes none.
pub fn Create() -> Contract { Contract }

impl SkillContract for Contract {

//! The three rules `validate` reports, and the only three it may report.
//!
//! They mirror `provides.errors` in `module.json`, which is what a harness
//! reads; the contract test compares the two so the manifest cannot drift from
//! the code.
//!
//! - `unparsable` - the frontmatter does not describe the skill that directory
//!   is: no `SKILL.md`, no frontmatter block, a block that is not a mapping, no
//!   `description`, or a `name` that disagrees with the directory. The
//!   directory is the skill's address, so a frontmatter name that contradicts
//!   it is a rename that stopped halfway.
//! - `unlisted-reference` - a file under `Reference/` that `Reference/Index.md`
//!   does not name, which is a reader's dead end.
//! - `uncited` - the corpus root's index - `Index.md`, or `AGENTS.md` - does
//!   not name this skill, which is a skill no reader is pointed at.
//!
//! Two rules the source implementation enforces are deliberately not here:
//! whether the body carries a claim-to-test table, and whether a second,
//! reader-facing index names the skill. Both are policies of one repository
//! rather than properties of a corpus, and a port that hard-coded them would
//! refuse every corpus that does not share that repository's house style.

use std::fmt;

/// The rules, in the order the manifest names them.
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub enum Rule {
	/// The frontmatter does not describe the skill that directory is.
	Unparsable,
	/// A file under `Reference/` that the index does not name.
	UnlistedReference,
	/// The corpus root's index does not name this skill.
	Uncited,
}

impl Rule {
	/// The string the manifest uses.
	pub const fn Name(self) -> &'static str {
		match self {
			Rule::Unparsable => "unparsable",
			Rule::UnlistedReference => "unlisted-reference",
			Rule::Uncited => "uncited",
		}
	}

	/// Every rule, so a harness can compare the manifest against the type.
	pub const fn All() -> [Rule; 3] { [Rule::Unparsable, Rule::UnlistedReference, Rule::Uncited] }

	/// The rule a manifest string names, if it names one of the three.
	pub fn Parse(Value:&str) -> Option<Rule> { Rule::All().into_iter().find(|Rule| Rule.Name() == Value) }
}

/// The three rules, in the order the manifest names them.
pub const Rules:[Rule; 3] = Rule::All();

/// The one failure this capability has, for the caller that must fail loudly.
///
/// `discover` and `validate` do not return it: a malformed skill is a finding.
/// A command that exits non-zero, though, needs one value to branch on, and
/// inventing a fourth vocabulary at the command layer is how a port stops being
/// the contract. It carries the same three codes as a finding.
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
	/// The frontmatter does not describe the skill that directory is.
	Unparsable {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
	/// A file under `Reference/` that the index does not name.
	UnlistedReference {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
	/// The corpus root's index does not name this skill.
	Uncited {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
}

impl Error {
	/// The code this failure carries.
	pub const fn Rule(&self) -> Rule {
		match self {
			Error::Unparsable { .. } => Rule::Unparsable,
			Error::UnlistedReference { .. } => Rule::UnlistedReference,
			Error::Uncited { .. } => Rule::Uncited,
		}
	}

	/// The sentence a human reads.
	pub fn Message(&self) -> &str {
		match self {
			Error::Unparsable { Message, .. }
			| Error::UnlistedReference { Message, .. }
			| Error::Uncited { Message, .. } => Message,
		}
	}

	/// What was observed, which may be empty.
	pub fn Detail(&self) -> &str {
		match self {
			Error::Unparsable { Detail, .. }
			| Error::UnlistedReference { Detail, .. }
			| Error::Uncited { Detail, .. } => Detail,
		}
	}
}

impl fmt::Display for Error {
	fn fmt(&self, Formatter:&mut fmt::Formatter<'_>) -> fmt::Result {
		write!(Formatter, "{}: {}", self.Rule().Name(), self.Message())
	}
}

impl std::error::Error for Error {}

//! The port `agentic.skill.validate` promises.
//!
//! The port states more than data shapes, so the semantics two adapters would
//! otherwise disagree about are fixed here rather than left to an
//! implementation:
//!
//! - `discover` is deterministic. Two runs over the same tree return the same
//!   order, sorted by skill name rather than by whatever the filesystem
//!   answered, because directory order is not stable across platforms or file
//!   systems.
//! - `validate` reports, it does not fail. A malformed skill is a finding, so
//!   one bad directory cannot abort a sweep over a corpus of good ones.
//! - `digest` is a pure function of the records it is handed. It reads no file,
//!   so a modification time cannot enter it, and `directory` is excluded from
//!   it, so the same corpus digests the same in two checkouts.
//! - Stateless means stateless. Nothing is cached between calls, and `validate`
//!   re-reads the skill's directory instead of trusting the record it is given:
//!   a record is a claim about a directory, and a cache of that claim is how a
//!   sweep reports yesterday's corpus.
//!
//! The methods are boxed futures rather than plain functions because the port
//! shape is preserved from the TypeScript capability, where every verb answered
//! a promise. Nothing in this crate awaits, and the work is filesystem work: a
//! caller on an async runtime should drive these from a blocking context rather
//! than assume the box makes the read asynchronous.

use AgenticRuntime::FutureBox;

use crate::Struct::Skill::{Finding, Skill};

/// The one port this capability offers.
pub trait SkillContract: Send + Sync {
	/// Every skill directory under the corpus root.
	fn Discover<'a>(&'a self, Root:&'a str) -> FutureBox<'a, Vec<Skill>>;
	/// Every rule a skill breaks. An empty vector is a valid corpus.
	fn Validate<'a>(&'a self, Skill:&'a Skill) -> FutureBox<'a, Vec<Finding>>;
	/// A stable digest of the corpus: names, order and reference lists.
	fn Digest(&self, Skills:&[Skill]) -> String;
}
