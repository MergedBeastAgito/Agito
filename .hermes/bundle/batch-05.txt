//! Whether a text names a token, as a whole name.
//!
//! A substring test is not enough: `Notes.md` is a substring of `Notes.md.bak`,
//! so an index could appear to name a file it never names. The boundaries are
//! the characters a file or directory name is built from, which is what the
//! TypeScript implementation spells as `(^|[^\w.-])token(?![\w.-])` - and this
//! is that rule, written without a regular-expression dependency so the crate
//! keeps its dependency list to a digest and a serialiser.

/// Whether a character may continue a file or directory name.
fn IsName(Character:char) -> bool { Character.is_alphanumeric() || matches!(Character, '_' | '.' | '-') }

/// Whether `Text` names `Token` between two name boundaries.
pub fn Names(Text:&str, Token:&str) -> bool {
	if Token.is_empty() {
		return false;
	}
	for (At, _) in Text.char_indices() {
		if !Text[At..].starts_with(Token) {
			continue;
		}
		let Before = Text[..At].chars().next_back();
		let After = Text[At + Token.len()..].chars().next();
		if !Before.is_some_and(IsName) && !After.is_some_and(IsName) {
			return true;
		}
	}
	false
}

--- FILE: rust/Crate/SkillContract/Source/Fn/Validate.rs ---
//! Every rule a skill breaks.
//!
//! Three checks, in the order the manifest names them, and each one reports
//! rather than returns early: a skill can be unparsable and uncited at once,
//! and a sweep that stopped at the first finding would hide half the corpus's
//! problems.

use std::path::Path;

use crate::{
	Fn::{
		Discover::{CorpusIndex, References},
		Entries::{BodyFile, CorpusIndexes, IndexFile, ReferenceDirectory, Text},
		Frontmatter::Split,
		Names::Names,
	},
	Struct::{
		Error::Rule,
		Skill::{Finding, Skill},
	},
};

/// Every rule `Skill` breaks. An empty vector is a valid skill.
pub fn Validate(Skill:&Skill) -> Vec<Finding> {
	let mut Findings = Vec::new();
	let Directory = Path::new(&Skill.Directory);

	match Text(&Directory.join(BodyFile)) {
		None => {
			Findings.push(Finding::Of(
				Skill,
				Rule::Unparsable,
				format!("{} carries no {BodyFile}", Skill.Directory),
			))
		},
		Some(Body) => {
			match Split(&Body) {
				Err(Reason) => {
					Findings.push(Finding::Of(
						Skill,
						Rule::Unparsable,
						format!("the frontmatter does not parse: {Reason}"),
					))
				},
				Ok(Data) => {
					let Description = Data.get("description").map(|Value| Value.trim()).unwrap_or_default();
					if Description.is_empty() {
						Findings.push(Finding::Of(Skill, Rule::Unparsable, "the frontmatter states no description"));
					}
					let Name = Data.get("name").map(|Value| Value.trim()).unwrap_or_default();
					if !Name.is_empty() && Name.to_lowercase() != Skill.Name.to_lowercase() {
						Findings.push(Finding::Of(
							Skill,
							Rule::Unparsable,
							format!(
								"the frontmatter names {} but the directory is {}",
								Quote(Name),
								Quote(&Skill.Name)
							),
						));
					}
				},
			}
		},
	}

	let Index = Text(&Directory.join(ReferenceDirectory).join(IndexFile));
	for Reference in References(Directory) {
		match &Index {
			None => {
				Findings.push(Finding::Of(
					Skill,
					Rule::UnlistedReference,
					format!(
						"{ReferenceDirectory}/{Reference} is not named in {ReferenceDirectory}/{IndexFile}, because \
						 there is no {ReferenceDirectory}/{IndexFile}"
					),
				))
			},
			Some(Index) if !Names(Index, &Reference) => {
				Findings.push(Finding::Of(
					Skill,
					Rule::UnlistedReference,
					format!("{ReferenceDirectory}/{Reference} is not named in {ReferenceDirectory}/{IndexFile}"),
				))
			},
			Some(_) => {},
		}
	}

	let Corpus = Directory.parent().and_then(CorpusIndex);
	match Corpus {
		None => {
			Findings.push(Finding::Of(
				Skill,
				Rule::Uncited,
				format!(
					"the corpus root carries no {}, so nothing points at {}",
					CorpusIndexes.join(" or "),
					Skill.Name
				),
			))
		},
		Some((Name, Text)) if !Names(&Text, &Skill.Name) => {
			Findings.push(Finding::Of(
				Skill,
				Rule::Uncited,
				format!("{Name} does not name {}/", Skill.Name),
			))
		},
		Some(_) => {},
	}

	Findings
}

/// A name as a quoted literal, so a finding cannot be misread.
fn Quote(Value:&str) -> String { serde_json::to_string(Value).unwrap_or_else(|_| format!("{Value:?}")) }

//!
//! A corpus is a directory of skill directories. A skill is a directory
//! carrying a `SKILL.md` and, when it has worked instances, a `Reference/`

//! directory of skill directories.

--- FILE: rust/Crate/SkillContract/Source/Struct/Error.rs ---
//! The three rules `validate` reports, and the only three it may report.
//!
//! They mirror `provides.errors` in `module.json`, which is what a harness
//! reads; the contract test compares the two so the manifest cannot drift from
//! the code.
//!
//! - `unparsable` - the frontmatter does not describe the skill that directory
//!   is: no `SKILL.md`, no frontmatter block, a block that is not a mapping, no
//!   `description`, or a `name` that disagrees with the directory. The
//!   directory is the skill's address, so a frontmatter name that contradicts
//!   it is a rename that stopped halfway.
//! - `unlisted-reference` - a file under `Reference/` that `Reference/Index.md`
//!   does not name, which is a reader's dead end.
//! - `uncited` - the corpus root's index - `Index.md`, or `AGENTS.md` - does
//!   not name this skill, which is a skill no reader is pointed at.
//!
//! Two rules the source implementation enforces are deliberately not here:
//! whether the body carries a claim-to-test table, and whether a second,
//! reader-facing index names the skill. Both are policies of one repository
//! rather than properties of a corpus, and a port that hard-coded them would
//! refuse every corpus that does not share that repository's house style.

use std::fmt;

/// The rules, in the order the manifest names them.
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash)]
pub enum Rule {
	/// The frontmatter does not describe the skill that directory is.
	Unparsable,
	/// A file under `Reference/` that the index does not name.
	UnlistedReference,
	/// The corpus root's index does not name this skill.
	Uncited,
}

impl Rule {
	/// The string the manifest uses.
	pub const fn Name(self) -> &'static str {
		match self {
			Rule::Unparsable => "unparsable",
			Rule::UnlistedReference => "unlisted-reference",
			Rule::Uncited => "uncited",
		}
	}

	/// Every rule, so a harness can compare the manifest against the type.
	pub const fn All() -> [Rule; 3] { [Rule::Unparsable, Rule::UnlistedReference, Rule::Uncited] }

	/// The rule a manifest string names, if it names one of the three.
	pub fn Parse(Value:&str) -> Option<Rule> { Rule::All().into_iter().find(|Rule| Rule.Name() == Value) }
}

/// The three rules, in the order the manifest names them.
pub const Rules:[Rule; 3] = Rule::All();

/// The one failure this capability has, for the caller that must fail loudly.
///
/// `discover` and `validate` do not return it: a malformed skill is a finding.
/// A command that exits non-zero, though, needs one value to branch on, and
/// inventing a fourth vocabulary at the command layer is how a port stops being
/// the contract. It carries the same three codes as a finding.
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
	/// The frontmatter does not describe the skill that directory is.
	Unparsable {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
	/// A file under `Reference/` that the index does not name.
	UnlistedReference {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
	/// The corpus root's index does not name this skill.
	Uncited {
		/// The sentence a human reads.
		Message:String,
		/// What was observed.
		Detail:String,
	},
}

impl Error {
	/// The code this failure carries.
	pub const fn Rule(&self) -> Rule {
		match self {
			Error::Unparsable { .. } => Rule::Unparsable,
			Error::UnlistedReference { .. } => Rule::UnlistedReference,
			Error::Uncited { .. } => Rule::Uncited,
		}
	}

	/// The sentence a human reads.
	pub fn Message(&self) -> &str {
		match self {
			Error::Unparsable { Message, .. }
			| Error::UnlistedReference { Message, .. }
			| Error::Uncited { Message, .. } => Message,
		}
	}

	/// What was observed, which may be empty.
	pub fn Detail(&self) -> &str {
		match self {
			Error::Unparsable { Detail, .. }
			| Error::UnlistedReference { Detail, .. }
			| Error::Uncited { Detail, .. } => Detail,
		}
	}
}

impl fmt::Display for Error {
	fn fmt(&self, Formatter:&mut fmt::Formatter<'_>) -> fmt::Result {
		write!(Formatter, "{}: {}", self.Rule().Name(), self.Message())
	}
}

impl std::error::Error for Error {}

--- FILE: rust/Crate/SkillContract/Source/Struct/Port.rs ---
//! The port `agentic.skill.validate` promises.
//!
//! The port states more than data shapes, so the semantics two adapters would
//! otherwise disagree about are fixed here rather than left to an
//! implementation:
//!
//! - `discover` is deterministic. Two runs over the same tree return the same
//!   order, sorted by skill name rather than by whatever the filesystem
//!   answered, because directory order is not stable across platforms or file
//!   systems.
//! - `validate` reports, it does not fail. A malformed skill is a finding, so
//!   one bad directory cannot abort a sweep over a corpus of good ones.
//! - `digest` is a pure function of the records it is handed. It reads no file,
//!   so a modification time cannot enter it, and `directory` is excluded from
//!   it, so the same corpus digests the same in two checkouts.
//! - Stateless means stateless. Nothing is cached between calls, and `validate`
//!   re-reads the skill's directory instead of trusting the record it is given:
//!   a record is a claim about a directory, and a cache of that claim is how a
//!   sweep reports yesterday's corpus.
//!
//! The methods are boxed futures rather than plain functions because the port
//! shape is preserved from the TypeScript capability, where every verb answered
//! a promise. Nothing in this crate awaits, and the work is filesystem work: a
//! caller on an async runtime should drive these from a blocking context rather
//! than assume the box makes the read asynchronous.

use AgenticRuntime::FutureBox;

use crate::Struct::Skill::{Finding, Skill};

/// The one port this capability offers.
pub trait SkillContract: Send + Sync {
	/// Every skill directory under the corpus root.
	fn Discover<'a>(&'a self, Root:&'a str) -> FutureBox<'a, Vec<Skill>>;
	/// Every rule a skill breaks. An empty vector is a valid corpus.
	fn Validate<'a>(&'a self, Skill:&'a Skill) -> FutureBox<'a, Vec<Finding>>;
	/// A stable digest of the corpus: names, order and reference lists.
	fn Digest(&self, Skills:&[Skill]) -> String;
}

//! One skill's record, and one rule it breaks.
//!
//! The record is what a sweep carries: the directory name is the skill's name,
//! because that is the address a reader types and an index cites, and
//! `Directory` is excluded from the digest so the same corpus digests the same
//! in two checkouts.

use serde::{Deserialize, Serialize};

use crate::Struct::Error::Rule;

/// One skill directory, as `discover` reports it.
#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub struct Skill {
	/// The directory name, which is the skill's name.
	pub Name:String,
	/// Where the skill lives. Excluded from the digest on purpose.
	pub Directory:String,
	/// The frontmatter's `description`, trimmed; empty when there is none.
	pub Description:String,
	/// The file names directly under `Reference/`, excluding its index, sorted.
	pub References:Vec<String>,
}

/// One rule a skill breaks, as `validate` reports it.
#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
pub struct Finding {
	/// The skill the finding is about.
	pub Skill:String,
	/// Which of the three rules it breaks.
	pub Rule:String,
	/// What was observed, kept apart from the sentence a human reads.
	pub Detail:String,
}

impl Finding {
	/// A finding for one skill, from a rule and what was observed.
	pub fn Of(Skill:&Skill, Rule:Rule, Detail:impl Into<String>) -> Self {
		Self { Skill:Skill.Name.clone(), Rule:Rule.Name().to_string(), Detail:Detail.into() }
	}

	/// The rule this finding reports, if it is one of the three.
	pub fn Code(&self) -> Option<Rule> { Rule::Parse(&self.Rule) }
}

//! `agentic.skill.validate`, exercised against corpora this test builds on
//! disk.
//!
//! No network, no browser and no fixture files checked in: a corpus is a
//! temporary directory the test writes, so a test can add a file between two
//! calls and prove that nothing was cached.

use std::{
	fs,
	path::{Path, PathBuf},
};

use AgenticRuntime::BlockOn;
use AgenticSkillContract::{Create, Finding, Rule, Rules, Skill, SkillContract};

/// A fresh, empty corpus root.
fn Root(Tag:&str) -> PathBuf {
	let Base = Path::new(env!("CARGO_TARGET_TMPDIR")).join(Tag);
	let _ = fs::remove_dir_all(&Base);
	fs::create_dir_all(&Base).expect("the corpus root is writable");
	Base
}

/// A skill directory carrying one `SKILL.md`.
fn SkillAt(Root:&Path, Name:&str, Frontmatter:&str) -> PathBuf {
	let Directory = Root.join(Name);
	fs::create_dir_all(&Directory).expect("the skill directory is writable");
	fs::write(Directory.join("SKILL.md"), format!("---\n{Frontmatter}\n---\n\n# {Name}\n"))
		.expect("SKILL.md is writable");
	Directory
}

/// One file under a skill's `Reference/`.
fn ReferenceAt(Directory:&Path, Name:&str, Body:&str) {
	let References = Directory.join("Reference");
	fs::create_dir_all(&References).expect("Reference/ is writable");
	fs::write(References.join(Name), Body).expect("the reference is writable");
}

/// A skill's `Reference/Index.md`.
fn ReferenceIndexAt(Directory:&Path, Body:&str) { ReferenceAt(Directory, "Index.md", Body); }

/// The corpus root's index.
fn CorpusIndexAt(Root:&Path, Name:&str, Body:&str) {
	fs::write(Root.join(Name), Body).expect("the corpus index is writable");
}

fn Discovered(Root:&Path, Name:&str) -> Skill {
	let Skills = BlockOn(Create().Discover(&Root.to_string_lossy()));
	Skills
		.into_iter()
		.find(|Candidate| Candidate.Name == Name)
		.unwrap_or_else(|| panic!("no skill named {Name} was discovered"))
}

fn Findings(Skill:&Skill) -> Vec<Finding> { BlockOn(Create().Validate(Skill)) }

fn RulesOf(Findings:&[Finding]) -> Vec<&str> { Findings.iter().map(|Finding| Finding.Rule.as_str()).collect() }

/// The fixture the recorded digest belongs to.
fn Valid(Root:&Path) -> Skill {
	let Directory = SkillAt(
		Root,
		"Alpha",
		"name: alpha\ndescription: One honest skill, with its reference listed.",
	);
	ReferenceAt(&Directory, "Notes.md", "# Notes\n");
	ReferenceIndexAt(&Directory, "- [Notes.md](Notes.md)\n");
	CorpusIndexAt(Root, "Index.md", "- [Alpha](Alpha/SKILL.md)\n");
	Discovered(Root, "Alpha")
}

--- FILE: rust/Crate/Workflow/Source/Fn/Measure.rs ---
//! The one expression that turns a page into measurements.
//!
//! This is a *workflow* decision, not a capability's: the verification package
//! must not know what an element is, so the convention for reading one lives
//! here, where a particular site can replace it.
//!
//! What it reads: `data-value` when the element carries one, parsed as a number
//! when it is numeric, and the element's trimmed text otherwise. A selector
//! that matches nothing is `present: false` with a `null` value, which is the
//! wire shape the capability's port asks for.

use serde_json::Value;

/// A JavaScript expression that answers one measurement per selector.
pub fn MeasurementExpression(Selectors:&[String]) -> String {
	let Selectors = serde_json::to_string(Selectors).unwrap_or_else(|_| "[]".to_string());
	format!(
		"(() => {{\n\tconst Read = (Selector) => {{\n\t\tconst Element = document.querySelector(Selector);\n\t\tif \
		 (Element === null) return {{ selector: Selector, present: false, value: null }};\n\t\tconst Raw = \
		 Element.getAttribute(\"data-value\");\n\t\tif (Raw !== null) {{\n\t\t\tconst Parsed = \
		 Number(Raw);\n\t\t\treturn {{ selector: Selector, present: true, value: Number.isNaN(Parsed) ? Raw : Parsed \
		 }};\n\t\t}}\n\t\treturn {{ selector: Selector, present: true, value: (Element.innerText || \"\").trim() \
		 }};\n\t}};\n\treturn {Selectors}.map(Read);\n}})()"
	)
}

/// The selectors a caller asked about, as a JSON array, for a harness that
/// wants to see what was sent without building a page.
pub fn MeasurementSelectors(Expression:&str) -> Option<Value> {
	let Start = Expression.rfind("return ")?;
	let Rest = &Expression[Start + "return ".len()..];
	let End = Rest.find(".map(Read)")?;
	serde_json::from_str(&Rest[..End]).ok()
}

--- FILE: rust/Crate/Workflow/Source/Fn/VerifySkillPage.rs ---
//! One composition that uses four capabilities in the order a release check
//! does.
//!
//! A corpus is swept, every skill is validated, the corpus is digested, and
//! then a built page is measured and judged. It is written here rather than in
//! a capability because it is the composition that knows all four, and it is
//! written at all because a workflow crate with no workflow in it is a
//! directory.
//!
//! The order is the interesting part: the corpus is read before the page is
//! touched, so a release check that finds a broken skill does not pay for a
//! browser, and a page that cannot be measured does not hide the findings that
//! were already collected.

use AgenticDesignVerification::{Invariant, Measurement, Verdict, Verify};
use AgenticRuntime::{Call, FutureBox};
use AgenticSkillContract::{Finding, Skill, SkillContract};

/// What one release check found.
#[derive(Clone, Debug, Default, PartialEq)]
pub struct Outcome {
	/// Every skill the corpus holds.
	pub Skills:Vec<Skill>,
	/// Every rule a skill broke.
	pub Findings:Vec<Finding>,
	/// The corpus digest, which is stable across checkouts.
	pub Digest:String,
	/// What the page measured, when there was a page to measure.
	pub Measurements:Vec<Measurement>,
	/// The verdict, when there was one.
	pub Verdict:Option<Verdict>,
	/// Why the page could not be measured, when it could not be.
	pub Failure:Option<String>,
}

impl Outcome {
	/// Whether the corpus is clean and the page held.
	pub fn Held(&self) -> bool {
		self.Findings.is_empty() && self.Failure.is_none() && self.Verdict.as_ref().is_none_or(|Verdict| Verdict.Held)
	}
}

/// Sweep a corpus, then measure and judge a page.
pub fn VerifySkillPage<'a>(
	Contract:&'a dyn SkillContract,
	Verify:&'a dyn Verify,
	Root:&'a str,
	Selectors:&'a [String],
	Invariants:&'a [Invariant],
	Call:Call,
) -> FutureBox<'a, Outcome> {
	Box::pin(async move {
		let Skills = Contract.Discover(Root).await;
		let mut Findings = Vec::new();
		for Skill in &Skills {
			Findings.extend(Contract.Validate(Skill).await);
		}
		let Digest = Contract.Digest(&Skills);
		let mut Outcome = Outcome { Skills, Findings, Digest, ..Outcome::default() };

		if Selectors.is_empty() {
			return Outcome;
		}

		match Verify.Measure(Selectors, Call).await {
			Ok(Measurements) => {
				Outcome.Verdict = Some(Verify.Check(Invariants, &Measurements));
				Outcome.Measurements = Measurements;
			},
			Err(Why) => Outcome.Failure = Some(format!("{}: {}", Why.Code().Name(), Why.Message())),
		}
		Outcome
	})
}

--- FILE: rust/Crate/Workflow/Source/Struct/Binding/Measure.rs ---
//! Design verification's `Browse`, over the browser substrate.
//!
//! The capability's port has one verb and no session, so this binding opens the
//! configured page for each measurement and releases it afterwards. That is not
//! an oversight: the manifest says `stateless`, and a binding that kept a page
//! alive between measurements would turn a stateless capability into a
//! `holds-transport` one behind its own manifest.
//!
//! The expression this binding sends is the one place a page convention lives,
//! and it is deliberately in the workflow rather than in the verification
//! package: the package must not know what an element is, and a workflow is
//! exactly the layer that decides how a particular site is read.

use std::sync::Arc;

use AgenticBrowserCDP::{Browse as BrowserBrowse, Error as BrowserError, OpenOptions};
use AgenticDesignVerification::{Browse, Failure};
use AgenticRuntime::{Call, FutureBox};
use serde_json::Value;

use crate::Fn::Measure::MeasurementExpression;

/// The verification capability's transport, served by the browser substrate.
pub struct Measure {
	Browse:Arc<dyn BrowserBrowse>,
	Url:String,
}

/// Bind design verification's `Browse` to one page of one browser.
pub fn Create(Browse:Arc<dyn BrowserBrowse>, Url:impl Into<String>) -> Measure { Measure { Browse, Url:Url.into() } }

/// A browser failure, said in the verification port's vocabulary.
fn Classify(Why:BrowserError) -> Failure {
	Failure {
		Message:format!("{}: {}", Why.Code().Name(), Why.Message()),
		Detail:Why.Detail().to_string(),
	}
}

impl Browse for Measure {
	fn Measure<'a>(&'a self, Selectors:&'a [String], Call:Call) -> FutureBox<'a, Result<Vec<Value>, Failure>> {
		Box::pin(async move {
			let Target = self
				.Browse
				.Open(&self.Url, OpenOptions::Launch(Call.clone()))
				.await
				.map_err(Classify)?;
			let Expression = MeasurementExpression(Selectors);
			let Read = self.Browse.Evaluate(&Target, &Expression, Call).await;
			// Released on every path: the capability is stateless, so the page
			// must not outlive the measurement that asked for it.
			self.Browse.Dispose(Target).await;
			match Read {
				Ok(Value::Array(Entries)) => Ok(Entries),
				Ok(Other) => {
					Err(Failure::Of(format!(
						"the page answered {Other} instead of a list of measurements"
					)))
				},
				Err(Why) => Err(Classify(Why)),
			}
		})
	}
}

--- FILE: rust/Crate/Workflow/Source/Struct/Binding/Page.rs ---
//! The chat driver's `Page`, over the browser substrate.
//!
//! Three translations, and each one is a decision this layer owns:
//!
//! 1. **Ownership.** `Page` has no ownership because a chat driver must not
//!    care whether the browser was launched or attached. This binding launches,
//!    because a capability that asks a product a question owns the page it
//!    opened; a caller that wants a shared browser supplies a `Browse` that
//!    attaches instead.
//! 2. **The handle.** The driver passes back whatever `Open` returned and never
//!    reads it, so the browser target lives here, keyed by the URL the driver
//!    does keep. That is the whole of "a provider may hang whatever it needs
//!    there".
//! 3. **The vocabulary.** A browser failure has three codes and the chat port
//!    has four; the browser's code and message travel in the failure's
//!    `message`, and its observation in `detail`, so the driver can map the
//!    whole thing onto `not-ready` without losing which of the three happened.

use std::{
	collections::HashMap,
	sync::{Arc, Mutex, MutexGuard},
};

use AgenticBrowserCDP::{Browse as BrowserBrowse, Error as BrowserError, OpenOptions, Target as BrowserTarget};
use AgenticChatDriver::{Failure, Page as ChatPage, Target};
use AgenticRuntime::{Call, FutureBox};
use serde_json::Value;

/// The chat driver's page, served by the browser substrate.
pub struct Page {
	Browse:Arc<dyn BrowserBrowse>,
	/// What each chat target was bound to, by URL.
	Held:Mutex<HashMap<String, BrowserTarget>>,
}

/// Bind the chat driver's `Page` to a browser.
pub fn Create(Browse:Arc<dyn BrowserBrowse>) -> Page { Page { Browse, Held:Mutex::new(HashMap::new()) } }

impl Page {
	fn Lock(&self) -> MutexGuard<'_, HashMap<String, BrowserTarget>> {
		self.Held.lock().unwrap_or_else(|Poison| Poison.into_inner())
	}
}

/// A browser failure, said in the chat port's vocabulary.
///
/// The code is kept: a consumer reading `detail` can still tell an unavailable
/// browser from a page that failed to load, which is the difference between
/// retrying and fixing the page.
fn Classify(Why:BrowserError) -> Failure {
	Failure {
		Message:format!("{}: {}", Why.Code().Name(), Why.Message()),
		Detail:Why.Detail().to_string(),
	}
}

impl ChatPage for Page {
	fn Open<'a>(&'a self, Url:&'a str, Call:Call) -> FutureBox<'a, Result<Target, Failure>> {
		Box::pin(async move {
			let Target = self.Browse.Open(Url, OpenOptions::Launch(Call)).await.map_err(Classify)?;
			self.Lock().insert(Target.Url.clone(), Target.clone());
			Ok(Target { Url:Target.Url })
		})
	}

	fn Evaluate<'a>(
		&'a self,
		Target:&'a Target,
		Expression:&'a str,
		Call:Call,
	) -> FutureBox<'a, Result<Value, Failure>> {
		Box::pin(async move {
			let Held = self.Lock().get(&Target.Url).cloned();
			let Some(Held) = Held else {
				return Err(Failure::Of("this binding holds no browser target for the page"));
			};
			self.Browse.Evaluate(&Held, Expression, Call).await.map_err(Classify)
		})
	}

	fn Dispose<'a>(&'a self, Target:Target) -> FutureBox<'a, ()> {
		Box::pin(async move {
			let Held = self.Lock().remove(&Target.Url);
			if let Some(Held) = Held {
				self.Browse.Dispose(Held).await;
			}
		})
	}
}

//!

};

fn Span(Milliseconds:u64) -> Duration { Duration::from_millis(Milliseconds) }

/// A clock the test owns.
struct TestPace {
	Now:Mutex<Instant>,
}

impl TestPace {
	fn New() -> Arc<Self> { Arc::new(Self { Now:Mutex::new(Instant::now()) }) }
}

impl Clock for TestPace {
	fn Now(&self) -> Instant { *self.Now.lock().unwrap() }
}

impl Pace for TestPace {
	fn Sleep(&self, Span:Duration) -> FutureBox<'_, ()> {
		*self.Now.lock().unwrap() += Span.max(Duration::from_millis(1));
		Box::pin(std::future::ready(()))
	}
}

/// A browser that answers the expression it is handed.
struct FakeBrowse {
	Answer:Mutex<Result<Value, BrowserError>>,
	Opens:AtomicU64,
	Disposes:AtomicU64,
	Expressions:Mutex<Vec<String>>,
}

impl FakeBrowse {
	fn Answering(Value:Value) -> Arc<Self> {
		Arc::new(Self {
			Answer:Mutex::new(Ok(Value)),
			Opens:AtomicU64::new(0),
			Disposes:AtomicU64::new(0),
			Expressions:Mutex::new(Vec::new()),
		})
	}

	fn Failing(Why:BrowserError) -> Arc<Self> {
		Arc::new(Self {
			Answer:Mutex::new(Err(Why)),
			Opens:AtomicU64::new(0),
			Disposes:AtomicU64::new(0),
			Expressions:Mutex::new(Vec::new()),
		})
	}
}

impl BrowserBrowse for FakeBrowse {
	fn Open<'a>(&'a self, Url:&'a str, _Options:OpenOptions) -> FutureBox<'a, Result<Target, BrowserError>> {
		self.Opens.fetch_add(1, Ordering::SeqCst);
		let Answer = self.Answer.lock().unwrap().clone();
		Box::pin(async move {
			// A fake that is scripted to fail fails everywhere, so a test can
			// say "this browser is unreachable" once and mean it.
			if let Err(Why) = Answer {
				return Err(Why);
			}
			Ok(Target {
				Url:Url.to_string(),
				Ws:"ws://fake".to_string(),
				Ownership:AgenticBrowserCDP::Ownership::Launched,
			})
		})
	}

	fn Evaluate<'a>(
		&'a self,
		_Target:&'a Target,
		Expression:&'a str,
		_Call:Call,
	) -> FutureBox<'a, Result<Value, BrowserError>> {
		self.Expressions.lock().unwrap().push(Expression.to_string());
		let Answer = self.Answer.lock().unwrap().clone();
		Box::pin(async move { Answer })
	}

	fn Dispose<'a>(&'a self, _Target:Target) -> FutureBox<'a, ()> {
		self.Disposes.fetch_add(1, Ordering::SeqCst);
		Box::pin(async move {})
	}
}

# Rust workspace

The TypeScript toolkit in `packages/` rebuilt as seven crates, keeping the port
shapes and the failure classification the TypeScript suites already enforce. The
TypeScript packages stay as they are: they are the specification, not a predecessor
to delete.

The layout follows the house style of `github.com/PlayForm/Run`: one directory per
crate under `Crate/`, `Source/Library.rs` as the library root, `Fn/` for operations
and `Struct/` for data and contracts, PascalCase module names, and one `Target/` at
the workspace root.

---

## Crates

**`agentic-runtime`** - `Crate/Runtime/`, library `AgenticRuntime`. The execution
core: `Call`, `Cancellation`, `Clock`, `Pace`, the `Bound`/`Race` combinators, the
std `BlockOn`, and the policy engine (admission, leases with a fence, per-run
budgets, release that never fails). Dependencies: none. An optional `tokio` feature
adds a `Pace` implementation and is off by default.

**`agentic-skill-contract`** - `Crate/SkillContract/`. Corpus discovery, validation
and a stable digest. Dependencies: `agentic-runtime`, `serde`, `serde_json`, `sha2`.

**`agentic-chat-driver`** - `Crate/ChatDriver/`. `agentic.chat.ask` over a page it
declares itself. Dependencies: `agentic-runtime`, `serde_json`. No browser, no
executor.

**`agentic-browser-cdp`** - `Crate/BrowserCDP/`. The concrete substrate: attach
versus launch, bounded evaluation, disposal that closes only what it launched.
Dependencies: `agentic-runtime`, `serde_json`, `tokio`, `tokio-util`, `futures-util`,
`chromiumoxide`. The only crate that names `chromiumoxide`.

**`agentic-design-verification`** - `Crate/DesignVerification/`. Measurement and
evidence, holding no design of its own. Dependencies: `agentic-runtime`,
`serde_json`.

**`agentic-workflow`** - `Crate/Workflow/`. The bridge from the substrate to the two
consumer-owned ports, and one composition that uses four capabilities.
Dependencies: `agentic-runtime`, `agentic-chat-driver`,
`agentic-design-verification`, `agentic-skill-contract`, `agentic-browser-cdp`,
`serde_json`.

**`agentic`** - `Crate/Agentic/`. Wiring, the manifest table and the executable.
Dependencies: all six above, `clap`, `tokio`, `serde_json`.

---

## The two rules the dependency graph enforces

**A capability's required surface is runtime-neutral.** `agentic-chat-driver`,
`agentic-skill-contract` and `agentic-design-verification` do not name `tokio` in
their dependency lists, and nothing they pull in resolves it. Time arrives as a
`Call` (a deadline and a cancellation) and a `Pace` (a clock and a wait) supplied by
the composition root.

**No capability imports an adapter.** `agentic-chat-driver` declares its own `Page`
trait and `agentic-design-verification` its own `Browse`; `agentic-workflow` owns the
bindings that implement them over `agentic-browser-cdp`'s `Browse`. There is no
shared ports crate, and the translations are deliberately not factored into a
"universal browser request" type.

Both are checkable, and the checks are the first two commands below.

---

## Verification

```
cargo test --workspace                                   # 161 tests, no browser, no network
cargo run -p agentic -- --help                           # the composition root's commands
cargo tree -p agentic-chat-driver | grep -c tokio         # 0
cargo tree -p agentic-chat-driver | grep -c browser       # 0
cargo clippy --workspace -- -D warnings
cargo fmt --check
```

The tests are fake-based, as the TypeScript suites are: a page is a scripted
responder, a browser is a scripted launcher, and a deadline passes because the test
advanced a clock it owns. `cargo test` never starts Chromium and never opens a
socket.

`agentic verify --module <name>` is the manifest check: it compares the errors a
manifest names against the codes its crate's own type declares, so a manifest that
has drifted from its code fails rather than being read by a harness.
