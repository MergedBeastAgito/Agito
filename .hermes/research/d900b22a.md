		/** Reply count at `lastScrapedAt`. */
		replies: t.u32(),
		/** Repost count at `lastScrapedAt`. */

		/** Like count at `lastScrapedAt`. */

		/** View count at `lastScrapedAt`. */

		firstSeenAt: t.string(),

		lastScrapedAt: t.string(),
		/** The only field that decides whether the post may reach the site. */

		/** The list slugs as a JSON array, because their number is the person's. */
		lists: t.string(),

import { t } from "spacetimedb/server";
// This sub-package resolves imports the way the SpacetimeDB module bundler
// does - without an extension - which is why it is the one place in the
// repository that does not spell `.ts`.
import { spacetimedb } from "./Schema";

/**
 * Timeline archive, live view.
 *
 * The tables are rendered from `Maintain/Source/Model/Schema/Entities.ts` into
 * `Schema.ts`; this file holds only behaviour. Three rules survive from the file
 * store, and each one is a reducer's job rather than a column's:
 *
 * 1. Observations and events are append-only. A re-scrape updates the post's
 *    metrics in place but never rewrites history.
 * 2. `visibility` is the only field that decides what may reach the public site.
 * 3. Every derived row id is built from its content, so replaying the same
 *    scrape is idempotent instead of duplicating rows.
 *
 * There is exactly one reducer that writes a post - `observe` - because a
 * second write path is a second set of rules waiting to disagree with the first.
 * `store publish`, `store hide` and the backfill all go through it.
 */

export default spacetimedb;

export const init = spacetimedb.init((_ctx) => {
	// Nothing to seed: the archive arrives through `observe`.
});

export const onConnect = spacetimedb.clientConnected((_ctx) => {});
export const onDisconnect = spacetimedb.clientDisconnected((_ctx) => {});

/**
 * Write one post record and append the observation that produced it.
 *
 * The arguments are the record, not a page reading: a caller that has already
 * folded a reading into the archive record (the scrape path) and a caller that
 * is restoring a record it holds (the export path) send the same thing. That is
 * what keeps the two paths from growing different rules.
 *
 * `at` is the observation's own moment, never the wall clock, so replaying a
 * scrape rewrites the same row instead of appending a second one.
 */
export const observe = spacetimedb.reducer(
	{
		id: t.string(),
		url: t.string(),
		authorHandle: t.string(),
		authorName: t.string(),
		authorVerified: t.bool(),
		text: t.string(),
		postedAt: t.string().optional(),
		firstSeenAt: t.string(),
		lastScrapedAt: t.string(),
		/** shadow | public | hidden */
		visibility: t.string(),
		/** The list slugs as a JSON array. */
		lists: t.string(),
		replies: t.u32(),
		reposts: t.u32(),
		likes: t.u32(),
		views: t.u32(),
		/** bookmark | like | history */
		source: t.string(),
		at: t.string(),
		/** observed | published | hidden */
		action: t.string(),
	},
	(ctx, args) => {
		const existing = ctx.db.post.id.find(args.id);

		if (existing === null) {
			ctx.db.post.insert({
				id: args.id,
				url: args.url,
				author: args.authorHandle,
				text: args.text,
				postedAt: args.postedAt,
				replies: args.replies,
				reposts: args.reposts,
				likes: args.likes,
				views: args.views,
				firstSeenAt: args.firstSeenAt,
				lastScrapedAt: args.lastScrapedAt,
				visibility: args.visibility,
				lists: args.lists,
			});
		} else {
			// `firstSeenAt` is the archive's own fact and survives every fold.
			ctx.db.post.id.update({
				...existing,
				url: args.url,
				author: args.authorHandle,
				text: args.text,
				postedAt: args.postedAt,
				replies: args.replies,
				reposts: args.reposts,
				likes: args.likes,
				views: args.views,
				lastScrapedAt: args.lastScrapedAt,
				visibility: args.visibility,
				lists: args.lists,
			});
		}

		const known = ctx.db.author.handle.find(args.authorHandle);
		if (known === null) {
			ctx.db.author.insert({
				handle: args.authorHandle,
				name: args.authorName,
				verified: args.authorVerified,
				firstSeenAt: args.firstSeenAt,
				postCount: existing === null ? 1 : 0,
			});
		} else {
			ctx.db.author.handle.update({
				...known,
				name: args.authorName,
				verified: args.authorVerified,
				// A new post is a new count; re-observing one already held is not.
				postCount: existing === null ? known.postCount + 1 : known.postCount,
			});
		}

		// Append-only, keyed by content: replaying the same scrape writes nothing.
		const observationId = `${args.id}:${args.source}:${args.at}:${args.action}`;
		if (ctx.db.observation.id.find(observationId) === null) {
			ctx.db.observation.insert({
				id: observationId,
				at: args.at,
				source: args.source,
				postId: args.id,
				action: args.action,
			});
		}
	},
);

/** Append one event. The payload is opaque JSON text. */
export const record = spacetimedb.reducer(
	{
		kind: t.string(),
		subject: t.string(),
		payload: t.string(),
		at: t.string(),
	},
	(ctx, args) => {
		const id = `${args.at}:${args.kind}:${args.subject}`;
		if (ctx.db.event.id.find(id) !== null) return;
		ctx.db.event.insert({
			id,
			at: args.at,
			kind: args.kind,
			subject: args.subject,
			payload: args.payload,
		});
	},
);

/** Record which curated list a handle belongs to. The latest decision for a pair wins. */
export const classify = spacetimedb.reducer(
	{
		handle: t.string(),
		list: t.string(),
		by: t.string(),
		confidence: t.f32(),
		reason: t.string(),
		at: t.string(),
	},
	(ctx, args) => {
		const id = `${args.list}:${args.handle}`;
		const row = {
			id,
			list: args.list,
			handle: args.handle,
			by: args.by,
			confidence: args.confidence,
			reason: args.reason,
			at: args.at,
		};
		if (ctx.db.membership.id.find(id) === null) {
			ctx.db.membership.insert(row);
		} else {
			ctx.db.membership.id.update(row);
		}
	},
);

/** Attach one media item to a post, at the position the page showed it. */
export const attach = spacetimedb.reducer(
	{ postId: t.string(), kind: t.string(), url: t.string(), position: t.u32() },
	(ctx, args) => {
		const id = `${args.postId}:${args.url}`;
		const row = {
			id,
			postId: args.postId,
			kind: args.kind,
			url: args.url,
			position: args.position,
		};
		if (ctx.db.media.id.find(id) === null) ctx.db.media.insert(row);
		else ctx.db.media.id.update(row);
	},
);

Use one SpacetimeDB database with three layers: a typed operational model, a versioned capability/schema catalog, and data-driven authorization. The essential distinction is between the logical schema that agents may change through authorized reducers and the physical schema compiled into the published module.

SpacetimeDB does not turn a row inserted into a “table definition” table into a new native table. Physical table and reducer changes require publishing an updated module. Consequently, runtime schema modification needs either a bounded, interpreted logical-schema layer or an authorized build-and-publish controller—not a reducer executing arbitrary DDL.[spacetimedb](https://spacetimedb.com/docs/intro/faq)

My recommendation is a **hybrid**: native tables for operational facts and concurrency; an interpreted schema for extensible capability state; and an explicitly privileged deployment path for physical changes.

## 1. Execution and trust boundaries

The database owns:

- Declared capabilities, versions, input/output contracts, state classes, admission policies.
- Identities, roles, scoped grants, approvals, and audit history.
- Tasks, execution attempts, resource leases, commands, and results.
- Browser observations, research conversations and visible step traces, verification evidence, skill manifests, and persisted UI state.
- Your existing archive.

An external worker owns the actual CDP connection, browser process, web-chat interaction, measurement tooling, and skill execution. Reducers cannot perform network requests or filesystem operations. Procedures can perform external I/O, but their external effects do not share the atomicity of a reducer transaction. A local CDP worker remains the appropriate boundary for your stated capabilities.spacetimedb+1

Every exposed mutation authenticates using `ctx.sender`, not an identity supplied in arguments. SpacetimeDB provides caller identity; your module supplies the role and permission interpretation.[spacetimedb](https://spacetimedb.com/docs/intro/key-architecture)

All new operational tables should be private. Expose caller-authorized views for subscriptions. A client-side `WHERE` filter is not authorization: public tables are readable by clients independently of the UI’s query choices. Views can filter private rows and omit sensitive columns.[spacetimedb](https://spacetimedb.com/docs/tables/access-permissions)

## 2. Table conventions

The following is a concrete proposed model, not a claim that SpacetimeDB supplies these application tables.

Notation:

- `PK`: native primary key.
- `UQ`: native unique key where applicable.
- `IX`: nonunique index.
- `?`: optional.
- IDs: opaque strings, except authenticated principals use native `Identity`.
- Times: native `Timestamp`; durations: integer microseconds.
- Counters/fencing epochs: `u64`; measurements: `f64`; ordinals: `u32`.
- Enumerated statuses: declared algebraic types where fixed by the module, not unrestricted strings.
- `…Json`: bounded, validated canonical JSON, never executable code.
- References are logical foreign keys checked by reducers. Do not assume automatic SQL foreign-key enforcement; the transaction documentation explicitly places foreign-key-like relationships in application logic.[spacetimedb](https://spacetimedb.com/docs/databases/transactions-atomicity/)

For tuple keys, store the component columns and derive one collision-safe key using a canonical encoding. Do not introduce more unescaped `${a}:${b}` keys.

All operational rows carry `scopeId IX`. A scope is the local workspace/security boundary. All mutable current-state rows carry `revision:u64`, checked on updates. Immutable records instead carry their provenance and recording timestamp.

## Catalog invariants

1. The catalog describes its own tables and types. Bootstrap a finite set of metatypes, then describe `type_def`, `field_def`, `table_def`, and the remaining catalog using those types. References between definitions express the recursion.
2. An activated release is immutable. Changes create a new release.
3. Capability versions pin exact schema and policy versions. Running work does not silently adopt the latest contract.
4. Native definitions must match the deployed module. Generate the initial catalog manifest alongside `Schema.ts` from your declaration source, and verify a catalog/module digest during activation.
5. Describing a native table does not authorize altering its actual structure.
6. Policies use a small, versioned, bounded expression language: role checks, ownership, equality, membership, numeric bounds, status checks, and approval requirements. No `eval`, arbitrary TypeScript, SQL fragments, or unrestricted recursion.
7. `handlerKey` selects an installed implementation. Declaring a new operation does not create executable behavior.
8. Fail closed on unknown types, unsupported evaluator versions, dangling references, invalid transitions, or mismatched module hashes.
9. Logical uniqueness, references, indexes, and document updates commit together.
10. Definitions distinguish durable state, append-only evidence, derived projections, and leased resources. These are behavior contracts, not merely descriptive labels.

The interpreter and bootstrap trust rules necessarily remain code. Making permissions data-driven does not remove the need for a trusted enforcement mechanism.

## 7. Reducers and their enforcement

Every reducer begins with the same trusted checks: authenticated principal, active scope membership, permission evaluation, request-id collision detection, payload bounds, and reference validation.

| Reducer family | What it enforces |
| --- | --- |
| `bootstrap`, `create_scope`, `register_principal` | Trusted initial ownership; no first-client takeover; explicit principal activation. |
| `bind_role`, `revoke_role`, `alter_permission` | Grant authority evaluated against pre-change permissions; delegation limits; no accidental removal of the last recovery administrator; audit in the same transaction. |
| `request_approval`, `decide_approval` | Exact action/target/content hash, eligible approver, expiry, separation of duties where required. Consumption occurs atomically with admission of the approved action. |
| `create_schema_draft`, `put_definition`, `validate_schema` | Draft-only editing, structural/type validity, supported policies/handlers, compatible references, bounded logical indexing. |
| `activate_schema`, `activate_capability_version` | Expected head revision, complete validation, immutable activated definitions, installed implementation compatibility. |
| `propose_physical_change`, `authorize_publish`, `record_publish_result` | Exact plan/artifact approval and trusted publisher identity. These record deployment authorization; they do not execute DDL. |
| `put_logical_entity`, `delete_logical_entity` | Type validation, field-level authority, revision match, declared retention/reference rules, atomic lookup maintenance. Append-only classes reject updates/deletes. |
| `register_agent`, `heartbeat_agent`, `advertise_capability` | Principal owns the instance; advertised contract/build is admitted. Heartbeat is liveness evidence, not authorization. |
| `create_task`, `add_dependency`, `cancel_task` | Pinned operation/input contract, dependency acyclicity, allowed transition, cancellation authority. Cancellation fences active work. |
| `claim_task_and_resources` | Ready dependencies, attempt limit, compatible worker, all required leases acquired atomically; no partial task/browser claim. |
| `renew_lease`, `release_lease`, `revoke_lease` | Holder, epoch, instance, expiry, TTL bounds, and authorized revocation; immutable lease history. |
| `issue_command`, `dispatch_command`, `complete_command` | Valid task/browser leases, operation admission, exact approval where needed, ordered dispatch, matching generation and epoch, immutable completed outcome. |
| `record_browser_snapshot`, `record_browser_tabs` | Trusted reporting worker, current generation/lease where applicable, monotonic observation ordering, snapshot-scoped element references. |
| `create_research_thread`, `submit_research_turn`, `record_research_capture` | Thread ownership, current browser authority, immutable captures, consistent ordered steps and sources, explicit completeness. |
| `create_verification_spec`, `start_verification`, `record_measurement`, `finish_verification` | Immutable versioned specifications, exact build/snapshot provenance, complete check accounting; errors cannot become passes. |
| `publish_skill_version`, `start_skill_validation`, `record_validation_result`, `activate_skill` | Manifest validation, immutable version/digest, exact suite results, compatibility and activation policy. |
| `register_artifact`, `link_evidence` | Scope access, provenance, size/hash format, permitted storage references. A submitted hash is not proof the bytes were inspected. |
| `set_ui_state` | Caller’s own UI key, bounded value, expected revision. |
| `expire_work`, `retry_work`, `apply_retention` | Scheduled maintenance, fencing before retry, bounded batches, retention/legal-hold rules, protected evidence references. |

Each reducer’s writes commit together or roll back together. Therefore claim, attempt creation, lease changes, command admission, and audit insertion belong in the same transaction where their consistency depends on one another.[spacetimedb](https://spacetimedb.com/docs/databases/transactions-atomicity/)

A rejected reducer cannot also preserve a newly inserted denial audit row if it throws: that insert rolls back. If durable denials are required, record a structured denial and return successfully without applying the requested mutation, or use a separate trusted audit channel.[spacetimedb](https://spacetimedb.com/docs/databases/transactions-atomicity/)

## 8. How two agents cannot claim one browser

The exclusivity unit is the browser session, not an individual tab, unless you deliberately implement stronger context isolation. Tabs can share authentication, profile state, and browser-level behavior.

Use one persistent `resource_lease` row keyed by the browser resource ID. Never delete its epoch history by deleting and recreating the current lease row.

Conceptually:

```
claim_task_and_resources(taskId, browserId, agentInstanceId, requestId):

  authenticate sender and verify ownership of agentInstanceId
  evaluate task and browser admission policies
  now := database context timestamp

  read task lease and browser lease
  require each is released, revoked, or expiresAt <= now
  require task is runnable

  increment each lease epoch
  assign both leases to this agent and new task attempt
  set bounded server-derived expirations

  insert task_attempt
  set task.currentAttemptId and task.status
  insert lease_history, execution_event, request_receipt
  commit
```

The claim is one reducer transaction—not “read availability in a client, then write ownership.” SpacetimeDB’s transactional execution and key constraints provide the database-side foundation for this exclusion.[spacetimedb](https://spacetimedb.com/docs/databases/transactions-atomicity/)

Every renewal, command dispatch, and result commit verifies:

```
sender owns holderAgentId
attemptId matches
lease epoch matches
lease has not expired
resource generation matches
task attempt remains active
```

Important consequences:

- An expired lease cannot be resurrected by renewal.
- A new claim increments the epoch.
- A reconnect does not restore ownership automatically.
- Disconnect does not need to release ownership; expiry is the recovery mechanism.
- Timer cleanup is not a correctness prerequisite. Every use checks expiry.
- Revocation increments the epoch immediately.

## Database exclusion is not browser fencing

The lease prevents two valid owners in the database. It cannot stop a paused old process from waking up and sending CDP commands directly.

Require a single browser-owning gateway/worker to enforce generations, lease epochs, and command ordering. Agents must not possess independent unrestricted CDP connections. The gateway rejects stale tokens and drains or quarantines outstanding old-generation work before handing the browser to a new owner.

Even this cannot undo a click or form submission that already happened. Treat external effects as a recoverable command protocol:

1. Persist intent.
2. Admit and dispatch under a valid lease.
3. Execute outside the transaction.
4. Record the outcome.
5. Reconcile ambiguous outcomes before retrying.

Do not promise exactly-once browser effects. In particular, a crash after submitting a web-chat prompt but before recording its external turn ID must trigger reconciliation, not blind resubmission. Procedures likewise cannot make multiple transactions and external effects one atomic operation.spacetimedb+1

## 9. What happens to the existing archive

Keep the archive as a domain, not as the universal storage format for agent state.

| --- | --- |
| `post` | Preserve its columns, primary key, and archive meaning. Move authoritative storage behind private visibility and expose a public projection containing only `visibility = public`. |
| `author` | Preserve columns/key. Expose publicly only as your publication policy permits; a count of all archived posts may itself disclose nonpublic archive activity. |
| `media` | Preserve columns/key and explicit `position`. Public projection must join through the parent post’s publication decision. |
| `observation` | Preserve columns/key and append-only meaning. Make private by default: bookmark/like/history provenance should not automatically be public. |
| `event` | Preserve as an archive event log. Do not move browser commands, grants, task attempts, or research traces into it. Make private unless an explicitly safe projection is required. |
| `membership` | Preserve columns/key as the latest archive classification projection. Keep its policy separate from IAM role membership. Make private unless intentionally published. |

Thus all six survive structurally; none should be repurposed. Their exposure and reducer enforcement need changes. Whether visibility can be changed in place on your deployed version must be verified before migration; if necessary, migrate into private successor tables and replace public reads with compatible projections.

The critical existing issue is that `post.public: true` exposes hidden and shadow posts to direct clients. The `visibility` column currently governs application publication, not database read access. The same leak applies to media and observations. Public/private access is enforced at the table/view boundary.[spacetimedb](https://spacetimedb.com/docs/tables/access-permissions)

## Harden the existing reducers

Keep `observe` as the single exported archive post-write path, but correct these behaviors:

- Authenticate callers and separately authorize publication/hiding. A scrape worker must not gain publication authority merely by passing `visibility: "public"`.
- Validate the documented closed sets. Comments do not constrain `visibility`, `source`, `action`, `kind`, or `by`.
- Reject or explicitly merge stale readings. The current reducer can overwrite newer text, metrics, and author details with an older replay.
- Define idempotency as same request key plus same payload hash. Reusing an observation key with different content must not silently mutate the post.
- Define trusted backfill semantics for `firstSeenAt`. Preserving the stored value is consistent with the existing contract; finding an earlier historical observation is a separate reconciliation decision.
- Reject changes to an established post’s author, or update both authors’ counts atomically. The current code does neither.
- Make `classify` compare decision order rather than arrival order. Currently “latest” means the last reducer invocation, not greatest `at`.
- Validate `confidence` as finite and within `[0,1]`, and validate list existence.
- Make `attach` require its parent post and recognized media kind. Add explicit replace/remove semantics for attachments absent from a new complete scrape.
- Keep `record` for archive events only; validate event kinds and payload shapes.

The existing key formats can remain for compatibility. New APIs should use safer IDs and receipts rather than propagate delimiter-based key construction.

## 10. Runtime schema modification and physical deployment

For logical changes, authorized reducers can create a draft, validate it, and atomically switch `schema_head`. Runtime workers can discover the new contracts through subscriptions without reading TypeScript.

For physical changes, use this separate flow:

1. Store a proposed migration and exact target manifest.
2. Authorize it using current roles and an approval bound to the plan hash.
3. A trusted external controller generates declarations, builds the module, tests it, and publishes the exact approved artifact.
4. The controller records the resulting module digest and migration outcome.
5. Activate the corresponding catalog only after compatibility checks.

Compatible changes may migrate automatically; more complicated changes can require successor tables and incremental migration. Publishing an updated module—not editing catalog rows—is the documented mechanism for physical schema evolution.[spacetimedb](https://spacetimedb.com/docs/intro/faq)

There is a hard trust limit: the publisher can replace the authorization reducers themselves. Database roles can govern the cooperative deployment workflow, but they cannot constrain someone who independently controls the database owner credentials or host. Keep publisher credentials outside the agent’s authority; the database owner can bypass ordinary table visibility restrictions.[spacetimedb](https://spacetimedb.com/docs/1.12.0/databases)

Also, publication and the preceding authorization transaction are not one atomic operation. Use a deployment lock/quiescence protocol and explicit recovery states.

## 11. Subscriptions and local completeness

Expose indexed, caller-authorized views such as:

- `my_tasks`, `task_detail`, `task_events`.
- `available_browser_sessions`, `browser_tabs`, `latest_page_snapshot`.
- `research_threads`, `research_turn_captures`, `research_capture_steps`.
- `verification_runs`, `verification_measurements`.
- `skill_catalog`, `skill_validation_history`.
- `capability_catalog`, `schema_release_detail`, `effective_permissions`.
- `my_ui_state`.
- Public archive projections.

Views can be subscribed to and update when their underlying data changes. Procedural views need indexed access rather than unrestricted `.iter()` scans, which is why scope, identity, parent IDs, and lookup keys are indexed throughout this design.[spacetimedb](https://spacetimedb.com/docs/functions/views)

“Everything locally answerable” should mean that current state, declared contracts, authority, outcomes, and evidence provenance can be read from this database. It must not mean that a database row is the actual browser connection or that an old page snapshot is current truth. Every external observation carries capture time and generation.

## 12. What I would refuse to put in the database

- Live CDP sockets, process handles, JavaScript execution contexts, closures, or browser object instances.
- Plaintext cookies, bearer tokens, private identity keys, browser-profile secrets, or unrestricted credential-bearing endpoints. Store secret references.
- Arbitrary executable code disguised as policy, schema defaults, or skill manifests.
- Hidden model chain-of-thought. Store supplied answers and visible product traces.
- Unbounded screenshots, video, network dumps, duplicate DOMs, or large package binaries by default. Store bounded extracts and content-addressed artifact metadata.
- Unsupported “verified” flags without a pinned specification, environment, tool, and evidence.
- Browser exactly-once guarantees that the external system cannot provide.
- Parallel, contradictory sources of ownership such as `browser.claimedBy`, `agent.currentBrowser`, and `lease.owner`. The lease is authoritative; other fields are projections.
- Raw secrets or operational state inside the existing public `event.payload`.
